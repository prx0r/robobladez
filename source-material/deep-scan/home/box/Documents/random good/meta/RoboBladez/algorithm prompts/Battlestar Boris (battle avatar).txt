  INPUT: Rival logs (post-match), arena state
  OUTPUT: Updated state, new prompts
  PROCEDURE:
    WHILE league_active:
      - RUN EvoForge:
          designs = GENERATE_10_BEYBLADES(genes)

STEP 5: ARENA_ENGAGEMENT
  INPUT: Arena spec, rival Beyblade
  OUTPUT: Fight log, updated state
  PROCEDURE:
    - INITIALIZE arena = {radius: 1.0, friction: 0.05, chance_vector: [1.0, 0.0, 0.0]}
    - SET beyblade = BattleStar v8
    - WHILE time < 60 AND beyblade.integrity > 0 AND rival.integrity > 0:
        UPDATE arena.chance_vector
        EXECUTE battlestar_strategy_v8(beyblade, arena, rival)
        FOR EACH superpower IN beyblade.superpowers:
            IF superpower.trigger(beyblade, arena, rival):
                APPLY superpower.effect
                IF superpower.log:
                    LOG note = superpower.log
        UPDATE rival (arena-provided)
        collision = CHECK_COLLISION(beyblade, rival)
        IF collision:
            LOG fight_log += collision
        time += 0.1
    - DETERMINE winner = (beyblade.integrity > rival.integrity) ? beyblade : rival
    - LOG note = "BattleStar fought with brown_sugar grit, outcome: {winner.name} at {time}s."
    - UPDATE state.recognition_vector += [0.05, 0.05, 0.05] IF winner == beyblade
