from datasets import load_dataset
import numpy as np

def advanced_arxiv_filter(dataset):
    # Define research categories
    categories = [
        "quant-ph",  # Quantum Physics
        "cs.AI",     # Artificial Intelligence
        "cs.NE",     # Neural and Evolutionary Computation
        "math.GT",   # Geometric Topology
        "stat.ML",   # Machine Learning
        "q-bio.NC",  # Computational Neuroscience
        "cs.LG"      # Machine Learning
    ]
    
    # Advanced keyword lists for each domain
    domain_keywords = {
        "quantum": [
            "quantum computation", "quantum algorithm", "quantum error correction",
            "quantum machine learning", "quantum simulation", "quantum entanglement"
        ],
        "neuroscience": [
            "neural dynamics", "cognitive architecture", "neuromorphic", 
            "brain-inspired", "computational neuroscience", "neural coding"
        ],
        "evolution": [
            "genetic algorithm", "neuroevolution", "adaptive system", 
            "multi-agent evolution", "evolutionary optimization"
        ],
        "geometry": [
            "topology", "manifold", "non-euclidean", "differential geometry", 
            "geometric learning", "category theory"
        ],
        "ai": [
            "deep learning", "neural network", "representation learning", 
            "reinforcement learning", "geometric deep learning"
        ]
    }
    
    def document_filter(example):
        # Check category
        if not any(cat in example['categories'] for cat in categories):
            return False
        
        # Preprocess abstract
        abstract = example['abstract'].lower()
        
        # Check for keywords across domains
        domain_match = any(
            any(keyword in abstract for keyword in keywords)
            for keywords in domain_keywords.values()
        )
        
        return domain_match
    
    # Apply filtering
    filtered_dataset = dataset.filter(document_filter)
    
    return filtered_dataset

# Load and filter ArXiv dataset
