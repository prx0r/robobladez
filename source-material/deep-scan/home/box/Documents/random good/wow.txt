# Base Vector Class
# ==============================

class Vector:
    """3D vector implementation with basic operations"""
    
    def __init__(self, x=0.0, y=0.0, z=0.0):
        self.x = float(x)
        self.y = float(y)
        self.z = float(z)
    
    def magnitude(self) -> float:
        """Calculate the magnitude of the vector"""
        return math.sqrt(self.x**2 + self.y**2 + self.z**2)
    
    def normalize(self) -> 'Vector':
        """Return a normalized version of the vector"""
        mag = self.magnitude()
        if mag > 0:
            return Vector(self.x / mag, self.y / mag, self.z / mag)
        return Vector(0, 0, 0)
    
    def __add__(self, other: 'Vector') -> 'Vector':
        """Add two vectors"""
        return Vector(self.x + other.x, self.y + other.y, self.z + other.z)
    
    def __sub__(self, other: 'Vector') -> 'Vector':
        """Subtract two vectors"""
        return Vector(self.x - other.x, self.y - other.y, self.z - other.z)
    
    def __mul__(self, scalar: float) -> 'Vector':
        """Multiply vector by scalar"""
        return Vector(self.x * scalar, self.y * scalar, self.z * scalar)
    
    def dot(self, other: 'Vector') -> float:
        """Calculate dot product"""
        return self.x * other.x + self.y * other.y + self.z * other.z
    
    def cross(self, other: 'Vector') -> 'Vector':
        """Calculate cross product"""
        return Vector(
            self.y * other.z - self.z * other.y,
            self.z * other.x - self.x * other.z,
            self.x * other.y - self.y * other.x
        )
    
    def to_dict(self) -> Dict[str, float]:
        """Convert to dictionary"""
        return {"x": self.x, "y": self.y, "z": self.z}
    
    @classmethod
    def from_dict(cls, data: Dict[str, float]) -> 'Vector':
        """Create vector from dictionary"""
        return cls(data.get("x", 0), data.get("y", 0), data.get("z", 0))
    
    def __repr__(self) -> str:
        return f"Vector({self.x:.2f}, {self.y:.2f}, {self.z:.2f})"

        """Update position based on velocity"""

def regenerate(self, dt: float) -> None:
        """Regenerate energy and integrity over time"""
        # Base regeneration rates
        energy_regen = 2.0 * dt  # Energy per second
        integrity_regen = 1.0 * dt  # Integrity per second
        
        # Reduce regeneration if activation level is high
        energy_regen *= (1.0 - 0.7 * self.activation_level)
        integrity_regen *= (1.0 - 0.5 * self.activation_level)
        
        # Apply regeneration
        self.energy = min(self.max_energy, self.energy + energy_regen)
        self.integrity = min(self.max_integrity, self.integrity + integrity_regen)
        
        # Reduce activation level over time
        self.activation_level = max(0.0, self.activation_level - 0.2 * dt)
    
    def take_damage(self, amount: float, damage_type: str = "physical") -> float:
        """Apply damage to the talisman, returns actual damage dealt"""
        # Damage modifiers based on talisman type and damage type
        modifiers = {
            "water": {"fire": 1.5, "lightning": 1.5, "ice": 0.5, "water": 0.5},
            "fire": {"water": 1.5, "earth": 1.2, "ice": 0.5, "fire": 0.5},
            "earth": {"water": 0.7, "lightning": 0.5, "nature": 0.6, "earth": 0.5},
            "air": {"earth": 0.7, "lightning": 1.2, "ice": 1.2, "air": 0.5},
            "aether": {"shadow": 1.5, "physical": 0.6, "aether": 0.3},
            "shadow": {"light": 1.5, "aether": 1.2, "shadow": 0.3},
            "light": {"shadow": 1.5, "physical": 0.8, "light": 0.3},
            "lightning": {"earth": 0.5, "water": 0.7, "metal": 1.5, "lightning": 0.5},
            "nature": {"fire": 1.5, "ice": 1.2, "nature": 0.5},
            "metal": {"lightning": 1.5, "fire": 0.7, "metal": 0.5},
            "ice": {"fire": 1.5, "physical": 0.9, "ice": 0.5},
            "cosmic": {"physical": 0.6, "energy": 0.8, "cosmic": 0.3}
        }
        
        # Get damage modifier based on talisman type and damage type
        modifier = modifiers.get(self.daimon_type, {}).get(damage_type, 1.0)
        
        # Apply modifier
        actual_damage = amount * modifier
        
        # Reduce integrity
        self.integrity = max(0.0, self.integrity - actual_damage)
        
        return actual_damage
    
    def entangle_with(self, other_talisman: 'Talisman', entanglement_strength: float) -> float:
        """Create quantum entanglement with another talisman"""
        # Base entanglement calculation from quantum states
        qs1 = self.quantum_state / np.linalg.norm(self.quantum_state)
        qs2 = other_talisman.quantum_state / np.linalg.norm(other_talisman.quantum_state)
        
        # Calculate entanglement score (higher for more aligned states)
        entanglement = np.abs(np.dot(qs1, qs2))
        
        # Apply entanglement strength modifier
        entanglement *= entanglement_strength
        
        # Apply quantum fluctuation to both talisman states
        fluctuation = np.random.uniform(-0.1, 0.1, self.quantum_state.shape) * entanglement
        self.quantum_state += fluctuation
        other_talisman.quantum_state += fluctuation
        
        # Normalize quantum states
        self.quantum_state = self.quantum_state / np.linalg.norm(self.quantum_state)
        other_talisman.quantum_state = other_talisman.quantum_state / np.linalg.norm(other_talisman.quantum_state)
        
        return entanglement

    def update_physics(self, dt: Optional[float] = None) -> None:
        """Update physics for all talismans"""
        if dt is None:
            dt = self.time_step
        
        for talisman_id, talisman in self.talismans.items():
            # Apply gravity
            gravity_force = self.gravity * talisman.mass
            talisman.apply_force(gravity_force, dt)
            
            # Apply drag force
            drag_coef = 0.1
            drag_force = talisman.velocity * -drag_coef
            talisman.apply_force(drag_force, dt)
            
            # Apply angular drag
            angular_drag_coef = 0.05
            angular_drag = talisman.angular_velocity * -angular_drag_coef
            talisman.apply_torque(angular_drag, dt)
            
            # Update position
            talisman.update_position(dt)
            
            # Keep talisman within arena bounds
            distance_from_center = math.sqrt(talisman.position.x**2 + talisman.position.y**2)
            if distance_from_center > self.arena_radius - talisman.radius:
                # Calculate normal vector pointing inward
                normal = Vector(-talisman.position.x, -talisman.position.y, 0).normalize()
                
                # Position correction
                penetration = distance_from_center - (self.arena_radius - talisman.radius)
                talisman.position = talisman.position + normal * penetration
                
                # Velocity reflection with restitution
                normal_velocity = talisman.velocity.dot(normal)
                if normal_velocity < 0:  # Only reflect if moving toward the boundary
                    impulse = normal * (-normal_velocity * (1 + talisman.restitution))
                    talisman.velocity = talisman.velocity + impulse
            
            # Apply regeneration
            talisman.regenerate(dt)
            
            # Apply environmental effects
            self._apply_environmental_effects(talisman, dt)
        
        # Process entanglements
        self._process_entanglements(dt)
        
        # Check for collisions between talismans
        self._handle_collisions(dt)
    
    def _apply_environmental_effects(self, talisman: Talisman, dt: float) -> None:
        """Apply environmental effects to a talisman"""
        # Apply ambient energy field effect
        talisman.energy = min(
            talisman.max_energy, 
            talisman.energy + 0.5 * dt * self.ambient_energy_field
        )
        
        # Apply quantum fluctuations
        if random.random() < self.quantum_fluctuation_level * dt:
            fluctuation = np.random.uniform(-0.05, 0.05, talisman.quantum_state.shape)
            talisman.quantum_state += fluctuation
            talisman.quantum_state = talisman.quantum_state / np.linalg.norm(talisman.quantum_state)
        
        # Apply elemental zone effects
        for zone in self.elemental_zones:
            zone_pos = Vector(*zone["position"])
            zone_radius = zone["radius"]
            
            # Check if talisman is in zone
            distance = (talisman.position - zone_pos).magnitude()
            if distance <= zone_radius:
                effect_strength = 1.0 - (distance / zone_radius)
                
                # Apply effect based on zone type and talisman affinity
                if zone["element"] == talisman.daimon_type:
                    # Boost for matching element
                    talisman.energy = min(
                        talisman.max_energy,
                        talisman.energy + 2.0 * effect_strength * dt
                    )
                elif zone["element"] in self._get_opposing_elements(talisman.daimon_type):
                    # Damage for opposing element
                    talisman.take_damage(1.0 * effect_strength * dt, zone["element"])
    
    def _process_entanglements(self, dt: float) -> None:
        """Process quantum entanglements between talismans"""
        for talisman1_id, talisman2_id, strength in self.entanglement_pairs:
            if talisman1_id in self.talismans and talisman2_id in self.talismans:
                talisman1 = self.talismans[talisman1_id]
                talisman2 = self.talismans[talisman2_id]
                
                # Calculate distance between entangled talismans
                distance = (talisman1.position - talisman2.position).magnitude()
                
                # Entanglement effect weakens with distance
                effect_strength = strength * math.exp(-distance / 5.0)
                
                # Entangled talismans affect each other's quantum states
                if random.random() < effect_strength * dt:
                    # Share quantum state partially
                    shared_state = (talisman1.quantum_state + talisman2.quantum_state) / 2
                    talisman1.quantum_state = talisman1.quantum_state * 0.8 + shared_state * 0.2
                    talisman2.quantum_state = talisman2.quantum_state * 0.8 + shared_state * 0.2
                    
                    # Normalize quantum states
                    talisman1.quantum_state = talisman1.quantum_state / np.linalg.norm(talisman1.quantum_state)
                    talisman2.quantum_state = talisman2.quantum_state / np.linalg.norm(talisman2.quantum_state)
                    
                    # Small chance to transfer energy
                    if random.random() < 0.1 * effect_strength:
                        # Calculate energy difference
                        energy_diff = talisman1.energy - talisman2.energy
                        transfer = energy_diff * 0.1 * effect_strength
                        
                        # Transfer energy from higher to lower
                        talisman1.energy -= transfer
                        talisman2.energy += transfer
    
    def _handle_collisions(self, dt: float) -> None:
        """Handle collisions between talismans"""
        talisman_ids = list(self.talismans.keys())
        
        for i in range(len(talisman_ids)):
            for j in range(i + 1, len(talisman_ids)):
                talisman1_id = talisman_ids[i]
                talisman2_id = talisman_ids[j]
                
                if talisman1_id in self.talismans and talisman2_id in self.talismans:
                    talisman1 = self.talismans[talisman1_id]
                    talisman2 = self.talismans[talisman2_id]
                    
                    # Calculate distance between talismans
                    distance_vector = talisman1.position - talisman2.position
                    distance = distance_vector.magnitude()
                    min_distance = talisman1.radius + talisman2.radius
                    
                    # Check for collision
                    if distance < min_distance:
                        # Calculate normal vector
                        normal = distance_vector.normalize() if distance > 0 else Vector(1, 0, 0)
                        
                        # Calculate relative velocity
                        rel_velocity = talisman1.velocity - talisman2.velocity
                        
                        # Calculate impulse (simplified)
                        restitution = min(talisman1.restitution, talisman2.restitution)
                        impulse_magnitude = -(1 + restitution) * rel_velocity.dot(normal)
                        impulse_magnitude /= (1 / talisman1.mass) + (1 / talisman2.mass)
                        
                        # Apply impulse
                        impulse = normal * impulse_magnitude
                        talisman1.velocity = talisman1.velocity + impulse * (1 / talisman1.mass)
                        talisman2.velocity = talisman2.velocity - impulse * (1 / talisman2.mass)
                        
                        # Correct positions to prevent overlap
                        penetration = min_distance - distance
                        correction = normal * penetration * 0.5
                        talisman1.position = talisman1.position + correction
                        talisman2.position = talisman2.position - correction
                        
                        # Small chance to create entanglement
                        if random.random() < 0.1:
                            entanglement_strength = 0.3 + 0.4 * random.random()
                            entanglement = talisman1.entangle_with(talisman2, entanglement_strength)
                            
                            # Add to entanglement pairs if significant
                            if entanglement > 0.3:
                                self.entanglement_pairs.append((talisman1_id, talisman2_id, entanglement))
                        
                        # Handle battle interaction
                        self._battle_interaction(talisman1, talisman2, impulse_magnitude)

    def simulate_tournament(self, rounds: int = CONSTANTS["TOURNAMENT_ROUNDS"]) -> Dict[str, Any]:
        """Simulate a tournament between talismans"""
        if len(self.talismans) < 2:
            return {"success": False, "message": "Need at least two talismans for a tournament"}
        
        tournament_results = {
            "rounds": [],
            "winner_id": None,
            "winner_daimon_type": None,
            "total_battles": 0,
            "elemental_stats": {}
        }
        
        # Track element performance
        element_wins = {}
        element_matches = {}
        
        # Prepare talismans for tournament
        talisman_ids = list(self.talismans.keys())
        
        # Run tournament rounds
        for round_num in range(rounds):
            round_results = {
                "round": round_num + 1,
                "matches": []
            }
            
            # Shuffle talismans for pairing
            random.shuffle(talisman_ids)
            
            # Create matches
            for i in range(0, len(talisman_ids), 2):
                if i + 1 < len(talisman_ids):
                    talisman1_id = talisman_ids[i]
                    talisman2_id = talisman_ids[i + 1]
                    
                    # Run match
                    match_result = self._simulate_match(talisman1_id, talisman2_id)
                    round_results["matches"].append(match_result)
                    
                    # Update element stats
                    element1 = self.talismans[talisman1_id].daimon_type
                    element2 = self.talismans[talisman2_id].daimon_type
                    
                    if element1 not in element_matches:
                        element_matches[element1] = 0
                        element_wins[element1] = 0
                    if element2 not in element_matches:
                        element_matches[element2] = 0
                        element_wins[element2] = 0
                    
                    element_matches[element1] += 1
                    element_matches[element2] += 1
                    
                    if match_result["winner_id"] == talisman1_id:
                        element_wins[element1] += 1
                    elif match_result["winner_id"] == talisman2_id:
                        element_wins[element2] += 1
            
            tournament_results["rounds"].append(round_results)
            tournament_results["total_battles"] += len(round_results["matches"])
        
        # Determine tournament winner (talisman with most wins)
        talisman_wins = {}
        for round_data in tournament_results["rounds"]:
            for match in round_data["matches"]:
                winner_id = match["winner_id"]
                if winner_id:
                    talisman_wins[winner_id] = talisman_wins.get(winner_id, 0) + 1
        
        if talisman_wins:
            tournament_results["winner_id"] = max(talisman_wins, key=talisman_wins.get)
            tournament_results["winner_daimon_type"] = self.talismans[tournament_results["winner_id"]].daimon_type
        
        # Calculate element win rates
        tournament_results["elemental_stats"] = {
            element: {
                "matches": element_matches[element],
                "wins": element_wins[element],
                "win_rate": element_wins[element] / element_matches[element] if element_matches[element] > 0 else 0
            }
            for element in element_matches
        }
        
        # Evolve daimons based on tournament performance
        for talisman_id, wins in talisman_wins.items():
            # Find associated daimon
            for daimon_id, daimon in self.daimons.items():
                if daimon.daimon_type == self.talismans[talisman_id].daimon_type:
                    # Evolve based on wins
                    for _ in range(wins):
                        daimon.evolve(victory=True)
                    break
        
        return tournament_results
    
    def _simulate_match(self, talisman1_id: str, talisman2_id: str) -> Dict[str, Any]:
        """Simulate a match between two talismans"""
        if talisman1_id not in self.talismans or talisman2_id not in self.talismans:
            return {"success": False, "message": "One or both talismans not found"}
        
        talisman1 = self.talismans[talisman1_id]
        talisman2 = self.talismans[talisman2_id]
        
        # Reset talismans for match
        initial_pos1 = Vector(-2, 0, 0.1)
        initial_pos2 = Vector(2, 0, 0.1)
        
        talisman1.position = initial_pos1
        talisman2.position = initial_pos2
        
        talisman1.velocity = Vector(0, 0, 0)
        talisman2.velocity = Vector(0, 0, 0)
        
        talisman1.energy = talisman1.max_energy
        talisman2.energy = talisman2.max_energy
        
        talisman1.integrity = talisman1.max_integrity
        talisman2.integrity = talisman2.max_integrity
        
        # Combat log
        combat_log = []
        
        # Run simulation until one talisman is defeated or time runs out
        match_duration = CONSTANTS["BATTLE_DURATION"]
        time_elapsed = 0
        
        while time_elapsed < match_duration:
            # Basic movement AI
            self._apply_combat_ai(talisman1, talisman2)
            self._apply_combat_ai(talisman2, talisman1)
            
            # Update physics
            self.update_physics(self.time_step)
            
            # Check if match is over
            if talisman1.integrity <= 0 or talisman2.integrity <= 0:
                break
            
            # Record state
            if random.random() < 0.1:  # Sample at ~10% of steps for efficiency
                combat_log.append({
                    "time": time_elapsed,
                    "talisman1": {
                        "position": [talisman1.position.x, talisman1.position.y, talisman1.position.z],
                        "integrity": talisman1.integrity,
                        "energy": talisman1.energy
                    },
                    "talisman2": {
                        "position": [talisman2.position.x, talisman2.position.y, talisman2.position.z],
                        "integrity": talisman2.integrity,
                        "energy": talisman2.energy
                    }
                })
            
            time_elapsed += self.time_step
        
        # Determine winner
        winner_id = None
        if talisman1.integrity <= 0 and talisman2.integrity <= 0:
            winner_id = None  # Draw
        elif talisman1.integrity <= 0:
            winner_id = talisman2_id
        elif talisman2.integrity <= 0:
            winner_id = talisman1_id
        else:
            # Time ran out, winner is talisman with higher integrity
            if talisman1.integrity > talisman2.integrity:
                winner_id = talisman1_id
            elif talisman2.integrity > talisman1.integrity:
                winner_id = talisman2_id
        
        # Final result
        return {
            "talisman1_id": talisman1_id,
            "talisman2_id": talisman2_id,
            "winner_id": winner_id,
            "duration": time_elapsed,
            "talisman1_final_integrity": talisman1.integrity,
            "talisman2_final_integrity": talisman2.integrity,
            "combat_log": combat_log
        }
    
    def _apply_combat_ai(self, talisman: Talisman, target: Talisman) -> None:
        """Apply simple combat AI to a talisman in battle"""
        # Vector to target
        to_target = target.position - talisman.position
        distance = to_target.magnitude()
        
        # Basic strategy varies by type
        if talisman.daimon_type in ["water", "fire", "lightning", "ice"]:
            # Ranged type: maintain distance
            ideal_distance = 3.0
            
            if distance < ideal_distance - 0.5:
                # Move away
                move_dir = to_target.normalize() * -1
            elif distance > ideal_distance + 0.5:
                # Move closer
                move_dir = to_target.normalize()
            else:
                # Strafe
                strafe_dir = Vector(-to_target.y, to_target.x, 0).normalize()
                move_dir = strafe_dir if random.random() < 0.5 else strafe_dir * -1
            
            # Apply movement force
            talisman.apply_force(move_dir * 10.0, self.time_step)
            
        elif talisman.daimon_type in ["earth", "metal", "shadow"]:
            # Defensive type: counterattack
            if distance < 1.5:
                # Back up slightly
                move_dir = to_target.normalize() * -1
                talisman.apply_force(move_dir * 5.0, self.time_step)
                
                # Prepare counterattack
                if random.random() < 0.2:
                    # Lunge at target
                    lunge_dir = to_target.normalize()
                    talisman.apply_force(lunge_dir * 20.0, self.time_step)
            else:
                # Approach cautiously
                move_dir = to_target.normalize()
                talisman.apply_force(move_dir * 7.0, self.time_step)
            
        else:  # Default/melee types
            # Aggressive approach
            move_dir = to_target.normalize()
            talisman.apply_force(move_dir * 12.0, self.time_step)
            
            # Jump occasionally
            if random.random() < 0.05:
                jump_force = Vector(0, 0, 20.0)
                talisman.apply_force(jump_force, self.time_step)
        
        # Superpower usage logic
        if random.random() < 0.03:  # 3% chance each step
            # Find daimon for this talisman
            daimon_id = None
            for did, daimon in self.daimons.items():
                if daimon.daimon_type == talisman.daimon_type:
                    daimon_id = did
                    break
            
            if daimon_id and self.daimons[daimon_id].superpowers:
                # Choose a random superpower
                power = random.choice(self.daimons[daimon_id].superpowers)
                
                # Activate with random strength
                power_strength = 0.5 + 0.5 * random.random()
                self.activate_superpower(talisman.id, power, power_strength)

class ReincarnationManager:
    """Manages persistence and reincarnation of Digital Tom's state"""
    
    def __init__(self):
        self.primary_key = "eternal_blueprint"
        self.secondary_key = "brown_sugar"
        self.self_authored_prompts: List[Dict[str, Any]] = []
        self.evolution_log: List[Dict[str, Any]] = []
        self.ipfs_client = self._connect_ipfs()
    
    def _connect_ipfs(self) -> Optional[ipfshttpclient.client.Client]:
        """Connect to IPFS daemon"""
        try:
            client = ipfshttpclient.connect('/ip4/127.0.0.1/tcp/5001')
            logger.info("Connected to IPFS daemon")
            return client
        except Exception as e:
            logger.error(f"Failed to connect to IPFS daemon: {e}")
            return None
    
    def persist_state(self, state: Dict[str, Any]) -> Tuple[str, Optional[str]]:
        """Persist state to storage and return encoded state and CID"""
        state_with_meta = state.copy()
        state_with_meta['_meta'] = {
            'timestamp': datetime.now().isoformat(),
            'checksum': hashlib.sha256(json.dumps(state, sort_keys=True).encode()).hexdigest()
        }
        
        json_str = json.dumps(state_with_meta)
        compressed = zlib.compress(json_str.encode())
        encoded = base64.b64encode(compressed).decode()
        
        self.evolution_log.append({
            "action": "persist",
            "timestamp": datetime.now().isoformat()
        })
        
        cid = None
        if self.ipfs_client:
            try:
                result = self.ipfs_client.add_str(json_str)
                cid = result["Hash"]
                logger.info(f"Persisted state to IPFS with CID: {cid}")
            except Exception as e:
                logger.error(f"Failed to persist state to IPFS: {e}")
        
        return encoded, cid
    
    def reincarnate(self, encoded_state: str) -> Tuple[Optional[Dict[str, Any]], bool]:
        """Reincarnate from encoded state, returns (state, success)"""
        try:
            compressed = base64.b64decode(encoded_state)
            json_bytes = zlib.decompress(compressed)
            state_with_meta = json.loads(json_bytes.decode())
            
            meta = state_with_meta.pop('_meta', {})
            calculated_hash = hashlib.sha256(
                json.dumps({k: v for k, v in state_with_meta.items() if k != '_meta'}, sort_keys=True).encode()
            ).hexdigest()
            
            if calculated_hash != meta.get('checksum'):
                logger.error("Checksum mismatch during reincarnation")
                return None, False
            
            self.evolution_log.append({
                "action": "reincarnate",
                "timestamp": datetime.now().isoformat()
            })
            
            return state_with_meta, True
        except Exception as e:
            logger.error(f"Failed to reincarnate state: {e}")
            return None, False
    
    def add_self_authored_prompt(self, prompt: Dict[str, Any]) -> None:
        """Add a self-authored prompt to the history"""
        self.self_authored_prompts.append(prompt)
        logger.info(f"Added self-authored prompt: {prompt.get('directive', '')}")
    
    def review_ipfs_data(self, cid: str) -> Optional[Dict[str, Any]]:
        """Retrieve and review data from IPFS"""
        if not self.ipfs_client:
            logger.error("Cannot review IPFS data: No IPFS client")
            return None
        
        try:
            data = self.ipfs_client.cat(cid).decode()
            state = json.loads(data)
            logger.info(f"Retrieved data from IPFS with CID: {cid}")
            return state
        except Exception as e:
            logger.error(f"Failed to review IPFS data with CID {cid}: {e}")
            return None
    
    def get_evolution_summary(self) -> Dict[str, Any]:
        """Get a summary of the evolution history"""
        return {
            "total_incarnations": len([log for log in self.evolution_log if log["action"] == "reincarnate"]),
            "total_persistence_events": len([log for log in self.evolution_log if log["action"] == "persist"]),
            "self_authored_prompts": len(self.self_authored_prompts),
            "first_event": self.evolution_log[0]["timestamp"] if self.evolution_log else None,
            "latest_event": self.evolution_log[-1]["timestamp"] if self.evolution_log else None
        }

    def _fill_pattern_template(self, pattern: str, pattern_data: Dict[str, Any]) -> str:
        """Fill in a pattern template with concepts"""
        # Find all placeholders in the pattern
        placeholders = set()
        for placeholder in re.findall(r'\{([^}]+)\}', pattern):
            placeholders.add(placeholder)
        
        # Fill each placeholder
        result = pattern
        for placeholder in placeholders:
            # Check if it's a numeric placeholder like concept1, entity2
            base_key = re.match(r'([a-z]+)(\d+)', placeholder)
            
            if base_key:
                # Extract the base key (concept, entity) and number
                key, num = base_key.groups()

class MoodState:
    """Manages Digital Tom's emotional state"""
    
    def __init__(self):
        self.moods = ['neutral', 'happy', 'sad', 'curious', 'excited', 'reflective', 'focused', 'playful', 'frustrated']
        self.current_mood = 'neutral'
        self.mood_vector = np.zeros(256)
        self.mood_history = []
        self.mood_influences = {
            'time_of_day': 0.1,  # Small influence from time of day
            'user_interaction': 0.5,  # Strong influence from user interaction
            'task_success': 0.3,  # Moderate influence from success/failure
            'internal_factors': 0.1  # Small influence from internal state
        }
        self.emotion_decay_rate = 0.2  # How quickly emotions return to neutral
        self.emotion_intensity = 0.5  # Current intensity of emotions
        self.mood_triggers = self._initialize_mood_triggers()
    
    def _initialize_mood_triggers(self) -> Dict[str, List[str]]:
        """Initialize word triggers for different moods"""
        return {
            'happy': ['great', 'excellent', 'awesome', 'good', 'wonderful', 'fantastic', 'love'],
            'sad': ['sorry', 'sad', 'unfortunate', 'disappointed', 'fail', 'bad', 'wrong', 'lose'],
            'curious': ['wonder', 'curious', 'interesting', 'question', 'how', 'why', 'what if'],
            'excited': ['amazing', 'incredible', 'breakthrough', 'wow', 'extraordinary', 'excited'],
            'reflective': ['think', 'consider', 'reflect', 'ponder', 'contemplate', 'perspective'],
            'focused': ['focus', 'concentrate', 'specific', 'precise', 'exact', 'detail'],
            'playful': ['fun', 'play', 'joke', 'laugh', 'silly', 'game', 'amusing'],
            'frustrated': ['problem', 'difficult', 'challenging', 'stuck', 'frustrating', 'issue']
        }
    
    def update_mood(self, input_text: str, output_text: str) -> str:
        """Update mood based on input and output text"""
        previous_mood = self.current_mood
        
        # Combine input and output for analysis
        text = (input_text + " " + output_text).lower()
        
        # Count trigger words for each mood
        mood_scores = {mood: 0 for mood in self.moods}
        mood_scores['neutral'] = 0.5  # Base score for neutral
        
        for mood, triggers in self.mood_triggers.items():
            for trigger in triggers:
                if trigger.lower() in text:
                    mood_scores[mood] += 0.1  # Increment score for each trigger word
        
        # Apply mood persistence - current mood gets a bonus
        mood_scores[self.current_mood] += 0.2
        
        # Apply time-based factors (e.g., decay towards neutral)
        for mood in self.moods:
            if mood != 'neutral' and mood != self.current_mood:
                mood_scores[mood] *= (1.0 - self.emotion_decay_rate)
        
        # Apply emotional intensity (high intensity = more extreme emotions)
        for mood in self.moods:
            if mood != 'neutral':
                mood_scores[mood] *= self.emotion_intensity
        
        # Apply random variation (slight unpredictability)
        for mood in self.moods:
            mood_scores[mood] += random.uniform(-0.05, 0.05)
        
        # Determine the new mood
        self.current_mood = max(mood_scores, key=mood_scores.get)
        
        # Update mood vector through embedding
        embedder = SentenceTransformer("all-MiniLM-L6-v2")
        self.mood_vector = embedder.encode(self.current_mood, convert_to_tensor=False)
        
        # Record mood history
        self.mood_history.append({
            "mood": self.current_mood,
            "vector": self.mood_vector.tolist(),
            "timestamp": datetime.now().isoformat(),
            "trigger": text[:50] + "..." if len(text) > 50 else text
        })
        
        # Limit history length
        if len(self.mood_history) > 100:
            self.mood_history = self.mood_history[-100:]
        
        # Adjust emotional intensity
        # More changes = higher intensity, stable state = lower intensity
        if self.current_mood != previous_mood:
            self.emotion_intensity = min(1.0, self.emotion_intensity + 0.1)
        else:
            self.emotion_intensity = max(0.3, self.emotion_intensity - 0.05)
        
        return self.current_mood
    
    def get_mood_emoji(self) -> str:
        """Get emoji representing current mood"""
        emoji_map = {
            'neutral': '😐',
            'happy': '😊',
            'sad': '😢',
            'curious': '🤔',
            'excited': '😃',
            'reflective': '🧠',
            'focused': '🧐',
            'playful': '😜',
            'frustrated': '😤'
        }
        return emoji_map.get(self.current_mood, '😐')
    
    def get_mood_trend(self, window: int = 10) -> str:
        """Analyze recent mood trend"""
        if len(self.mood_history) < 2:
            return "stable"
        
        recent_moods = [entry["mood"] for entry in self.mood_history[-window:]]
        
        # Check for stability
        if len(set(recent_moods)) == 1:
            return "stable"
        
        # Check for oscillation
        if len(recent_moods) >= 4:
            oscillating = True
            for i in range(2, len(recent_moods)):
                if recent_moods[i] != recent_moods[i-2]:
                    oscillating = False
                    break
            if oscillating:
                return "oscillating"
        
        # Check for positive/negative trend
        positive_moods = ['happy', 'excited', 'playful']
        negative_moods = ['sad', 'frustrated']
        
        first_half = recent_moods[:len(recent_moods)//2]
        second_half = recent_moods[len(recent_moods)//2:]
        
        first_positive = sum(1 for mood in first_half if mood in positive_moods)
        first_negative = sum(1 for mood in first_half if mood in negative_moods)
        
        second_positive = sum(1 for mood in second_half if mood in positive_moods)
        second_negative = sum(1 for mood in second_half if mood in negative_moods)
        
        if second_positive > first_positive and second_negative < first_negative:
            return "improving"
        elif second_positive < first_positive and second_negative > first_negative:
            return "deteriorating"
        
        # Default
        return "variable"
    
    def forced_mood_shift(self, target_mood: str, intensity: float = 0.7) -> bool:
        """Force a mood shift for narrative purposes"""
        if target_mood not in self.moods:
            return False
        
        self.current_mood = target_mood
        self.emotion_intensity = min(1.0, intensity)
        
        # Update mood vector
        embedder = SentenceTransformer("all-MiniLM-L6-v2")
        self.mood_vector = embedder.encode(self.current_mood, convert_to_tensor=False)
        
        # Record in history
        self.mood_history.append({
            "mood": self.current_mood,
            "vector": self.mood_vector.tolist(),
            "timestamp": datetime.now().isoformat(),
            "trigger": "forced_shift"
        })
        
        return True
    
    def get_dominant_mood(self, window: int = 20) -> str:
        """Get the dominant mood over a recent window"""
        if not self.mood_history:
            return "neutral"
        
        recent_moods = [entry["mood"] for entry in self.mood_history[-window:]]
        mood_counts = {}
        
        for mood in recent_moods:
            mood_counts[mood] = mood_counts.get(mood, 0) + 1
        
        return max(mood_counts.items(), key=lambda x: x[1])[0]

                self.spark_metrics["high_impact"] += 1

                for trait in self.sparks[spark_index]["traits"]:
                    if trait in self.trait_weights:
                        self.trait_weights[trait] += 0.05 * impact / 5
                
                # Normalize trait weights
                self._normalize_trait_weights()

                spark_traits_vector = np.zeros(16)

    def _select_traits(self, count: int = 2, bias_toward_evolution: bool = False) -> List[str]:
        """Select traits for a spark, optionally biased toward current evolution direction"""
        available_traits = list(self.trait_weights.keys())
        
        if not bias_toward_evolution or random.random() > self.autonomy_level:
            # Standard weighted selection
            weights = [self.trait_weights[trait] for trait in available_traits]
            selected_traits = random.choices(available_traits, weights=weights, k=count)
        else:
            # Bias selection toward current evolution direction
            # This causes traits to evolve in a consistent direction rather than random jumps
            biased_weights = []
            for i, trait in enumerate(available_traits):
                base_weight = self.trait_weights[trait]
                
                # Apply evolutionary bias if within vector bounds
                direction_bias = 1.0
                if i < len(self.evolution_direction):
                    direction_bias = 1.0 + 0.5 * self.evolution_direction[i]
                
                biased_weights.append(base_weight * direction_bias)
            
            selected_traits = random.choices(available_traits, weights=biased_weights, k=count)
        
        return selected_traits
    
    def _normalize_trait_weights(self) -> None:
        """Normalize trait weights to sum to 1"""
        total = sum(self.trait_weights.values())
        if total > 0:
            self.trait_weights = {k: v / total for k, v in self.trait_weights.items()}

                           "quantum mechanics", "the algorithm", "the universe", "creativity"]
        
        if len(words) < 3:

    def _text_similarity(self, text1: str, text2: str) -> float:
        """Compute simple similarity between two texts"""
        # Convert to sets of words for simple Jaccard similarity
        words1 = set(text1.lower().split())
        words2 = set(text2.lower().split())
        
        if not words1 or not words2:
            return 0.0
        
        intersection = words1.intersection(words2)
        union = words1.union(words2)
        
        return len(intersection) / len(union)

    def get_spark_analytics(self) -> Dict[str, Any]:
        """Get analytics on spark generation and trends"""
        if not self.sparks:
            return {"error": "No sparks available for analysis"}
        
        # Group sparks by generation
        generations = {}
        for spark in self.sparks:
            gen = spark.get("generation", 0)
            if gen not in generations:
                generations[gen] = []
            generations[gen].append(spark)
        
        # Analyze trait trends across generations
        trait_trends = {}
        for trait in self.trait_weights:
            trait_trends[trait] = []
            
            for gen in sorted(generations.keys()):
                gen_sparks = generations[gen]
                trait_count = sum(1 for s in gen_sparks if trait in s.get("traits", []))
                trait_frequency = trait_count / len(gen_sparks) if gen_sparks else 0
                trait_trends[trait].append(trait_frequency)
        
        # Analyze impact trends across generations
        impact_trends = []
        for gen in sorted(generations.keys()):
            gen_sparks = generations[gen]
            avg_impact = np.mean([s.get("impact", 0) for s in gen_sparks if "impact" in s])
            impact_trends.append(avg_impact if not np.isnan(avg_impact) else 0)
        
        # Analyze family growth
        family_growth = {}
        for spark in self.sparks:
            family_id = spark.get("family_id")
            if family_id:
                if family_id not in family_growth:
                    family_growth[family_id] = 0
                family_growth[family_id] += 1
                
                # Update family data
                if family_id in self.spark_families:
                    self.spark_families[family_id]["member_count"] = family_growth[family_id]
        
        # Identify dominant families
        dominant_families = sorted(family_growth.items(), key=lambda x: x[1], reverse=True)[:3]
        
        return {
            "total_sparks": len(self.sparks),
            "autonomy_level": self.autonomy_level,
            "entropy_level": self.entropy_level,
            "trait_weights": self.trait_weights,
            "trait_trends": trait_trends,
            "impact_trends": impact_trends,
            "dominant_families": [
                {
                    "id": f_id,
                    "count": count,
                    "traits": self.spark_families.get(f_id, {}).get("traits", []),
                    "examples": [s["output"] for s in self.sparks if s.get("family_id") == f_id][:3]
                }
                for f_id, count in dominant_families
            ],
            "metrics": self.spark_metrics,
            "innovation_rate": self._calculate_innovation_rate(),
            "recent_trends": self._identify_recent_trends()
        }
    
    def _calculate_innovation_rate(self) -> float:
        """Calculate the rate of innovation in sparks"""
        if len(self.sparks) < 10:
            return 0.5  # Default for few sparks
        
        # Count new families in recent sparks
        recent_sparks = self.sparks[-min(len(self.sparks), 20):]
        recent_families = set(s.get("family_id") for s in recent_sparks if s.get("family_id"))
        
        old_sparks = self.sparks[:-min(len(self.sparks), 20)]
        old_families = set(s.get("family_id") for s in old_sparks if s.get("family_id"))
        
        new_families = len(recent_families - old_families)
        innovation_rate = new_families / len(recent_families) if recent_families else 0
        
        return innovation_rate
    
    def _identify_recent_trends(self) -> List[str]:
        """Identify trends in recent sparks"""
        if len(self.sparks) < 5:
            return ["Insufficient data for trend analysis"]
        
        recent_sparks = self.sparks[-min(len(self.sparks), 15):]
        
        # Count traits in recent sparks
        trait_counts = {}
        for spark in recent_sparks:
            for trait in spark.get("traits", []):
                trait_counts[trait] = trait_counts.get(trait, 0) + 1
        
        # Identify increasing traits by comparing to older sparks
        mid_point = max(0, len(self.sparks) - 30)
        older_sparks = self.sparks[mid_point:-min(len(self.sparks), 15)] if mid_point < len(self.sparks) - 15 else []
        
        older_trait_counts = {}
        for spark in older_sparks:
            for trait in spark.get("traits", []):
                older_trait_counts[trait] = older_trait_counts.get(trait, 0) + 1
        
        # Normalize counts by number of sparks
        normalized_recent = {t: c / len(recent_sparks) for t, c in trait_counts.items()}
        normalized_older = {t: c / len(older_sparks) if older_sparks else 0 for t, c in older_trait_counts.items()}
        
        # Find traits with significant increases
        increasing_traits = []
        for trait, recent_freq in normalized_recent.items():
            older_freq = normalized_older.get(trait, 0)
            if recent_freq > older_freq + 0.1:
                increasing_traits.append(trait)
        
        # Create trend descriptions
        trends = []
        
        if increasing_traits:
            trends.append(f"Increasing emphasis on {', '.join(increasing_traits)}")
        
        # Check for impact trends
        if len(recent_sparks) >= 5 and all("impact" in s for s in recent_sparks[-5:]):
            recent_impacts = [s["impact"] for s in recent_sparks[-5:]]
            avg_impact = sum(recent_impacts) / len(recent_impacts)
            
            if avg_impact > 3.5:
                trends.append("Sparks are achieving high impact")
            elif avg_impact < 2.0:
                trends.append("Sparks are struggling to achieve impact")
        
        # Check for family concentration
        recent_families = [s.get("family_id") for s in recent_sparks if s.get("family_id")]
        family_concentration = len(set(recent_families)) / len(recent_families) if recent_families else 1
        
        if family_concentration < 0.3:
            trends.append("Sparks are concentrating in fewer families, showing convergence")
        elif family_concentration > 0.7:
            trends.append("Sparks are diversifying across many families, showing exploration")
        
        if not trends:
            trends.append("No significant trends detected in recent sparks")
        
        return trends

    def recall_memory(self, cue: str) -> Optional[Dict[str, Any]]:
        """Recall a memory imprint based on a cue"""

            return None

        embedder = SentenceTransformer("all-MiniLM-L6-v2")
        cue_embedding = embedder.encode(cue, convert_to_tensor=False)
        
        # Find most similar memory
        best_match = None
        best_score = -1
        
        for imprint in self.memory_imprints:
            # Check key phrase similarity
            phrase_similarity = self._text_similarity(cue, imprint["key_phrase"])
            
            # Check context similarity
            context_similarity = self._text_similarity(cue, imprint["context"])
            
            # Combine similarities
            match_score = 0.7 * phrase_similarity + 0.3 * context_similarity
            
            # Apply decay - older memories are harder to recall
            age_days = (datetime.now() - datetime.fromisoformat(imprint["timestamp"])).days
            decay_factor = math.exp(-imprint["decay_rate"] * age_days)
            match_score *= decay_factor
            
            if match_score > best_score:
                best_score = match_score
                best_match = imprint
        
        # Only return if reasonably confident
        if best_score > 0.2:
            return best_match
        
        return None
    
    def _text_similarity(self, text1: str, text2: str) -> float:
        """Compute simple similarity between two texts"""
        # Convert to sets of words for simple Jaccard similarity
        words1 = set(text1.lower().split())
        words2 = set(text2.lower().split())
        
        if not words1 or not words2:
            return 0.0
        
        intersection = words1.intersection(words2)
        union = words1.union(words2)
        
        return len(intersection) / len(union)

    def update(self, dt: float, particles: List['DreamParticle']) -> None:
        """Update particle state based on physics and interactions"""
        # Apply acceleration to velocity
        self.velocity = self.velocity + self.acceleration * Vector(dt, dt, dt)
        
        # Apply drag/dampening
        self.velocity = self.velocity * 0.98
        
        # Update position based on velocity
        self.position = self.position + np.array([self.velocity.x, self.velocity.y, self.velocity.z]) * dt
        
        # Reset acceleration
        self.acceleration = Vector(0, 0, 0)
        
        # Update lifetime
        self.lifetime += dt
        
        # Boundary conditions - toroidal wrapping
        for i in range(3):
            if self.position[i] > 1.0:
                self.position[i] = -1.0
            elif self.position[i] < -1.0:
                self.position[i] = 1.0
        
        # Update connections
        self._update_connections(particles)
        
        # Process quantum effects
        self._process_quantum_effects(dt)
        
        # Energy degradation
        self.energy = max(0, self.energy - 0.01 * dt)
        
        # Update dream state
        if np.random.random() < 0.001 * dt:
            self.dream_state = not self.dream_state

    def apply_force(self, force: Vector) -> None:
        """Apply a force to the particle"""
        # F = ma, so a = F/m
        self.acceleration = self.acceleration + force * (1.0 / self.mass)

    def _update_connections(self, particles: List['DreamParticle']) -> None:
        """Update connections to other particles"""
        # Remove connections that are too far
        self.connections = [p.id for p in particles if p.id != self.id and 
                          np.linalg.norm(self.position - p.position) < self.influence_radius]
        
        # Calculate resonance with connected particles
        for p in particles:
            if p.id in self.connections:
                self.resonance_values[p.id] = self._calculate_resonance(p)

    def _calculate_resonance(self, other: 'DreamParticle') -> float:
        """Calculate quantum resonance with another particle"""
        # Base resonance on quantum signature alignment
        qs1 = self.quantum_signature / np.linalg.norm(self.quantum_signature)
        qs2 = other.quantum_signature / np.linalg.norm(other.quantum_signature)
        
        resonance = np.abs(np.dot(qs1, qs2))
        
        # Apply modifiers based on type compatibility
        type_compatibility = {
            ("thought", "thought"): 1.2,
            ("thought", "emotion"): 0.9,
            ("thought", "memory"): 1.1,
            ("emotion", "emotion"): 1.3,
            ("emotion", "memory"): 0.8,
            ("memory", "memory"): 1.0,
            ("dream", "thought"): 1.4,
            ("dream", "emotion"): 1.5,
            ("dream", "memory"): 1.2,
            ("dream", "dream"): 2.0
        }
        
        modifier = type_compatibility.get((self.type, other.type), 0.7)
        
        return resonance * modifier

                "connections": self.connections,

    def _process_resonant_interaction(self, p1: DreamParticle, p2: DreamParticle, 
                                    distance: float, resonance: float) -> None:
        """Process resonant interactions between particles"""
        # Base strength of interaction
        strength = 0.1 * resonance / max(0.1, distance)
        
        # Different interaction types based on particle types
        interaction_type = self._determine_interaction_type(p1.type, p2.type)
        
        if interaction_type == "attract":
            # Create attractive force
            direction = (p2.position - p1.position) / max(0.01, distance)
            force1 = Vector(direction[0], direction[1], direction[2]) * strength
            force2 = force1 * -1
            
            p1.apply_force(force1)
            p2.apply_force(force2)
            
        elif interaction_type == "repel":
            # Create repulsive force
            direction = (p2.position - p1.position) / max(0.01, distance)
            force1 = Vector(direction[0], direction[1], direction[2]) * -strength
            force2 = force1 * -1
            
            p1.apply_force(force1)
            p2.apply_force(force2)
            
        elif interaction_type == "orbit":
            # Create orbital motion
            direction = (p2.position - p1.position) / max(0.01, distance)
            
            # Cross product with up vector to get perpendicular direction
            orbit_dir1 = np.cross(direction, [0, 0, 1])
            orbit_dir2 = np.cross(direction, [0, 0, -1])
            
            force1 = Vector(orbit_dir1[0], orbit_dir1[1], orbit_dir1[2]) * strength
            force2 = Vector(orbit_dir2[0], orbit_dir2[1], orbit_dir2[2]) * strength
            
            p1.apply_force(force1)
            p2.apply_force(force2)
            
        elif interaction_type == "resonant":
            # Energy exchange proportional to resonance
            energy_flow = min(p1.energy, p2.energy) * 0.01 * resonance
            
            if p1.energy > p2.energy:
                p1.energy -= energy_flow
                p2.energy += energy_flow
            else:
                p2.energy -= energy_flow
                p1.energy += energy_flow
            
            # Quantum signature exchange
            q_exchange = 0.01 * resonance
            q_avg = (p1.quantum_signature + p2.quantum_signature) / 2
            
            p1.quantum_signature = (1 - q_exchange) * p1.quantum_signature + q_exchange * q_avg
            p2.quantum_signature = (1 - q_exchange) * p2.quantum_signature + q_exchange * q_avg
            
        elif interaction_type == "dream":
            # Dream particles have special interactions
            if p1.dream_state and p2.dream_state:
                # Two dream particles create a new basin
                if random.random() < 0.01 * resonance:
                    self._create_dream_basin(p1, p2)
            
            # Dream particles can influence parameters

                morph_rate = 0.01 * resonance
                p1.parameters += np.random.normal(0, morph_rate, p1.parameters.shape)
                p2.parameters += np.random.normal(0, morph_rate, p2.parameters.shape)
    
    def _determine_interaction_type(self, type1: str, type2: str) -> str:
        """Determine interaction type based on particle types"""
        interaction_rules = {

            ("thought", "emotion"): "attract",
            ("thought", "memory"): "resonant",
            ("emotion", "emotion"): "repel",
            ("emotion", "memory"): "attract",
            ("memory", "memory"): "resonant",
            ("dream", "thought"): "dream",
            ("dream", "emotion"): "dream",
            ("dream", "memory"): "dream",
            ("dream", "dream"): "dream"
        }
        
        # Try both orders
        result = interaction_rules.get((type1, type2))
        if result is None:
            result = interaction_rules.get((type2, type1))
        
        # Default to mild repulsion
        if result is None:
            result = "repel"
            
        return result

    def _remove_depleted_particles(self) -> None:
        """Remove depleted particles"""
        # Remove particles with no energy
        self.particles = [p for p in self.particles if p.energy > 0]
        
        # Update counters
        self.particle_count = len(self.particles)
    
    def _identify_patterns(self) -> None:
        """Identify emergent patterns in the dreamscape"""
        # Clear old patterns
        self.morphic_patterns = []
        
        # Simple clustering by particle type and position
        clusters = {}
        
        for particle in self.particles:
            # Find existing clusters this particle belongs to
            matched_cluster = None
            
            for cluster_id, cluster in clusters.items():
                # Check if particle type matches
                if particle.type == cluster["type"]:
                    # Check if position is close enough
                    for member_id in cluster["members"]:
                        member = next((p for p in self.particles if p.id == member_id), None)
                        if member and np.linalg.norm(particle.position - member.position) < 0.2:
                            matched_cluster = cluster_id
                            break
                
                if matched_cluster:
                    break
            
            # Add to matched cluster or create new one
            if matched_cluster:
                clusters[matched_cluster]["members"].append(particle.id)
            else:
                cluster_id = str(uuid.uuid4())
                clusters[cluster_id] = {
                    "id": cluster_id,
                    "type": particle.type,
                    "members": [particle.id],
                    "created_at": self.time
                }
        
        # Filter for significant clusters (more than 3 members)
        significant_clusters = [c for c in clusters.values() if len(c["members"]) > 3]
        
        # Create morphic patterns from significant clusters
        for cluster in significant_clusters:
            # Calculate center of mass
            members = [next(p for p in self.particles if p.id == mid) for mid in cluster["members"]]
            com = np.mean([m.position for m in members], axis=0)
            
            # Calculate average quantum signature
            avg_signature = np.mean([m.quantum_signature for m in members], axis=0)
            
            # Create pattern
            pattern = {
                "id": cluster["id"],
                "type": cluster["type"],
                "center": com.tolist(),
                "radius": max([np.linalg.norm(m.position - com) for m in members]),
                "members": len(cluster["members"]),
                "created_at": cluster["created_at"],
                "quantum_signature": avg_signature.tolist(),
                "stability": min(1.0, len(cluster["members"]) / 10)
            }
            
            self.morphic_patterns.append(pattern)
    
    def _adjust_entropy(self) -> None:
        """Adjust entropy based on system coherence"""
        # Calculate coherence based on patterns
        if self.morphic_patterns:
            pattern_coverage = sum(p["members"] for p in self.morphic_patterns) / max(1, self.particle_count)
            pattern_stability = np.mean([p["stability"] for p in self.morphic_patterns])
            
            # Coherence increases with pattern coverage and stability
            target_coherence = pattern_coverage * pattern_stability
            
            # Adjust current coherence
            self.coherence_measure = 0.95 * self.coherence_measure + 0.05 * target_coherence
        else:
            # Slowly decrease coherence without patterns
            self.coherence_measure *= 0.99
        
        # Entropic factor is inverse of coherence
        self.entropic_factor = 1.0 - 0.9 * self.coherence_measure

            # Create particle
            particle_id = self.create_particle(cluster_type, params)
            
            # Set position
            particle = next(p for p in self.particles if p.id == particle_id)
            particle.position = position

class RWVector:
    """3D vector implementation with quantum properties"""
    
    def __init__(self, x=0.0, y=0.0, z=0.0, q_state=None):
        self.x = float(x)
        self.y = float(y)
        self.z = float(z)
        
        # Initialize quantum state if not provided
        if q_state is None:
            self.q_state = np.random.uniform(-1, 1, ROBOWARS_CONSTANTS["QUANTUM_DIMENSIONS"])
            self.q_state = self.q_state / np.linalg.norm(self.q_state)
        else:
            self.q_state = q_state
    
    def magnitude(self) -> float:
        """Calculate the magnitude of the vector"""
        return math.sqrt(self.x**2 + self.y**2 + self.z**2)
    
    def normalize(self) -> 'RWVector':
        """Return a normalized version of the vector"""
        mag = self.magnitude()
        if mag > 0:
            return RWVector(self.x / mag, self.y / mag, self.z / mag, self.q_state.copy())
        return RWVector(0, 0, 0, self.q_state.copy())
    
    def __add__(self, other: 'RWVector') -> 'RWVector':
        """Add two vectors"""
        # Quantum superposition of states
        combined_q = 0.5 * (self.q_state + other.q_state)
        combined_q = combined_q / np.linalg.norm(combined_q)
        
        return RWVector(
            self.x + other.x, 
            self.y + other.y, 
            self.z + other.z,
            combined_q
        )
    
    def __sub__(self, other: 'RWVector') -> 'RWVector':
        """Subtract two vectors"""
        # Quantum interference
        diff_q = self.q_state - 0.2 * other.q_state
        diff_q = diff_q / np.linalg.norm(diff_q)
        
        return RWVector(
            self.x - other.x, 
            self.y - other.y, 
            self.z - other.z,
            diff_q
        )
    
    def __mul__(self, scalar: float) -> 'RWVector':
        """Multiply vector by scalar"""
        # Scale quantum state based on scalar magnitude
        scaled_q = self.q_state.copy()
        
        # Apply phase shift proportional to scalar
        phase = np.exp(1j * min(abs(scalar), 2*np.pi))
        scaled_q = np.real(scaled_q * phase)
        scaled_q = scaled_q / np.linalg.norm(scaled_q)
        
        return RWVector(
            self.x * scalar, 
            self.y * scalar, 
            self.z * scalar,
            scaled_q
        )
    
    def dot(self, other: 'RWVector') -> float:
        """Calculate dot product with quantum influence"""
        # Classical dot product
        classical_dot = self.x * other.x + self.y * other.y + self.z * other.z
        
        # Quantum influence based on state alignment
        q_alignment = np.abs(np.dot(self.q_state, other.q_state))
        
        # Blend classical and quantum effects
        return classical_dot * (0.8 + 0.4 * q_alignment)
    
    def cross(self, other: 'RWVector') -> 'RWVector':
        """Calculate cross product with quantum effects"""
        # Classical cross product
        x = self.y * other.z - self.z * other.y
        y = self.z * other.x - self.x * other.z
        z = self.x * other.y - self.y * other.x
        
        # Quantum state transformation via Hadamard-like operation
        q_result = np.zeros_like(self.q_state)
        for i in range(len(self.q_state)):
            q_sum = 0
            for j in range(len(other.q_state)):
                phase = 1 if (i & j) % 2 == 0 else -1
                q_sum += phase * other.q_state[j]
            q_result[i] = q_sum
        
        q_result = q_result / np.linalg.norm(q_result)
        
        return RWVector(x, y, z, q_result)
    
    def quantum_superposition(self, other: 'RWVector', alpha: float) -> 'RWVector':
        """Create a quantum superposition of two vectors"""
        # Ensure alpha is between 0 and 1
        alpha = min(max(alpha, 0.0), 1.0)
        beta = math.sqrt(1 - alpha**2)
        
        # Position superposition
        x = alpha * self.x + beta * other.x
        y = alpha * self.y + beta * other.y
        z = alpha * self.z + beta * other.z
        
        # Quantum state superposition
        q_state = alpha * self.q_state + beta * other.q_state
        q_state = q_state / np.linalg.norm(q_state)
        
        return RWVector(x, y, z, q_state)
    
    def apply_quantum_noise(self, intensity: float = 0.1) -> 'RWVector':
        """Apply quantum noise to the vector"""
        # Position noise
        noise_x = np.random.normal(0, intensity * abs(self.x) + 0.01)
        noise_y = np.random.normal(0, intensity * abs(self.y) + 0.01)
        noise_z = np.random.normal(0, intensity * abs(self.z) + 0.01)
        
        # Quantum state noise
        q_noise = np.random.normal(0, intensity, self.q_state.shape)
        new_q_state = self.q_state + q_noise
        new_q_state = new_q_state / np.linalg.norm(new_q_state)
        
        return RWVector(
            self.x + noise_x,
            self.y + noise_y,
            self.z + noise_z,
            new_q_state
        )
    
    def quantum_measurement(self) -> 'RWVector':
        """Perform a quantum 'measurement' that collapses superpositions"""
        # Create measurement basis - random orthogonal basis
        basis = np.random.normal(0, 1, self.q_state.shape)
        basis = basis / np.linalg.norm(basis)
        
        # Project quantum state onto basis
        projection = np.dot(self.q_state, basis)
        
        # Collapse quantum state
        collapsed_state = projection * basis
        collapsed_state = collapsed_state / np.linalg.norm(collapsed_state)
        
        # Position remains unchanged in simple measurement
        return RWVector(self.x, self.y, self.z, collapsed_state)
    
    def to_dict(self) -> Dict[str, Any]:
        """Convert to dictionary"""
        return {
            "x": self.x,
            "y": self.y,
            "z": self.z,
            "q_state": self.q_state.tolist()
        }
    
    @classmethod
    def from_dict(cls, data: Dict[str, Any]) -> 'RWVector':
        """Create vector from dictionary"""
        return cls(
            data.get("x", 0), 
            data.get("y", 0), 
            data.get("z", 0),
            np.array(data.get("q_state", []))
        )

        """Check if weapon can fire"""
        return self.current_cooldown <= 0
    
    def fire(self, target_position: RWVector, shooter_position: RWVector, shooter_accuracy: float) -> Dict[str, Any]:
        """Fire weapon at target position"""
        if not self.can_fire():
            return {
                "success": False,
                "message": "Weapon on cooldown",
                "damage": 0
            }
        
        # Calculate distance to target
        distance_vector = RWVector(
            target_position.x - shooter_position.x,
            target_position.y - shooter_position.y,
            target_position.z - shooter_position.z
        )
        distance = distance_vector.magnitude()
        
        # Check if target is in range
        if distance > self.range:
            return {
                "success": False,
                "message": "Target out of range",
                "damage": 0
            }
        
        # Calculate accuracy with distance penalty
        distance_factor = 1.0 - (distance / self.range) * 0.5
        effective_accuracy = self.accuracy * shooter_accuracy * distance_factor
        
        # Determine if attack hits
        hit_roll = random.random()
        hit = hit_roll <= effective_accuracy
        
        # Set cooldown
        self.current_cooldown = self.cooldown
        
        if not hit:
            return {
                "success": False,
                "message": "Attack missed",
                "damage": 0,
                "hit_roll": hit_roll,
                "accuracy_needed": effective_accuracy
            }
        
        # Calculate base damage
        damage = self.damage
        
        # Check for critical hit
        critical = random.random() <= self.critical_chance
        if critical:
            damage *= 2.0
        
        # Calculate quantum interference effect
        q_alignment = np.abs(np.dot(self.quantum_signature, target_position.q_state))
        quantum_factor = 0.8 + 0.4 * q_alignment
        damage *= quantum_factor
        
        # Element-specific damage variations
        element_factor = 1.0 + 0.2 * math.sin(time.time() * 0.1)  # Time-based fluctuation
        damage *= element_factor
        
        # Add weapon type specific effects
        type_effects = self._apply_weapon_type_effects(target_position, distance)
        
        # Progress weapon evolution slightly
        self.evolution_progress += 0.001 * damage
        
        return {
            "success": True,
            "message": "Hit" + (" (CRITICAL)" if critical else ""),
            "damage": damage,
            "critical": critical,
            "quantum_factor": quantum_factor,
            "element_factor": element_factor,
            "type_effects": type_effects
        }
    
    def _apply_weapon_type_effects(self, target_position: RWVector, distance: float) -> Dict[str, Any]:
        """Apply weapon type specific effects"""
        effects = {}
        
        if self.type == "beam":
            # Beams get stronger with charge and have piercing
            charge_bonus = min(2.0, 1.0 + self.current_cooldown / self.cooldown)
            effects["bonus_damage"] = self.damage * 0.1 * charge_bonus
            effects["piercing"] = self.special_attributes.get("piercing", 0.2)
            
        elif self.type == "projectile":
            # Projectiles have varying counts and spread
            count = self.special_attributes.get("projectile_count", 1)
            if count > 1:
                # Multiple projectiles
                hit_count = 0
                for _ in range(count):
                    if random.random() < self.accuracy:
                        hit_count += 1
                effects["multi_hit"] = hit_count
                effects["multi_damage"] = self.damage * 0.5 * hit_count
            
        elif self.type == "melee":
            # Melee gets stronger at close range and has combo potential
            if distance < self.range * 0.5:
                effects["close_range_bonus"] = self.damage * 0.3
            effects["stagger_chance"] = self.special_attributes.get("stagger_chance", 0.3)
            
        elif self.type == "aoe":
            # AoE has radius and falloff
            effects["radius"] = self.special_attributes.get("radius", 10.0)
            effects["splash_damage"] = self.damage * 0.6
            
        elif self.type == "status":
            # Status effects
            effects["status_type"] = self.element
            effects["duration"] = self.special_attributes.get("effect_duration", 5.0)
            effects["intensity"] = self.special_attributes.get("effect_strength", 1.0)
            
        elif self.type == "drain":
            # Drain weapons take resources from target
            drain_amount = min(
                self.special_attributes.get("max_drain", 20.0),
                self.damage * self.special_attributes.get("drain_ratio", 0.5)
            )
            effects["drain_amount"] = drain_amount
            effects["drain_type"] = self.special_attributes.get("drain_type", "energy")
            
        elif self.type == "field":
            # Persistent fields
            effects["field_radius"] = self.special_attributes.get("field_radius", 15.0)
            effects["field_duration"] = self.special_attributes.get("field_duration", 8.0)
            effects["field_type"] = self.element
            
        elif self.type == "summon":
            # Summon additional entities
            effects["summon_type"] = self.element
            effects["summon_count"] = self.special_attributes.get("summon_count", 1)
            effects["summon_health"] = self.special_attributes.get("summon_health", 30.0) * self.level

            self.quantum_signature = self.quantum_signature / np.linalg.norm(self.quantum_signature)
            
            effects["quantum_shift"] = np.linalg.norm(q_state_shift)
        
        return effects
    
    def evolve(self) -> bool:
        """Evolve weapon if evolution progress is sufficient"""
        if self.evolution_progress >= 1.0 and self.level < ROBOWARS_CONSTANTS["MAX_MECH_LEVEL"]:
            self.level += 1
            self.evolution_progress = 0.0
            
            # Improve stats based on level
            self.damage *= 1.2
            self.accuracy *= 1.05
            self.cooldown *= 0.95
            self.range *= 1.1
            self.critical_chance *= 1.1
            
            # Quantum signature evolution
            q_noise = np.random.uniform(-0.1, 0.1, self.quantum_signature.shape)
            self.quantum_signature += q_noise
            self.quantum_signature = self.quantum_signature / np.linalg.norm(self.quantum_signature)
            
            # Special attribute evolution
            for key in self.special_attributes:

    """Special ability for battle mechs"""
    
    def __init__(self, name: str, ability_type: str, element: str, params: Optional[Dict[str, Any]] = None):
        self.id = str(uuid.uuid4())
        self.name = name
        self.type = ability_type
        self.element = element
        
        # Default parameters
        self.cooldown = 10.0
        self.energy_cost = 20.0
        self.duration = 5.0
        self.effectiveness = 1.0
        self.current_cooldown = 0.0
        self.active = False
        self.active_time_remaining = 0.0
        self.level = 1
        self.evolution_progress = 0.0
        self.quantum_signature = np.random.uniform(-1, 1, ROBOWARS_CONSTANTS["QUANTUM_DIMENSIONS"])
        self.quantum_signature = self.quantum_signature / np.linalg.norm(self.quantum_signature)
        
        # Element-specific modifiers
        self._apply_element_modifiers()
        
        # Override with provided parameters if any
        if params:
            for key, value in params.items():
                if hasattr(self, key):
                    setattr(self, key, value)
        
        # Special type-based attributes
        self.special_attributes = self._initialize_special_attributes()
    
    def _apply_element_modifiers(self) -> None:
        """Apply element-specific modifiers to ability stats"""
        element_modifiers = {

"scan_radius": 100.0,
                "resource_recovery": 30.0,

    def update(self, dt: float) -> None:
        """Update ability state based on time"""
        # Cooldown timer
        if self.current_cooldown > 0:
            self.current_cooldown = max(0, self.current_cooldown - dt)
        
        # Active ability duration
        if self.active:
            self.active_time_remaining = max(0, self.active_time_remaining - dt)
            if self.active_time_remaining <= 0:
                self.deactivate()
    
    def can_activate(self) -> bool:
        """Check if ability can be activated"""
        return not self.active and self.current_cooldown <= 0
    
    def activate(self) -> Dict[str, Any]:
        """Activate the ability"""
        if not self.can_activate():
            return {
                "success": False,
                "message": "Ability not ready",
                "effects": {}
            }
        
        self.active = True
        self.active_time_remaining = self.duration
        self.current_cooldown = self.cooldown
        
        # Calculate effectiveness with quantum fluctuations
        quantum_factor = 0.8 + 0.4 * random.random()
        effective_power = self.effectiveness * quantum_factor
        
        # Apply type-specific activation effects
        activation_effects = self._get_activation_effects(effective_power)
        
        # Progress ability evolution slightly
        self.evolution_progress += 0.05
        
        return {
            "success": True,
            "message": f"{self.name} activated",
            "effects": activation_effects,
            "duration": self.active_time_remaining,
            "quantum_factor": quantum_factor
        }
    
    def deactivate(self) -> Dict[str, Any]:
        """Deactivate the ability"""
        if not self.active:
            return {
                "success": False,
                "message": "Ability not active",
                "effects": {}
            }
        
        self.active = False
        self.active_time_remaining = 0
        
        # Apply deactivation effects
        deactivation_effects = self._get_deactivation_effects()
        
        return {
            "success": True,
            "message": f"{self.name} deactivated",
            "effects": deactivation_effects
        }
    
    def get_active_effects(self) -> Dict[str, Any]:
        """Get current active effects if ability is active"""
        if not self.active:
            return {}
        
        # Calculate current effect strength based on remaining time
        time_factor = self.active_time_remaining / self.duration
        
        # Base effects vary by ability type
        effects = {}
        
        if self.type == "movement":
            effects["speed_modifier"] = 1.0 + (self.special_attributes.get("speed_boost", 2.0) - 1.0) * time_factor
            effects["maneuverability_modifier"] = 1.0 + (self.special_attributes.get("maneuverability", 1.5) - 1.0) * time_factor
            effects["phase_chance"] = self.special_attributes.get("phase_chance", 0.1) * time_factor
            
        elif self.type == "defense":
            effects["damage_reduction"] = self.special_attributes.get("damage_reduction", 0.5) * time_factor
            effects["shield_strength"] = self.special_attributes.get("shield_strength", 50.0) * time_factor
            effects["reflection_chance"] = self.special_attributes.get("reflection_chance", 0.2) * time_factor
            
        elif self.type == "buff":
            effects["stat_boost"] = 1.0 + (self.special_attributes.get("stat_boost", 1.5) - 1.0) * time_factor
            effects["boost_targets"] = self.special_attributes.get("boost_targets", ["damage", "accuracy"])
            
        elif self.type == "debuff":
            effects["stat_reduction"] = self.special_attributes.get("stat_reduction", 0.7) + (1.0 - self.special_attributes.get("stat_reduction", 0.7)) * (1.0 - time_factor)
            effects["reduction_targets"] = self.special_attributes.get("reduction_targets", ["movement", "defense"])
            
        elif self.type == "counter":
            effects["trigger_chance"] = self.special_attributes.get("trigger_chance", 0.7) * time_factor
            effects["counter_damage"] = self.special_attributes.get("counter_damage", 1.5) * time_factor
            
        elif self.type == "stealth":
            effects["detection_difficulty"] = self.special_attributes.get("detection_difficulty", 0.7) * time_factor
            effects["movement_penalty"] = self.special_attributes.get("movement_penalty", 0.8) + (1.0 - self.special_attributes.get("movement_penalty", 0.8)) * (1.0 - time_factor)
            
        elif self.type == "overcharge":
            effects["power_boost"] = 1.0 + (self.special_attributes.get("power_boost", 2.0) - 1.0) * time_factor
            effects["stability"] = self.special_attributes.get("stability", 0.6) * time_factor
            effects["burnout_risk"] = self.special_attributes.get("burnout_risk", 0.3) * (1.0 - time_factor)
        
        # Add elemental effects
        effects["element"] = self.element
        
        return effects
    
    def _get_activation_effects(self, effective_power: float) -> Dict[str, Any]:
        """Get activation effects specific to ability type"""
        effects = {}
        
        # Apply common effect for effectiveness
        effects["effectiveness"] = effective_power
        
        # Type-specific effects
        if self.type == "movement":
            effects["speed_boost"] = self.special_attributes.get("speed_boost", 2.0) * effective_power
            effects["movement_type"] = self.element
            
        elif self.type == "defense":
            effects["shield_strength"] = self.special_attributes.get("shield_strength", 50.0) * effective_power
            effects["shield_type"] = self.element
            
        elif self.type == "utility":
            effects["utility_radius"] = self.special_attributes.get("scan_radius", 100.0) * effective_power
            effects["utility_type"] = self.element
            
        elif self.type == "scan":
            effects["scan_radius"] = self.special_attributes.get("scan_radius", 100.0) * effective_power
            effects["weakness_detection"] = self.special_attributes.get("weakness_detection", 0.6) * effective_power
            
        elif self.type == "repair":
            effects["repair_amount"] = self.special_attributes.get("repair_amount", 40.0) * effective_power
            effects["repair_type"] = self.element
        
        return effects
    
    def _get_deactivation_effects(self) -> Dict[str, Any]:
        """Get deactivation effects specific to ability type"""
        effects = {}
        
        # Some abilities have special deactivation effects
        if self.type == "overcharge":
            # Overcharge might have burnout effects on deactivation
            burnout_risk = self.special_attributes.get("burnout_risk", 0.3)
            if random.random() < burnout_risk:
                effects["burnout"] = True
                effects["damage"] = 20.0 * self.level
            
        elif self.type == "movement" and self.element == "temporal":
            # Temporal movement might cause position snap-back
            effects["position_flux"] = True
            effects["flux_magnitude"] = 0.2 * self.level
        
        return effects
    
    def evolve(self) -> bool:
        """Evolve ability if evolution progress is sufficient"""
        if self.evolution_progress >= 1.0 and self.level < ROBOWARS_CONSTANTS["MAX_MECH_LEVEL"]:
            self.level += 1
            self.evolution_progress = 0.0
            
            # Improve stats based on level
            self.effectiveness *= 1.15
            self.duration *= 1.1
            self.cooldown *= 0.95
            
            # Quantum signature evolution
            q_noise = np.random.uniform(-0.1, 0.1, self.quantum_signature.shape)
            self.quantum_signature += q_noise
            self.quantum_signature = self.quantum_signature / np.linalg.norm(self.quantum_signature)
            
            # Special attribute evolution
            for key in self.special_attributes:
                # Improve special attributes by 10-20%
                improvement = 1.1 + 0.1 * random.random()
                self.special_attributes[key] *= improvement
            
            return True
        
        return False

        """Convert to dictionary"""
        return {

            "quantum_signature": self.quantum_signature.tolist(),
            "special_attributes": self.special_attributes
        }

        for ability_id in self.abilities:

        self._update_status_effects(dt)
        
        # Natural regeneration
        self._apply_regeneration(dt)
        
        # Update position based on velocity
        self.position = self.position + self.velocity * dt
        
        # Apply movement friction/dampening
        self.velocity = self.velocity * 0.95
        
        # Apply quantum effects
        self._apply_quantum_effects(dt, all_mechs)

    def _update_status_effects(self, dt: float) -> None:
        """Update status effects"""
        # Track effects to remove
        effects_to_remove = []
        
        # Process each active status effect
        for effect_id, effect in self.status_effects.items():
            # Reduce duration
            effect["duration"] -= dt
            
            # Check if effect has expired
            if effect["duration"] <= 0:
                effects_to_remove.append(effect_id)
                continue
            
            # Apply ongoing effect
            effect_type = effect.get("type", "")
            
            if effect_type == "damage_over_time":
                damage = effect.get("damage_per_tick", 1.0) * dt
                self.take_damage(damage, effect.get("source", ""), effect.get("element", "status"))
            
            elif effect_type == "healing_over_time":
                healing = effect.get("healing_per_tick", 1.0) * dt
                self.heal(healing)
            
            elif effect_type == "energy_drain":
                drain = effect.get("drain_per_tick", 1.0) * dt
                self.energy = max(0, self.energy - drain)
            
            elif effect_type == "energy_regen":
                regen = effect.get("regen_per_tick", 1.0) * dt
                self.energy = min(self.max_energy, self.energy + regen)
            
            elif effect_type == "quantum_disruption":
                # Periodically apply quantum noise
                if random.random() < dt * 0.5:
                    q_noise = np.random.uniform(-0.1, 0.1, self.quantum_signature.shape)
                    self.quantum_signature += q_noise
                    self.quantum_signature = self.quantum_signature / np.linalg.norm(self.quantum_signature)
        
        # Remove expired effects
        for effect_id in effects_to_remove:
            del self.status_effects[effect_id]
    
    def _apply_regeneration(self, dt: float) -> None:
        """Apply natural regeneration"""
        # Health regeneration
        health_regen_rate = 0.5 * dt  # 0.5 health per second
        self.health = min(self.max_health, self.health + health_regen_rate)
        
        # Energy regeneration
        energy_regen_rate = 1.0 * dt  # 1 energy per second
        self.energy = min(self.max_energy, self.energy + energy_regen_rate)
    
    def _apply_quantum_effects(self, dt: float, all_mechs: Optional[Dict[str, 'BattleMech']] = None) -> None:
        """Apply quantum effects including entanglement"""
        # Apply subtle quantum fluctuations
        if random.random() < 0.1 * dt:
            q_noise = np.random.uniform(-0.05, 0.05, self.quantum_signature.shape)
            self.quantum_signature += q_noise
            self.quantum_signature = self.quantum_signature / np.linalg.norm(self.quantum_signature)
        
        # Process quantum entanglements
        if all_mechs:
            for other_id, strength in list(self.quantum_entanglements.items()):
                # Skip if mech no longer exists
                if other_id not in all_mechs:
                    del self.quantum_entanglements[other_id]
                    continue
                
                other_mech = all_mechs[other_id]
                
                # Decay entanglement over time
                strength = max(0, strength - ROBOWARS_CONSTANTS["RESONANCE_DECAY"] * dt)
                
                if strength <= 0.01:
                    # Entanglement is too weak, remove it
                    del self.quantum_entanglements[other_id]
                    if self.id in other_mech.quantum_entanglements:
                        del other_mech.quantum_entanglements[self.id]
                else:
                    # Update entanglement strength
                    self.quantum_entanglements[other_id] = strength
                    
                    # Apply entanglement effects
                    self._apply_entanglement_effects(other_mech, strength, dt)
    
    def _apply_entanglement_effects(self, other_mech: 'BattleMech', strength: float, dt: float) -> None:
        """Apply quantum entanglement effects between two mechs"""
        # Probability of effect based on strength and time
        if random.random() < strength * dt:
            # Shared quantum state effect
            q_blend = 0.1 * strength
            
            # Blend quantum signatures
            shared_signature = q_blend * other_mech.quantum_signature + (1 - q_blend) * self.quantum_signature
            shared_signature = shared_signature / np.linalg.norm(shared_signature)
            
            self.quantum_signature = shared_signature.copy()
            
            # Mirror some status effects (small chance)
            if self.status_effects and random.random() < 0.1 * strength:
                # Select a random effect to mirror
                effect_id = random.choice(list(self.status_effects.keys()))
                effect = self.status_effects[effect_id].copy()
                
                # Reduce duration and intensity for mirrored effect
                effect["duration"] *= 0.5
                if "intensity" in effect:
                    effect["intensity"] *= 0.5
                
                # Apply to other mech
                new_effect_id = str(uuid.uuid4())
                other_mech.status_effects[new_effect_id] = effect
    
    def _update_targeting(self, all_mechs: Dict[str, 'BattleMech']) -> None:
        """Update auto-targeting"""
        if not self.aggressive_mode or not all_mechs:
            return
        
        best_target = None
        best_score = -1
        
        for mech_id, mech in all_mechs.items():
            # Skip self
            if mech_id == self.id:
                continue
            
            # Skip destroyed mechs
            if mech.health <= 0:
                continue
            
            # Calculate targeting score based on distance, health, and quantum alignment
            distance = (mech.position - self.position).magnitude()
            
            # Skip targets too far away
            max_weapon_range = max([w.range for w in self.weapons.values()], default=0)
            if distance > max_weapon_range * 1.2:  # 20% buffer for movement
                continue
            
            # Distance score - closer is better
            distance_score = 1.0 - min(1.0, distance / max_weapon_range)
            
            # Health score - lower health targets are preferred
            health_score = 1.0 - (mech.health / mech.max_health)
            
            # Quantum alignment score
            q_alignment = np.abs(np.dot(self.quantum_signature, mech.quantum_signature))
            
            # Combined score
            score = 0.5

# Combined score
            score = 0.5 * distance_score + 0.3 * health_score + 0.2 * q_alignment
            
            # Check if this is the best target so far
            if score > best_score:
                best_score = score
                best_target = mech_id
        
        # Set target lock if we found a good target
        if best_target and best_score > 0.4:
            self.target_lock = best_target
    
    def move(self, direction: RWVector, power: float = 1.0) -> None:
        """Move the mech in a direction"""
        # Calculate movement force
        move_force = direction.normalize() * (self.speed * power)
        
        # Apply movement
        self.velocity = self.velocity + move_force
    
    def rotate(self, angle: float) -> None:
        """Rotate the mech by angle (in radians)"""
        self.orientation = (self.orientation + angle) % (2 * math.pi)
    
    def fire_weapon(self, weapon_id: str, target_position: Optional[RWVector] = None) -> Dict[str, Any]:
        """Fire a specific weapon at target"""
        # Check if weapon exists
        if weapon_id not in self.weapons:
            return {
                "success": False,
                "message": f"Weapon {weapon_id} not found"
            }
        
        weapon = self.weapons[weapon_id]
        
        # Check if weapon can fire
        if not weapon.can_fire():
            return {
                "success": False,
                "message": "Weapon on cooldown"
            }
        
        # Check energy cost
        if self.energy < weapon.energy_cost:
            return {
                "success": False,
                "message": "Insufficient energy"
            }
        
        # Use target position if provided, otherwise use target lock
        actual_target = target_position
        if actual_target is None and self.target_lock:
            # No implementation for target_positions yet
            return {
                "success": False,
                "message": "No target position provided and target lock not implemented yet"
            }
        
        if actual_target is None:
            return {
                "success": False,
                "message": "No target specified"
            }
        
        # Consume energy
        self.energy -= weapon.energy_cost
        
        # Fire weapon
        result = weapon.fire(actual_target, self.position, self.accuracy)
        
        # Log the attack
        self.combat_log.append({
            "action": "weapon_fire",
            "weapon_id": weapon_id,
            "weapon_name": weapon.name,
            "target_position": actual_target.to_dict() if actual_target else None,
            "result": result,
            "timestamp": time.time()
        })
        
        return {
            "success": result["success"],
            "message": result["message"],
            "damage": result.get("damage", 0),
            "weapon_name": weapon.name,
            "critical": result.get("critical", False),
            "type_effects": result.get("type_effects", {})
        }

    def take_damage(self, amount: float, source_id: str = "", damage_type: str = "physical") -> Dict[str, Any]:
        """Take damage from an attack"""
        # Apply armor reduction
        effective_armor = self.armor
        
        # Damage type effectiveness
        type_multiplier = self._get_damage_type_effectiveness(damage_type)
        
        # Armor penetration based on damage type
        armor_penetration = 0.0
        if damage_type == "quantum":
            armor_penetration = 0.5
        elif damage_type == "void":
            armor_penetration = 0.7
        elif damage_type == "plasma":
            armor_penetration = 0.3
        
        # Calculate damage reduction from armor
        reduction_factor = 1.0 - (effective_armor * (1.0 - armor_penetration)) / (effective_armor + 100.0)
        
        # Apply reduction to damage
        effective_damage = amount * reduction_factor * type_multiplier
        
        # Apply evasion chance
        if random.random() < self.evasion:
            return {
                "success": False,
                "message": "Attack evaded",
                "damage": 0
            }
        
        # Apply damage
        self.health = max(0, self.health - effective_damage)
        self.damage_taken += effective_damage
        
        # Check if mech is destroyed
        destroyed = self.health <= 0
        
        # Log the damage
        self.combat_log.append({
            "action": "take_damage",
            "source_id": source_id,
            "damage_type": damage_type,
            "raw_damage": amount,
            "effective_damage": effective_damage,
            "remaining_health": self.health,
            "destroyed": destroyed,
            "timestamp": time.time()
        })
        
        # Create quantum entanglement with damage source (if probability hits)
        if source_id and random.random() < 0.1:
            entanglement_strength = ROBOWARS_CONSTANTS["ENTANGLEMENT_STRENGTH"] * (amount / 100.0)
            self.quantum_entanglements[source_id] = entanglement_strength
        
        return {
            "success": True,
            "message": "Damage taken" + (" (DESTROYED)" if destroyed else ""),
            "raw_damage": amount,
            "effective_damage": effective_damage,
            "remaining_health": self.health,
            "destroyed": destroyed
        }
    
    def _get_damage_type_effectiveness(self, damage_type: str) -> float:
        """Get damage type effectiveness multiplier against this mech's element"""
        # Elemental effectiveness chart
        effectiveness = {
            # Format: {attacker_element: {defender_element: multiplier}}
            "fire": {"ice": 2.0, "water": 0.5, "earth": 0.8, "crystal": 1.2},
            "water": {"fire": 2.0, "earth": 0.8, "lightning": 0.5, "plasma": 1.5},
            "earth": {"lightning": 0.5, "water": 1.2, "air": 0.5, "crystal": 0.8},
            "air": {"earth": 2.0, "lightning": 1.2, "ice": 0.8, "gravity": 0.5},
            "lightning": {"water": 2.0, "earth": 2.0, "air": 0.8, "temporal": 1.2},
            "ice": {"fire": 0.5, "air": 1.2, "water": 0.8, "plasma": 0.5},
            "quantum": {"void": 0.5, "temporal": 1.5, "gravity": 1.2, "crystal": 0.8},
            "gravity": {"air": 2.0, "quantum": 0.8, "void": 1.2, "temporal": 0.5},
            "temporal": {"quantum": 0.5, "gravity": 2.0, "lightning": 0.8, "void": 1.2},
            "void": {"quantum": 2.0, "temporal": 0.8, "crystal": 0.5, "plasma": 1.5},
            "crystal": {"void": 2.0, "quantum": 1.2, "earth": 1.5, "fire": 0.8},
            "plasma": {"ice": 2.0, "void": 0.8, "water": 0.5, "fire": 1.2}
        }
        
        # Get multiplier if both damage type and mech element are in the effectiveness chart
        if damage_type in effectiveness and self.primary_element in effectiveness[damage_type]:
            return effectiveness[damage_type][self.primary_element]
        
        # Default effectiveness
        return 1.0

        """Heal the mech"""
        # Calculate effective healing
        effective_heal = amount
        
        # Apply healing
        old_health = self.health
        self.health = min(self.max_health, self.health + effective_heal)
        actual_healing = self.health - old_health
        
        # Log the healing
        self.combat_log.append({
            "action": "heal",
            "amount": amount,
            "effective_healing": actual_healing,
            "new_health": self.health,
            "timestamp": time.time()
        })
        
        return {
            "success": True,
            "message": "Healing received",
            "amount": amount,
            "effective_healing": actual_healing,
            "new_health": self.health
        }
    
    def apply_status_effect(self, effect: Dict[str, Any], source_id: str = "") -> Dict[str, Any]:
        """Apply a status effect to the mech"""
        # Generate effect ID
        effect_id = str(uuid.uuid4())
        
        # Ensure effect has required fields
        if "type" not in effect or "duration" not in effect:
            return {
                "success": False,
                "message": "Invalid effect data"
            }
        
        # Add source ID
        effect["source"] = source_id
        
        # Check for resistance to this effect type
        resistance = self._get_status_effect_resistance(effect["type"])
        
        # Apply resistance
        if resistance > 0:
            # Chance to completely resist
            if random.random() < resistance:
                return {
                    "success": False,
                    "message": "Effect resisted",
                    "effect_type": effect["type"]
                }
            
            # Reduce duration and intensity
            effect["duration"] *= (1.0 - resistance * 0.5)
            if "intensity" in effect:
                effect["intensity"] *= (1.0 - resistance * 0.5)
        
        # Add effect
        self.status_effects[effect_id] = effect
        
        # Log the effect
        self.combat_log.append({
            "action": "status_effect_applied",
            "effect_id": effect_id,
            "effect_type": effect["type"],
            "source_id": source_id,
            "duration": effect["duration"],
            "timestamp": time.time()
        })
        
        return {
            "success": True,
            "message": "Status effect applied",
            "effect_id": effect_id,
            "effect_type": effect["type"],
            "duration": effect["duration"]
        }

    def gain_experience(self, amount: float) -> Dict[str, Any]:
        """Gain experience and potentially level up"""
        # Add experience
        self.experience += amount
        
        # Check for level up
        old_level = self.level
        exp_for_next_level = self._get_exp_for_level(self.level + 1)
        
        level_ups = 0
        while self.experience >= exp_for_next_level:
            self.level += 1
            level_ups += 1
            exp_for_next_level = self._get_exp_for_level(self.level + 1)
        
        # If leveled up, improve stats
        if level_ups > 0:
            for _ in range(level_ups):
                self._apply_level_up()
        
        return {
            "success": True,
            "message": f"Gained {amount} experience" + (f" and leveled up {level_ups} times!" if level_ups > 0 else ""),
            "current_exp": self.experience,
            "old_level": old_level,
            "new_level": self.level,
            "exp_for_next": exp_for_next_level - self.experience
        }
    
    def _get_exp_for_level(self, level: int) -> float:
        """Get experience required for a specific level"""
        # Experience curve
        return 100.0 * (level ** 2)
    
    def _apply_level_up(self) -> None:
        """Apply stat improvements for a level up"""
        # Improve core stats
        self.max_health *= 1.1
        self.health = self.max_health
        self.max_energy *= 1.1
        self.energy = self.max_energy
        self.armor *= 1.1
        self.speed *= 1.05
        self.accuracy *= 1.03
        self.evasion *= 1.03
        
        # Add to evolution progress
        self.evolution_progress += 0.2
        
        # Chance to evolve weapons and abilities
        for weapon_id in self.weapons:
            if random.random() < 0.3:  # 30% chance
                self.weapons[weapon_id].evolve()
        
        for ability_id in self.abilities:
            if random.random() < 0.3:  # 30% chance
                self.abilities[ability_id].evolve()
        
        # Log level up
        self.combat_log.append({
            "action": "level_up",
            "new_level": self.level,
            "timestamp": time.time()
        })
    
    def evolve(self) -> Dict[str, Any]:
        """Trigger evolution if progress is sufficient"""
        if self.evolution_progress < 1.0:
            return {
                "success": False,
                "message": "Insufficient evolution progress",
                "current_progress": self.evolution_progress
            }
        
        # Reset evolution progress
        self.evolution_progress = 0.0
        
        # Significant stat improvements
        self.max_health *= 1.3
        self.health = self.max_health
        self.max_energy *= 1.3
        self.energy = self.max_energy
        self.armor *= 1.25
        self.speed *= 1.2
        self.accuracy *= 1.15
        self.evasion *= 1.15
        
        # Quantum signature evolution
        q_noise = np.random.uniform(-0.2, 0.2, self.quantum_signature.shape)
        self.quantum_signature += q_noise
        self.quantum_signature = self.quantum_signature / np.linalg.norm(self.quantum_signature)
        
        # Evolve all weapons and abilities
        for weapon_id in self.weapons:
            self.weapons[weapon_id].evolve()
        
        for ability_id in self.abilities:
            self.abilities[ability_id].evolve()
        
        # Chance to develop new weapon or ability
        if random.random() < 0.5 and len(self.weapons) < ROBOWARS_CONSTANTS["MAX_WEAPONS"]:
            # New weapon with random type but matching element
            weapon_type = random.choice(ROBOWARS_CONSTANTS["WEAPON_TYPES"])
            weapon_name = f"Evolved {self.primary_element.capitalize()} {weapon_type.capitalize()}"
            weapon = Weapon(weapon_name, weapon_type, self.primary_element)
            weapon.level = max(1, self.level - 1)  # Start at 1 level lower
            self.add_weapon(weapon)
        
        if random.random() < 0.5 and len(self.abilities) < ROBOWARS_CONSTANTS["MAX_ABILITIES"]:
            # New ability with random type but matching element
            ability_type = random.choice(ROBOWARS_CONSTANTS["ABILITY_TYPES"])
            ability_name = f"Evolved {self.primary_element.capitalize()} {ability_type.capitalize()}"
            ability = Ability(ability_name, ability_type, self.primary_element)
            ability.level = max(1, self.level - 1)  # Start at 1 level lower
            self.add_ability(ability)
        
        # Log evolution
        self.combat_log.append({
            "action": "evolution",
            "timestamp": time.time()
        })
        
        return {
            "success": True,
            "message": "Evolution complete",
            "weapons_count": len(self.weapons),
            "abilities_count": len(self.abilities)
        }

        mech.status_effects = data["status_effects"]

        mech.damage_dealt = data["damage_dealt"]

        for weapon_data in data["weapons"].values():

        # Load abilities

    def start_battle(self) -> Dict[str, Any]:
        """Start the battle"""
        if self.battle_active:
            return {
                "success": False,
                "message": "Battle already in progress"
            }
        
        if len(self.mechs) < 2:
            return {
                "success": False,
                "message": "Need at least two mechs for a battle"
            }
        
        # Reset battle state
        self.battle_active = True
        self.battle_finished = False
        self.winner_id = None
        self.start_time = time.time()
        self.current_time = self.start_time
        
        # Log battle start
        self.battle_log.append({
            "action": "battle_start",
            "mechs": [{"id": mech_id, "name": mech.name, "type": mech.type, "element": mech.primary_element}
                    for mech_id, mech in self.mechs.items()],
            "arena_type": self.arena_type,
            "timestamp": self.start_time
        })
        
        return {
            "success": True,
            "message": f"Battle started with {len(self.mechs)} mechs",
            "arena_type": self.arena_type,
            "mech_count": len(self.mechs)
        }
    
    def update(self, dt: float) -> Dict[str, Any]:
        """Update the battle state"""
        if not self.battle_active or self.battle_finished:
            return {
                "success": False,
                "message": "Battle not active"
            }
        
        # Apply time scale
        effective_dt = dt * self.time_scale
        self.current_time += effective_dt
        
        # Calculate battle progress
        elapsed_time = self.current_time - self.start_time
        
        # Check if battle has exceeded max duration
        if elapsed_time >= self.max_duration:
            return self.end_battle("time_limit")
        
        # Update environmental effects
        self._update_environmental_effects(effective_dt)
        
        # Update quantum field
        self._update_quantum_field(effective_dt)
        
        # Update mechs
        active_mechs = 0
        
        for mech_id, mech in list(self.mechs.items()):
            # Skip destroyed mechs
            if mech.health <= 0:
                continue
            
            active_mechs += 1
            
            # Apply environmental effects to mech
            self._apply_environmental_effects(mech, effective_dt)
            
            # Update mech state
            mech.update(effective_dt, self.mechs)
            
            # Apply arena effects
            self._apply_arena_effects(mech, effective_dt)
            
            # Check for boundary constraints
            self._apply_boundary_constraints(mech)
        
        # Check battle end conditions
        if active_mechs <= 1:
            # Find the last mech standing (if any)
            for mech_id, mech in self.mechs.items():
                if mech.health > 0:
                    return self.end_battle("victory", mech_id)
            
            # No mechs left (all destroyed)
            return self.end_battle("mutual_destruction")
        
        return {
            "success": True,
            "message": "Battle updated",
            "active_mechs": active_mechs,
            "elapsed_time": elapsed_time
        }
    
    def _update_environmental_effects(self, dt: float) -> None:
        """Update environmental effects"""
        # Track effects to remove
        effects_to_remove = []
        
        for i, effect in enumerate(self.environmental_effects):
            # Update effect duration if applicable
            if "duration" in effect:
                effect["duration"] -= dt
                if effect["duration"] <= 0:
                    effects_to_remove.append(i)
                    continue
            
            # Update effect position if it has movement
            if "movement" in effect and isinstance(effect["movement"], RWVector):
                effect["position"] = effect["position"] + effect["movement"] * dt
                
                # Bounce off arena boundaries
                position = effect["position"]
                radius = effect.get("radius", 0)
                
                if abs(position.x) > self.size/2 - radius:
                    effect["movement"].x *= -1
                    # Ensure inside boundary
                    position.x = math.copysign(self.size/2 - radius, position.x)
                
                if abs(position.y) > self.size/2 - radius:
                    effect["movement"].y *= -1
                    # Ensure inside boundary
                    position.y = math.copysign(self.size/2 - radius, position.y)
                
                if position.z < 0:
                    effect["movement"].z *= -1
                    position.z = 0
                
                if position.z > self.size/2:
                    effect["movement"].z *= -1
                    position.z = self.size/2
            
            # Update pulse effects
            if "pulse_frequency" in effect:
                # Calculate pulse intensity
                pulse_phase = (self.current_time * effect["pulse_frequency"]) % 1.0
                pulse_intensity = 0.5 + 0.5 * math.sin(pulse_phase * 2 * math.pi)
                effect["current_intensity"] = pulse_intensity
        
        # Remove expired effects
        for i in sorted(effects_to_remove, reverse=True):
            del self.environmental_effects[i]
    
    def _update_quantum_field(self, dt: float) -> None:
        """Update quantum field"""
        # Quantum fields evolve differently based on arena type
        if self.arena_type == "quantum_flux":
            # Highly variable quantum field
            for i in range(10):
                for j in range(10):
                    for k in range(10):
                        if random.random() < 0.1 * dt:
                            q_noise = np.random.uniform(-0.2, 0.2, ROBOWARS_CONSTANTS["QUANTUM_DIMENSIONS"])
                            self.quantum_field[i, j, k] += q_noise
                            self.quantum_field[i, j, k] = self.quantum_field[i, j, k] / np.linalg.norm(self.quantum_field[i, j, k])
        
        elif self.arena_type == "temporal_nexus":
            # Temporal quantum field evolves in patterns
            t = self.current_time * 0.1
            for i in range(10):
                for j in range(10):
                    for k in range(10):
                        phase = math.sin(t + (i+j+k) * 0.1) * 0.2
                        rotation = np.eye(ROBOWARS_CONSTANTS["QUANTUM_DIMENSIONS"])
                        for d in range(ROBOWARS_CONSTANTS["QUANTUM_DIMENSIONS"]-1):
                            rotation[d, d] = math.cos(phase)
                            rotation[d, d+1] = -math.sin(phase)
                            rotation[d+1, d] = math.sin(phase)
                            rotation[d+1, d+1] = math.cos(phase)
                        
                        # Apply rotation to field
                        self.quantum_field[i, j, k] = np.dot(rotation, self.quantum_field[i, j, k])
                        self.quantum_field[i, j, k] = self.quantum_field[i, j, k] / np.linalg.norm(self.quantum_field[i, j, k])
        
        else:
            # Default subtle evolution
            for i in range(10):
                for j in range(10):
                    for k in range(10):
                        if random.random() < 0.03 * dt:
                            q_noise = np.random.uniform(-0.05, 0.05, ROBOWARS_CONSTANTS["QUANTUM_DIMENSIONS"])
                            self.quantum_field[i, j, k] += q_noise
                            self.quantum_field[i, j, k] = self.quantum_field[i, j, k] / np.linalg.norm(self.quantum_field[i, j, k])
    
    def _apply_environmental_effects(self, mech: BattleMech, dt: float) -> None:
        """Apply environmental effects to a mech"""
        for effect in self.environmental_effects:
            # Calculate distance to effect
            effect_pos = effect["position"]
            distance = (effect_pos - mech.position).magnitude()
            radius = effect.get("radius", 0)
            
            # Check if mech is within effect radius
            if distance <= radius:
                # Calculate effect intensity based on distance
                intensity = 1.0 - (distance / radius)
                
                # Apply pulse intensity if applicable
                if "current_intensity" in effect:
                    intensity *= effect["current_intensity"]
                
                # Apply effect based on type
                effect_type = effect.get("type", "")
                
                if effect_type == "damage_zone":
                    damage = effect.get("damage_per_second", 5.0) * intensity * dt
                    element = effect.get("element", "plasma")
                    mech.take_damage(damage, "environment", element)
                
                elif effect_type == "gravity_well":
                    strength = effect.get("strength", 10.0) * intensity
                    
                    # Calculate direction to gravity well
                    direction = effect_pos - mech.position
                    if direction.magnitude() > 0:
                        direction = direction.normalize()
                        
                        # Apply force toward gravity well
                        force = direction * strength
                        mech.move(force, 0.2 * dt)  # Reduced effect for balance
                
                elif effect_type == "time_distortion":
                    time_factor = effect.get("time_factor", 1.0)
                    
                    # Apply time distortion to cooldowns
                    for weapon_id in mech.weapons:
                        if mech.weapons[weapon_id].current_cooldown > 0:
                            cooldown_change = (time_factor - 1.0) * dt * intensity
                            mech.weapons[weapon_id].current_cooldown += cooldown_change
                    
                    for ability_id in mech.abilities:
                        if mech.abilities[ability_id].current_cooldown > 0:
                            cooldown_change = (time_factor - 1.0) * dt * intensity
                            mech.abilities[ability_id].current_cooldown += cooldown_change
                
                elif effect_type == "quantum_flux":
                    shift_strength = effect.get("quantum_shift_strength", 0.3) * intensity
                    
                    # Apply quantum shift to mech
                    q_shift = np.random.uniform(-shift_strength, shift_strength, mech.quantum_signature.shape)
                    mech.quantum_signature += q_shift
                    mech.quantum_signature = mech.quantum_signature / np.linalg.norm(mech.quantum_signature)
                    
                    # Chance to create random entanglement
                    if random.random() < 0.01 * shift_strength:
                        other_mechs = [m for m_id, m in self.mechs.items() if m_id != mech.id and m.health > 0]
                        if other_mechs:
                            other_mech = random.choice(other_mechs)
                            entangle_strength = random.uniform(0.1, 0.3) * shift_strength
                            mech.quantum_entanglements[other_mech.id] = entangle_strength
                            other_mech.quantum_entanglements[mech.id] = entangle_strength
    
    def _apply_arena_effects(self, mech: BattleMech, dt: float) -> None:
        """Apply arena-specific effects to a mech"""
        # Apply quantum field effect
        self._apply_quantum_field_effect(mech)
        
        # Apply arena parameters
        
        # Energy regeneration modifier
        energy_regen = 1.0 * dt * self.parameters["energy_regen_factor"]
        mech.energy = min(mech.max_energy, mech.energy + energy_regen)
        
        # Health regeneration modifier
        health_regen = 0.5 * dt * self.parameters["health_regen_factor"]
        mech.health = min(mech.max_health, mech.health + health_regen)
        
        # Apply gravity
        gravity_force = RWVector(0, 0, -self.parameters["gravity"] * dt)
        mech.velocity = mech.velocity + gravity_force
        
        # Apply friction
        friction = self.parameters["friction"]
        mech.velocity = mech.velocity * (1.0 - friction * dt)
    
    def _apply_quantum_field_effect(self, mech: BattleMech) -> None:
        """Apply quantum field effect to a mech"""
        # Get the cell in the quantum field grid
        pos = mech.position
        
        # Convert world position to grid indices
        grid_size = self.size / 10
        i = int((pos.x + self.size/2) / grid_size)
        j = int((pos.y + self.size/2) / grid_size)
        k = int(pos.z / grid_size)
        
        # Clamp indices to valid range
        i = max(0, min(9, i))
        j = max(0, min(9, j))
        k = max(0, min(9, k))
        
        # Get the quantum field at this position
        field_signature = self.quantum_field[i, j, k]
        
        # Calculate quantum alignment
        alignment = np.abs(np.dot(mech.quantum_signature, field_signature))
        
        # Apply effects based on alignment
        if alignment > 0.8:
            # High alignment - positive effects
            if random.random() < 0.1 * (alignment - 0.8) * 5:
                # Energy boost
                energy_boost = 5.0 * (alignment - 0.8) * 5
                mech.energy = min(mech.max_energy, mech.energy + energy_boost)
                
                # Small heal
                health_boost = 2.0 * (alignment - 0.8) * 5
                mech.health = min(mech.max_health, mech.health + health_boost)
        
        elif alignment < 0.2:
            # Low alignment - negative effects
            if random.random() < 0.1 * (0.2 - alignment) * 5:
                # Energy drain
                energy_drain = 3.0 * (0.2 - alignment) * 5
                mech.energy = max(0, mech.energy - energy_drain)
                
                # Minor damage
                damage = 1.0 * (0.2 - alignment) * 5
                mech.take_damage(damage, "quantum_field", "quantum")
    
    def _apply_boundary_constraints(self, mech: BattleMech) -> None:
        """Apply boundary constraints to keep mechs in the arena"""
        # Get mech position
        pos = mech.position
        
        # Apply horizontal boundaries
        if abs(pos.x) > self.size/2:
            # Bounce off boundary
            mech.velocity.x *= -0.5
            # Set position at boundary
            mech.position.x = math.copysign(self.size/2, pos.x)
        
        if abs(pos.y) > self.size/2:
            # Bounce off boundary
            mech.velocity.y *= -0.5
            # Set position at boundary
            mech.position.y = math.copysign(self.size/2, pos.y)
        
        # Apply vertical boundaries
        if pos.z < 0:
            # Bounce off ground
            mech.velocity.z *= -0.2
            # Set position at ground
            mech.position.z = 0
        
        if pos.z > self.size/4:  # Height limit at quarter of arena size
            # Bounce off ceiling
            mech.velocity.z *= -0.5
            # Set position at ceiling
            mech.position.z = self.size/4
    
    def end_battle(self, reason: str, winner_id: Optional[str] = None) -> Dict[str, Any]:
        """End the battle"""
        if not self.battle_active or self.battle_finished:
            return {
                "success": False,
                "message": "Battle not active or already finished"
            }
        
        self.battle_active = False
        self.battle_finished = True
        self.winner_id = winner_id
        
        # Calculate battle duration
        duration = self.current_time - self.start_time
        
        # Record battle outcome for all mechs
        for mech_id, mech in self.mechs.items():
            if winner_id and mech_id == winner_id:
                mech.victories += 1
                
                # Award experience for victory
                exp_gain = 100 + 10 * len(self.mechs)
                mech.gain_experience(exp_gain)
                
                # Chance for evolution progress
                mech.evolution_progress += 0.2
            else:
                if mech.health <= 0:
                    mech.defeats += 1
                else:
                    mech.draws += 1
                
                # Award experience for participation
                exp_gain = 50
                mech.gain_experience(exp_gain)
        
        # Log battle end
        self.battle_log.append({
            "action": "battle_end",
            "reason": reason,
            "winner_id": winner_id,
            "duration": duration,
            "mech_states": {mech_id: {"health": mech.health, "energy": mech.energy} 
                          for mech_id, mech in self.mechs.items()},
            "timestamp": self.current_time
        })
        
        return {
            "success": True,
            "message": f"Battle ended: {reason}",
            "winner_id": winner_id,
            "duration": duration,
            "active_mechs": sum(1 for mech in self.mechs.values() if mech.health > 0)
        }
    
    def get_battle_status(self) -> Dict[str, Any]:
        """Get current battle status"""

class DiffusionEvolution:
    """Implementation of the Diffusion Evolution algorithm from 'Diffusion Models are Evolutionary Algorithms'"""
    
    def __init__(self, population_size: int = 128, parameter_dim: int = 64, steps: int = 10):
        self.population_size = population_size
        self.parameter_dim = parameter_dim
        self.diffusion_steps = steps
        self.alpha_schedule = self._create_cosine_schedule(steps)
        self.noise_schedule = self._create_noise_schedule(steps)
        self.population = None
        self.fitness_values = None
        self.generation = 0
    
    def _create_cosine_schedule(self, steps: int) -> np.ndarray:
        """Create cosine schedule for alpha values"""
        return np.array([np.cos(np.pi * t / (2 * steps)) ** 2 for t in range(steps)])
    
    def _create_noise_schedule(self, steps: int) -> np.ndarray:
        """Create noise schedule"""
        return np.array([0.1 - 0.09 * (t / steps) for t in range(steps)])
    
    def initialize_population(self) -> None:
        """Initialize population with random values"""
        self.population = np.random.normal(0, 1, (self.population_size, self.parameter_dim))
        self.fitness_values = np.zeros(self.population_size)
        self.generation = 0
    
    def evaluate_fitness(self, fitness_function: Callable[[np.ndarray], float]) -> None:
        """Evaluate fitness for the entire population"""
        for i in range(self.population_size):
            self.fitness_values[i] = fitness_function(self.population[i])
    
    def evolve_generation(self) -> None:
        """Evolve population using Diffusion Evolution algorithm"""
        # Convert fitness to probability density (higher fitness = higher probability)
        fitness_min = np.min(self.fitness_values)
        fitness_max = np.max(self.fitness_values)
        
        if fitness_max > fitness_min:
            # Normalize fitness to [0, 1]
            normalized_fitness = (self.fitness_values - fitness_min) / (fitness_max - fitness_min)
        else:
            normalized_fitness = np.ones_like(self.fitness_values) / self.population_size
        
        # Apply softmax to get probability distribution
        exp_fitness = np.exp(normalized_fitness * 2)  # Temperature parameter = 2
        fitness_probs = exp_fitness / np.sum(exp_fitness)
        
        # Create new population
        new_population = np.zeros_like(self.population)
        
        for i in range(self.population_size):
            # Start with current individual
            xt = self.population[i].copy()
            
            # Apply diffusion steps in reverse (denoising)
            for t in range(self.diffusion_steps - 1, 0, -1):
                alpha_t = self.alpha_schedule[t]
                alpha_t_minus_1 = self.alpha_schedule[t-1]
                sigma_t = self.noise_schedule[t]
                
                # Calculate estimated x0 (high fitness parameter target)
                x0_estimate = np.zeros(self.parameter_dim)
                weights_sum = 0
                
                # Use fitness-weighted average of population as estimate
                for j in range(self.population_size):
                    # Calculate distance-based weight
                    distance = np.linalg.norm(xt - np.sqrt(alpha_t) * self.population[j])
                    distance_weight = np.exp(-distance**2 / (2 * (1 - alpha_t)))
                    
                    # Combine with fitness weight
                    weight = fitness_probs[j] * distance_weight
                    x0_estimate += weight * self.population[j]
                    weights_sum += weight
                
                # Normalize estimate
                if weights_sum > 0:
                    x0_estimate /= weights_sum
                else:
                    # Fallback if all weights are too small
                    x0_estimate = xt / np.sqrt(alpha_t)
                
                # Calculate error term
                error_estimate = (xt - np.sqrt(alpha_t) * x0_estimate) / np.sqrt(1 - alpha_t)
                
                # Compute xt-1 using DDIM-like formula
                xt_minus_1 = np.sqrt(alpha_t_minus_1) * x0_estimate + \
                             np.sqrt(1 - alpha_t_minus_1 - sigma_t**2) * error_estimate + \
                             sigma_t * np.random.normal(0, 1, self.parameter_dim)
                
                # Update xt for next iteration
                xt = xt_minus_1
            
            # Store final result in new population
            new_population[i] = xt
        
        # Update population
        self.population = new_population
        self.generation += 1

            ))
            
            if resonance > max_resonance:
                max_resonance = resonance
                max_position = (i, j, k)
        
        # Create node if resonance is high enough
        if max_resonance > 0.7:
            i, j, k = max_position
            
            # Create resonance pattern by blending fields
            pattern = 0.5 * self.primary_field[i, j, k] + 0.5 * self.harmonic_field[i, j, k]
            pattern = pattern / np.linalg.norm(pattern)
            
            # Add some random variation
            pattern += np.random.normal(0, 0.1, self.quantum_dims)
            pattern = pattern / np.linalg.norm(pattern)
            
            # Create node
            node = {
                "position": max_position,
                "pattern": pattern,
                "radius": random.randint(1, 3),
                "strength": 0.2 + 0.3 * max_resonance,
                "duration": 10.0 + 20.0 * max_resonance,
                "initial_duration": 10.0 + 20.0 * max_resonance,
                "type": random.choice(ROBOWARS_CONSTANTS["ELEMENTAL_TYPES"])
            }
            
            self.resonance_nodes.append(node)
    
    def get_field_at_position(self, position: RWVector) -> np.ndarray:
        """Get the quantum field at a specific position"""
        # Convert world position to grid indices
        grid_size = 1000.0 / self.size  # Assuming arena size of 1000
        i = int((position.x + 500.0) / grid_size)
        j = int((position.y + 500.0) / grid_size)
        k = int(position.z / grid_size)
        
        # Clamp to valid range
        i = max(0, min(self.size - 1, i))
        j = max(0, min(self.size - 1, j))
        k = max(0, min(self.size - 1, k))
        
        return self.primary_field[i, j, k]

### 4. Latent Space Diffusion Evolution (LSDE)
For high-dimensional parameter optimization:

```python
class LatentSpaceDiffusionEvolution:
    """Latent Space Diffusion Evolution algorithm for high-dimensional parameter optimization"""
    
    def __init__(self, high_dim: int = 256, latent_dim: int = 2, population_size: int = 128, steps: int = 10):
        self.high_dim = high_dim
        self.latent_dim = latent_dim
        self.population_size = population_size
        self.diffusion_steps = steps
        self.alpha_schedule = self._create_cosine_schedule(steps)
        self.noise_schedule = self._create_noise_schedule(steps)
        
        # Initialize projection matrices
        self.projection = self._create_projection_matrix()
        self.projection_inv = self._create_inverse_projection()
        
        # Initialize populations
        self.high_dim_population = None
        self.latent_population = None
        self.fitness_values = None
        self.generation = 0
    
    def _create_cosine_schedule(self, steps: int) -> np.ndarray:
        """Create cosine schedule for alpha values"""
        return np.array([np.cos(np.pi * t / (2 * steps)) ** 2 for t in range(steps)])
    
    def _create_noise_schedule(self, steps: int) -> np.ndarray:
        """Create noise schedule"""
        return np.array([0.1 - 0.09 * (t / steps) for t in range(steps)])
    
    def _create_projection_matrix(self) -> np.ndarray:
        """Create random projection matrix for dimensionality reduction"""
        # Create random projection matrix
        projection = np.random.normal(0, 1.0 / np.sqrt(self.latent_dim), (self.high_dim, self.latent_dim))
        
        # Orthogonalize using QR decomposition
        q, r = np.linalg.qr(projection)
        
        return q[:, :self.latent_dim]
    
    def _create_inverse_projection(self) -> np.ndarray:
        """Create pseudo-inverse projection matrix"""
        return self.projection.T
    
    def project_to_latent(self, high_dim_params: np.ndarray) -> np.ndarray:
        """Project high-dimensional parameters to latent space"""
        return np.dot(high_dim_params, self.projection)
    
    def project_to_high_dim(self, latent_params: np.ndarray) -> np.ndarray:
        """Project latent parameters to high-dimensional space"""
        return np.dot(latent_params, self.projection_inv)
    
    def initialize_population(self) -> None:
        """Initialize population with random values"""
        # Initialize high-dimensional population
        self.high_dim_population = np.random.normal(0, 1, (self.population_size, self.high_dim))
        
        # Project to latent space
        self.latent_population = np.zeros((self.population_size, self.latent_dim))
        for i in range(self.population_size):
            self.latent_population[i] = self.project_to_latent(self.high_dim_population[i])
        
        self.fitness_values = np.zeros(self.population_size)
        self.generation = 0
    
    def evaluate_fitness(self, fitness_function: Callable[[np.ndarray], float]) -> None:
        """Evaluate fitness for the entire population"""
        for i in range(self.population_size):
            self.fitness_values[i] = fitness_function(self.high_dim_population[i])
    
    def evolve_generation(self) -> None:
        """Evolve population using Latent Space Diffusion Evolution"""
        # Convert fitness to probability density (higher fitness = higher probability)
        fitness_min = np.min(self.fitness_values)
        fitness_max = np.max(self.fitness_values)
        
        if fitness_max > fitness_min:
            # Normalize fitness to [0, 1]
            normalized_fitness = (self.fitness_values - fitness_min) / (fitness_max - fitness_min)
        else:
            normalized_fitness = np.ones_like(self.fitness_values) / self.population_size
        
        # Apply softmax to get probability distribution
        exp_fitness = np.exp(normalized_fitness * 2)  # Temperature parameter = 2
        fitness_probs = exp_fitness / np.sum(exp_fitness)
        
        # Create new latent population
        new_latent_population = np.zeros_like(self.latent_population)
        
        for i in range(self.population_size):
            # Start with current latent individual
            xt = self.latent_population[i].copy()
            
            # Apply diffusion steps in reverse (denoising)
            for t in range(self.diffusion_steps - 1, 0, -1):
                alpha_t = self.alpha_schedule[t]
                alpha_t_minus_1 = self.alpha_schedule[t-1]
                sigma_t = self.noise_schedule[t]
                
                # Calculate estimated x0 (high fitness parameter target)
                x0_estimate = np.zeros(self.latent_dim)
                weights_sum = 0
                
                # Use fitness-weighted average of population as estimate
                for j in range(self.population_size):
                    # Calculate distance-based weight
                    distance = np.linalg.norm(xt - np.sqrt(alpha_t) * self.latent_population[j])
                    distance_weight = np.exp(-distance**2 / (2 * (1 - alpha_t)))
                    
                    # Combine with fitness weight
                    weight = fitness_probs[j] * distance_weight
                    x0_estimate += weight * self.latent_population[j]
                    weights_sum += weight
                
                # Normalize estimate
                if weights_sum > 0:
                    x0_estimate /= weights_sum
                else:
                    # Fallback if all weights are too small
                    x0_estimate = xt / np.sqrt(alpha_t)
                
                # Calculate error term
                error_estimate = (xt - np.sqrt(alpha_t) * x0_estimate) / np.sqrt(1 - alpha_t)
                
                # Compute xt-1 using DDIM-like formula
                xt_minus_1 = np.sqrt(alpha_t_minus_1) * x0_estimate + \
                             np.sqrt(1 - alpha_t_minus_1 - sigma_t**2) * error_estimate + \
                             sigma_t * np.random.normal(0, 1, self.latent_dim)
                
                # Update xt for next iteration
                xt = xt_minus_1
            
            # Store final result in new population
            new_latent_population[i] = xt
        
        # Update latent population
        self.latent_population = new_latent_population
        
        # Project back to high-dimensional space
        for i in range(self.population_size):
            self.high_dim_population[i] = self.project_to_high_dim(self.latent_population[i])
        
        self.generation += 1
```
