import pybullet as p
import numpy as np
import math
import random

@dataclass
class SuperpowerEffect:
    """Defines the effect of a superpower on physics parameters"""
    name: str
    duration: float  # in seconds
    cooldown: float  # in seconds
    energy_cost: float  # 0-100 scale
    integrity_cost: float  # percentage of integrity lost when used
    
    # Physics parameter modifiers (multiplicative)
    mass_modifier: float = 1.0
    friction_modifier: float = 1.0
    restitution_modifier: float = 1.0
    gravity_modifier: float = 1.0
    
    # Special effect function
    special_effect: Optional[Callable] = None
    
    # Visual effect parameters
    visual_effect_id: Optional[str] = None
    sound_effect_id: Optional[str] = None

class QuantumSuperpower:
    """Manages quantum-inspired superpowers for RoboBladez"""
    
    def __init__(self, simulator_backend='statevector_simulator'):
        """Initialize the quantum superpower system"""
        self.simulator = Aer.get_backend(simulator_backend)
        self.active_superpowers = {}  # blade_id -> {superpower_name: end_time}
        self.cooldowns = {}  # blade_id -> {superpower_name: end_cooldown_time}
        self.blade_states = {}  # blade_id -> original physics state
        self.time_step = 0.01  # physics simulation time step
        
        # Register built-in superpowers
        self.superpower_registry = {
            "quantum_superposition": self._create_quantum_superposition(),
            "quantum_tunneling": self._create_quantum_tunneling(),
            "quantum_entanglement": self._create_quantum_entanglement(),
            "decoherence_blast": self._create_decoherence_blast(),
            "probability_amplification": self._create_probability_amplification(),
        }
    
    def _create_quantum_superposition(self) -> SuperpowerEffect:
        """
        Creates a superposition effect where the RoboBlade exists in multiple states
        simultaneously for a short duration
        """
        def superposition_effect(blade_id, blade_data, arena_data, time):
            # Run a simple quantum circuit to generate a superposition state
            qc = QuantumCircuit(2, 2)
            qc.h(0)  # Apply Hadamard to create superposition
            qc.h(1)  # Apply Hadamard to create superposition
            
            # Simulate and get state vector
            result = execute(qc, self.simulator).result()
            state_vector = result.get_statevector()
            
            # Calculate probabilities for each state
            probabilities = np.abs(state_vector)**2
            
            # Create phantom instances (only visual, actual physics still uses original)
            blade_pos, blade_orn = p.getBasePositionAndOrientation(blade_id)
            blade_vel, blade_ang_vel = p.getBaseVelocity(blade_id)
            
            phantom_ids = []
            for i, prob in enumerate(probabilities):
                if prob > 0.05:  # Only create phantoms for significant probabilities
                    # Offset position based on state index
                    x_offset = 0.05 * ((i & 1) * 2 - 1)
                    y_offset = 0.05 * (((i >> 1) & 1) * 2 - 1)
                    
                    # Create visual-only phantom
                    phantom_pos = (blade_pos[0] + x_offset, blade_pos[1] + y_offset, blade_pos[2])
                    
                    # Adjust physics parameters for the phantom
                    phantom_id = p.createMultiBody(
                        baseMass=0.01,  # Almost no mass to minimize physics impact
                        baseCollisionShapeIndex=-1,  # No collision
                        baseVisualShapeIndex=p.getVisualShapeData(blade_id)[0][1],
                        basePosition=phantom_pos,
                        baseOrientation=blade_orn
                    )
                    
                    # Apply velocity and angular velocity
                    p.resetBaseVelocity(
                        phantom_id,
                        linearVelocity=blade_vel,
                        angularVelocity=blade_ang_vel
                    )
                    
                    # Set transparency based on probability
                    alpha = min(0.7, prob + 0.3)
                    p.changeVisualShape(phantom_id, -1, rgbaColor=[1, 1, 1, alpha])
                    
                    phantom_ids.append(phantom_id)
            
            # Store phantom IDs for cleanup
            if blade_id not in self.blade_states:
                self.blade_states[blade_id] = {}
            self.blade_states[blade_id]["phantom_ids"] = phantom_ids
            
            # For actual physics effect, make blade harder to predict
            # Randomly adjust velocity by small amounts to simulate quantum uncertainty
            random_vel_modifier = [random.uniform(-0.5, 0.5) for _ in range(3)]
            current_vel = list(blade_vel)
            new_vel = [current_vel[i] + random_vel_modifier[i] for i in range(3)]
            p.resetBaseVelocity(blade_id, linearVelocity=new_vel)
            
            return True  # Effect successfully applied
        
        return SuperpowerEffect(
            name="Quantum Superposition",
            duration=2.5,
            cooldown=10.0,
            energy_cost=50.0,
            integrity_cost=5.0,
            mass_modifier=0.8,  # Slightly reduced mass during superposition
            friction_modifier=0.6,  # Reduced friction
            restitution_modifier=1.2,  # Enhanced bounce
            gravity_modifier=0.9,  # Slightly reduced gravity
            special_effect=superposition_effect,
            visual_effect_id="superposition_glow",
            sound_effect_id="quantum_hum"
        )
    
    def _create_quantum_tunneling(self) -> SuperpowerEffect:
        """
        Creates a tunneling effect where the RoboBlade can temporarily pass through
        obstacles with a probability based on quantum tunneling equations
        """
        def tunneling_effect(blade_id, blade_data, arena_data, time):
            # Store original collision state
            if blade_id not in self.blade_states:
                self.blade_states[blade_id] = {}
            
            # Get original collision shape
            collision_shapes = p.getCollisionShapeData(blade_id, -1)
            self.blade_states[blade_id]["original_collision"] = collision_shapes
            
            # Temporarily disable collisions
            p.setCollisionFilterGroupMask(blade_id, -1, 0, 0)
            
            # Apply visual effect - make blade semi-transparent
            p.changeVisualShape(blade_id, -1, rgbaColor=[0.3, 0.7, 1.0, 0.7])
            
            # Calculate tunneling probability based on "barrier" thickness
            # In a real quantum system, tunneling probability decreases exponentially with barrier width
            # Here we simulate a simplified version
            tunneling_probability = np.exp(-0.5)  # ~0.61 probability of successful tunneling
            
            # Store additional state for tunneling
            self.blade_states[blade_id]["tunneling_probability"] = tunneling_probability
            self.blade_states[blade_id]["tunneling_successful"] = random.random() < tunneling_probability
            
            return True
        
        return SuperpowerEffect(
            name="Quantum Tunneling",
            duration=1.0,
            cooldown=15.0,
            energy_cost=70.0,
            integrity_cost=8.0,
            mass_modifier=0.5,  # Half mass during tunneling
            friction_modifier=0.1,  # Almost no friction
            gravity_modifier=0.7,  # Reduced gravity
            special_effect=tunneling_effect,
            visual_effect_id="tunnel_wave",
            sound_effect_id="phase_shift"
        )
    
    def _create_quantum_entanglement(self) -> SuperpowerEffect:
        """
        Creates an entanglement between two RoboBladez, causing their states to be
        correlated even when separated in the arena
        """
        def entanglement_effect(blade_id, blade_data, arena_data, time):
            # Find closest opponent to entangle with
            closest_opponent = None
            min_distance = float('inf')
            blade_pos, _ = p.getBasePositionAndOrientation(blade_id)
            
            for other_id in arena_data.get("active_blades", []):
                if other_id != blade_id:
                    other_pos, _ = p.getBasePositionAndOrientation(other_id)
                    distance = math.sqrt(sum((a - b) ** 2 for a, b in zip(blade_pos, other_pos)))
                    if distance < min_distance:
                        min_distance = distance
                        closest_opponent = other_id
            
            if closest_opponent is None or min_distance > 0.5:
                # No opponent close enough to entangle with
                return False
            
            # Create entanglement
            if blade_id not in self.blade_states:
                self.blade_states[blade_id] = {}
            
            # Store entanglement relationship
            self.blade_states[blade_id]["entangled_with"] = closest_opponent
            
            # Create quantum circuit for entangled states
            qc = QuantumCircuit(2, 2)
            qc.h(0)  # Put qubit 0 in superposition
            qc.cx(0, 1)  # CNOT gate to entangle qubits
            
            # Run circuit and store results
            result = execute(qc, self.simulator).result()
            state_vector = result.get_statevector()
            
            # Store state vector for later use in entanglement effects
            self.blade_states[blade_id]["entanglement_state"] = state_vector
            
            # Apply visual effect to show entanglement
            # Create a visible "tether" between the two RoboBladez
            blade_pos, _ = p.getBasePositionAndOrientation(blade_id)
            other_pos, _ = p.getBasePositionAndOrientation(closest_opponent)
            
            # Create line visual between entangled RoboBladez
            line_id = p.addUserDebugLine(
                blade_pos,
                other_pos,
                lineColorRGB=[0.8, 0.2, 0.8],  # Purple color
                lineWidth=3.0,
                lifeTime=0.0  # Persistent until removed
            )
            
            self.blade_states[blade_id]["entanglement_line"] = line_id
            
            # Mirror the entanglement on the other blade
            if closest_opponent not in self.blade_states:
                self.blade_states[closest_opponent] = {}
            self.blade_states[closest_opponent]["entangled_with"] = blade_id
            self.blade_states[closest_opponent]["entanglement_state"] = state_vector
            
            return True
        
        return SuperpowerEffect(
            name="Quantum Entanglement",
            duration=8.0,
            cooldown=20.0,
            energy_cost=60.0,
            integrity_cost=5.0,
            special_effect=entanglement_effect,
            visual_effect_id="entanglement_beam",
            sound_effect_id="quantum_link"
        )
    
    def _create_decoherence_blast(self) -> SuperpowerEffect:
        """
        Creates a decoherence blast that disrupts other quantum superpowers
        and causes instability in nearby RoboBladez
        """
        def decoherence_effect(blade_id, blade_data, arena_data, time):
            # Get blade position
            blade_pos, _ = p.getBasePositionAndOrientation(blade_id)
            
            # Define blast radius
            blast_radius = 0.8
            
            # Affect all blades within radius
            affected_blades = []
            for other_id in arena_data.get("active_blades", []):
                if other_id != blade_id:
                    other_pos, _ = p.getBasePositionAndOrientation(other_id)
                    distance = math.sqrt(sum((a - b) ** 2 for a, b in zip(blade_pos, other_pos)))
                    
                    if distance <= blast_radius:
                        affected_blades.append(other_id)
                        
                        # Cancel active superpowers on affected blade
                        if other_id in self.active_superpowers:
                            self.clear_superpower_effects(other_id)
                        
                        # Apply destabilizing effect
                        current_vel, current_ang_vel = p.getBaseVelocity(other_id)
                        
                        # Random disruption to velocity and angular velocity
                        disrupt_factor = 1.0 - (distance / blast_radius)  # Stronger effect closer to center
                        random_vel = [random.uniform(-3, 3) * disrupt_factor for _ in range(3)]
                        random_ang = [random.uniform(-5, 5) * disrupt_factor for _ in range(3)]
                        
                        # Apply disrupted velocity
                        new_vel = [current_vel[i] * 0.5 + random_vel[i] for i in range(3)]
                        new_ang_vel = [current_ang_vel[i] * 0.5 + random_ang[i] for i in range(3)]
                        
                        p.resetBaseVelocity(
                            other_id,
                            linearVelocity=new_vel,
                            angularVelocity=new_ang_vel
                        )
            
            # Create visual blast effect
            p.addUserDebugLine(
                [blade_pos[0] - blast_radius, blade_pos[1], blade_pos[2]],
                [blade_pos[0] + blast_radius, blade_pos[1], blade_pos[2]],
                lineColorRGB=[1.0, 0.5, 0.0],
                lineWidth=2.0,
                lifeTime=0.3
            )
            p.addUserDebugLine(
                [blade_pos[0], blade_pos[1] - blast_radius, blade_pos[2]],
                [blade_pos[0], blade_pos[1] + blast_radius, blade_pos[2]],
                lineColorRGB=[1.0, 0.5, 0.0],
                lineWidth=2.0,
                lifeTime=0.3
            )
            
            # Store affected blades for cleanup
            if blade_id not in self.blade_states:
                self.blade_states[blade_id] = {}
            self.blade_states[blade_id]["affected_blades"] = affected_blades
            
            return True
        
        return SuperpowerEffect(
            name="Decoherence Blast",
            duration=0.5,  # Short duration as it's an instant effect
            cooldown=12.0,
            energy_cost=80.0,
            integrity_cost=10.0,
            special_effect=decoherence_effect,
            visual_effect_id="decoherence_wave",
            sound_effect_id="reality_crack"
        )

            original_chance = arena_data.get("chance_vector", [1.0, 0.0, 0.0])
            self.blade_states[blade_id]["original_chance"] = original_chance
            
            # Create quantum circuit to calculate enhanced probability
            qc = QuantumCircuit(3, 3)
            
            # Apply amplitude amplification (similar to Grover's algorithm)
            for i in range(3):
                qc.h(i)  # Put all qubits in superposition
            
            # Apply oracle (mark desired states)
            qc.x(0)
            qc.x(1)
            qc.h(2)
            qc.ccx(0, 1, 2)  # Controlled-controlled-X (Toffoli)
            qc.h(2)
            qc.x(0)
            qc.x(1)
            
            # Apply diffusion operator
            for i in range(3):
                qc.h(i)
                qc.x(i)
            qc.h(2)
            qc.ccx(0, 1, 2)
            qc.h(2)
            for i in range(3):
                qc.x(i)
                qc.h(i)
            
            # Simulate and get probabilities
            result = execute(qc, self.simulator).result()
            counts = result.get_counts()
            
            # Calculate enhanced chance vector
            enhanced_chance = [
                1.5,  # Significantly increase luck
                original_chance[1] * 0.5,  # Reduce negative friction variations
                original_chance[2] * 0.5   # Reduce negative quirks

            # Store enhanced chance
            self.blade_states[blade_id]["enhanced_chance"] = enhanced_chance
            
            # Apply visual effect - particles moving toward the blade

            self.blade_states[blade_id]["particle_ids"] = particle_ids
            return True
        
        return SuperpowerEffect(

    def activate_superpower(self, blade_id: int, superpower_name: str, blade_data: Dict, arena_data: Dict) -> bool:
        """Activate a superpower for a RoboBlade"""
        # Check if superpower can be activated
        if not self.can_activate_superpower(blade_id, superpower_name, blade_data):
            return False
        
        # Get superpower effect
        effect = self.superpower_registry[superpower_name]
        
        # Deduct costs
        blade_data["energy"] -= effect.energy_cost
        blade_data["integrity"] -= effect.integrity_cost
        
        # Store original physics parameters if not already stored
        if blade_id not in self.blade_states:
            self.blade_states[blade_id] = self._store_physics_state(blade_id)
        
        # Apply physics modifiers
        self._apply_physics_modifiers(blade_id, effect)
        
        # Apply special effect if defined
        current_time = blade_data.get("simulation_time", 0)
        if effect.special_effect:
            effect.special_effect(blade_id, blade_data, arena_data, current_time)
        
        # Set active superpower and cooldown
        if blade_id not in self.active_superpowers:
            self.active_superpowers[blade_id] = {}
        if blade_id not in self.cooldowns:
            self.cooldowns[blade_id] = {}
        
        self.active_superpowers[blade_id][superpower_name] = current_time + effect.duration
        self.cooldowns[blade_id][superpower_name] = current_time + effect.duration + effect.cooldown
        
        return True
    
    def _store_physics_state(self, blade_id: int) -> Dict:
        """Store original physics state of a RoboBlade"""
        dynamics_info = p.getDynamicsInfo(blade_id, -1)
        return {
            "original_mass": dynamics_info[0],
            "original_lateral_friction": dynamics_info[1],
            "original_restitution": dynamics_info[5],
            "original_gravity": p.getGravity()
        }
    
    def _apply_physics_modifiers(self, blade_id: int, effect: SuperpowerEffect) -> None:
        """Apply physics modifiers from superpower effect"""
        # Get original values
        original_state = self.blade_states[blade_id]
        
        # Apply modifiers
        p.changeDynamics(
            blade_id, -1,
            mass=original_state["original_mass"] * effect.mass_modifier,
            lateralFriction=original_state["original_lateral_friction"] * effect.friction_modifier,
            restitution=original_state["original_restitution"] * effect.restitution_modifier
        )
        
        # Apply gravity modifier if not 1.0
        if effect.gravity_modifier != 1.0:
            original_gravity = original_state["original_gravity"]
            modified_gravity = [g * effect.gravity_modifier for g in original_gravity]
            p.setGravity(*modified_gravity)
    
    def update(self, blade_data_dict: Dict[int, Dict], arena_data: Dict) -> None:
        """Update all active superpowers"""
        current_time = arena_data.get("simulation_time", 0)
        expired_powers = []
        
        # Check for expired superpowers
        for blade_id, active_powers in self.active_superpowers.items():
            for power_name, end_time in active_powers.items():
                if current_time >= end_time:
                    expired_powers.append((blade_id, power_name))
                else:
                    # Handle special ongoing effects that need updates
                    effect = self.superpower_registry[power_name]
                    if power_name == "quantum_entanglement" and effect.special_effect:
                        # Update entanglement effects (e.g., update tether line position)
                        self._update_entanglement(blade_id, blade_data_dict.get(blade_id, {}), arena_data)
        
        # Remove expired superpowers
        for blade_id, power_name in expired_powers:
            self.clear_superpower_effect(blade_id, power_name)
    
    def _update_entanglement(self, blade_id: int, blade_data: Dict, arena_data: Dict) -> None:
        """Update entanglement visual and effects"""
        if blade_id not in self.blade_states or "entangled_with" not in self.blade_states[blade_id]:
            return
        
        entangled_id = self.blade_states[blade_id]["entangled_with"]
        if entangled_id not in self.blade_states:
            return
        
        # Update visual tether
        if "entanglement_line" in self.blade_states[blade_id]:
            line_id = self.blade_states[blade_id]["entanglement_line"]
            p.removeUserDebugItem(line_id)
            
            # Create new line with updated positions
            blade_pos, _ = p.getBasePositionAndOrientation(blade_id)
            other_pos, _ = p.getBasePositionAndOrientation(entangled_id)
            
            new_line_id = p.addUserDebugLine(
                blade_pos,
                other_pos,
                lineColorRGB=[0.8, 0.2, 0.8],
                lineWidth=3.0,
                lifeTime=0.0
            )
            
            self.blade_states[blade_id]["entanglement_line"] = new_line_id
        
        # Implement entanglement effects
        # 1. Velocity correlation: if one blade speeds up, the other follows
        blade_vel, blade_ang_vel = p.getBaseVelocity(blade_id)
        other_vel, other_ang_vel = p.getBaseVelocity(entangled_id)
        
        # Calculate average linear and angular velocities
        avg_vel = [(a + b) * 0.5 for a, b in zip(blade_vel, other_vel)]
        avg_ang_vel = [(a + b) * 0.5 for a, b in zip(blade_ang_vel, other_ang_vel)]
        
        # Apply correlated velocities (with some random variation to make it interesting)
        blade_random = [random.uniform(-0.1, 0.1) for _ in range(3)]
        other_random = [random.uniform(-0.1, 0.1) for _ in range(3)]
        
        p.resetBaseVelocity(
            blade_id,
            linearVelocity=[avg_vel[i] + blade_random[i] for i in range(3)],
            angularVelocity=avg_ang_vel
        )
        
        p.resetBaseVelocity(
            entangled_id,
            linearVelocity=[avg_vel[i] + other_random[i] for i in range(3)],
            angularVelocity=avg_ang_vel
        )
    
    def clear_superpower_effect(self, blade_id: int, power_name: str) -> None:
        """Clear a specific superpower effect"""
        if blade_id not in self.active_superpowers or power_name not in self.active_superpowers[blade_id]:
            return
        
        # Remove from active superpowers
        del self.active_superpowers[blade_id][power_name]
        
        # Get original state
        original_state = self.blade_states.get(blade_id, {})
        if not original_state:
            return
        
        # Restore original physics parameters
        if "original_mass" in original_state:
            p.changeDynamics(
                blade_id, -1,
                mass=original_state["original_mass"],
                lateralFriction=original_state["original_lateral_friction"],
                restitution=original_state["original_restitution"]
            )
        
        if "original_gravity" in original_state:
            p.setGravity(*original_state["original_gravity"])
        
        # Cleanup special effect artifacts
        self._cleanup_power_artifacts(blade_id, power_name)
    
    def clear_superpower_effects(self, blade_id: int) -> None:
        """Clear all superpower effects for a blade"""
        if blade_id not in self.active_superpowers:
            return
        
        active_powers = list(self.active_superpowers[blade_id].keys())
        for power_name in active_powers:
            self.clear_superpower_effect(blade_id, power_name)
    
    def _cleanup_power_artifacts(self, blade_id: int, power_name: str) -> None:
        """Clean up artifacts created by superpowers"""
        if blade_id not in self.blade_states:
            return
        
        # Superposition cleanup
        if power_name == "quantum_superposition" and "phantom_ids" in self.blade_states[blade_id]:
            for phantom_id in self.blade_states[blade_id]["phantom_ids"]:
                p.removeBody(phantom_id)
            del self.blade_states[blade_id]["phantom_ids"]
        
        # Tunneling cleanup
        if power_name == "quantum_tunneling":
            # Restore collision properties
            p.setCollisionFilterGroupMask(blade_id, -1, 1, 0)  # Reset to default group
            
            # Restore original appearance
            p.changeVisualShape(blade_id, -1, rgbaColor=[1.0, 1.0, 1.0, 1.0])
        
        # Entanglement cleanup
        if power_name == "quantum_entanglement" and "entanglement_line" in self.blade_states[blade_id]:
            p.removeUserDebugItem(self.blade_states[blade_id]["entanglement_line"])
            
            if "entangled_with" in self.blade_states[blade_id]:
                entangled_id = self.blade_states[blade_id]["entangled_with"]
                if entangled_id in self.blade_states and "entangled_with" in self.blade_states[entangled_id]:
                    del self.blade_states[entangled_id]["entangled_with"]
            
            del self.blade_states[blade_id]["entanglement_line"]
            del self.blade_states[blade_id]["entangled_with"]
            
        # Probability amplification cleanup
        if power_name == "probability_amplification" and "particle_ids" in self.blade_states[blade_id]:
            for particle_id in self.blade_states[blade_id]["particle_ids"]:
                p.removeUserDebugItem(particle_id)
            del self.blade_states[blade_id]["particle_ids"]

class SuperpowerTrigger:
    """Defines triggers for automatic superpower activation based on battle conditions"""
    
    def __init__(self):
        self.triggers = {}  # blade_id -> {trigger_name: trigger_condition}
    
    def add_trigger(self, blade_id: int, superpower_name: str, condition_function: Callable) -> None:
        """Add a trigger for a superpower"""
        if blade_id not in self.triggers:
            self.triggers[blade_id] = {}
        
        self.triggers[blade_id][superpower_name] = condition_function
    
    def check_triggers(self, blade_id: int, blade_data: Dict, arena_data: Dict, quantum_powers: QuantumSuperpower) -> None:
        """Check all triggers for a blade and activate superpowers if conditions are met"""
        if blade_id not in self.triggers:
            return
        
        for power_name, condition_func in self.triggers[blade_id].items():
            # Check if trigger condition is met
            if condition_func(blade_data, arena_data):
                # Activate superpower if possible
                quantum_powers.activate_superpower(blade_id, power_name, blade_data, arena_data)

# Example trigger conditions
def low_health_trigger(blade_data: Dict, arena_data: Dict) -> bool:
    """Trigger when blade integrity is below 30%"""
    return blade_data.get("integrity", 100) < 30

def center_control_trigger(blade_data: Dict, arena_data: Dict) -> bool:
    """Trigger when blade is near the center of the arena"""
    blade_pos = blade_data.get("position", [0, 0, 0])
    distance_from_center = math.sqrt(blade_pos[0]**2 + blade_pos[1]**2)
    return distance_from_center < 0.3

def opponent_nearby_trigger(blade_data: Dict, arena_data: Dict) -> bool:
    """Trigger when an opponent is nearby"""
    blade_id = blade_data.get("id", -1)
    blade_pos = blade_data.get("position", [0, 0, 0])
    
    for other_id in arena_data.get("active_blades", []):
        if other_id != blade_id:
            other_data = arena_data.get("blade_data", {}).get(other_id, {})
            other_pos = other_data.get("position", [0, 0, 0])
            distance = math.sqrt(sum((a - b)**2 for a, b in zip(blade_pos, other_pos)))
            if distance < 0.4:
                return True
    
    return False

    def _create_quantum_superposition(self) -> SuperpowerEffect:
        """
        Creates a superposition effect where the RoboBlade exists in multiple states
        simultaneously for a short duration
        """
        def superposition_effect(blade_id, blade_data, arena_data, time):
            # Run a simple quantum circuit to generate a superposition state
            qc = QuantumCircuit(2, 2)
            qc.h(0)  # Apply Hadamard to create superposition
            qc.h(1)  # Apply Hadamard to create superposition
            
            # Simulate and get state vector
            result = execute(qc, self.simulator).result()
            state_vector = result.get_statevector()
            
            # Calculate probabilities for each state
            probabilities = np.abs(state_vector)**2
            
            # Create phantom instances (only visual, actual physics still uses original)
            blade_pos, blade_orn = p.getBasePositionAndOrientation(blade_id)
            blade_vel, blade_ang_vel = p.getBaseVelocity(blade_id)
            
            phantom_ids = []
            for i, prob in enumerate(probabilities):
                if prob > 0.05:  # Only create phantoms for significant probabilities
                    # Offset position based on state index
                    x_offset = 0.05 * ((i & 1) * 2 - 1)
                    y_offset = 0.05 * (((i >> 1) & 1) * 2 - 1)
                    
                    # Create visual-only phantom
                    phantom_pos = (blade_pos[0] + x_offset, blade_pos[1] + y_offset, blade_pos[2])
                    
                    # Adjust physics parameters for the phantom
                    phantom_id = p.createMultiBody(
                        baseMass=0.01,  # Almost no mass to minimize physics impact
                        baseCollisionShapeIndex=-1,  # No collision
                        baseVisualShapeIndex=p.getVisualShapeData(blade_id)[0][1],
                        basePosition=phantom_pos,
                        baseOrientation=blade_orn
                    )
                    
                    # Apply velocity and angular velocity
                    p.resetBaseVelocity(
                        phantom_id,
                        linearVelocity=blade_vel,
                        angularVelocity=blade_ang_vel
                    )
                    
                    # Set transparency based on probability
                    alpha = min(0.7, prob + 0.3)
                    p.changeVisualShape(phantom_id, -1, rgbaColor=[1, 1, 1, alpha])
                    
                    phantom_ids.append(phantom_id)
            
            # Store phantom IDs for cleanup
            if blade_id not in self.blade_states:
                self.blade_states[blade_id] = {}
            self.blade_states[blade_id]["phantom_ids"] = phantom_ids
            
            # For actual physics effect, make blade harder to predict
            # Randomly adjust velocity by small amounts to simulate quantum uncertainty
            random_vel_modifier = [random.uniform(-0.5, 0.5) for _ in range(3)]
            current_vel = list(blade_vel)
            new_vel = [current_vel[i] + random_vel_modifier[i] for i in range(3)]
            p.resetBaseVelocity(blade_id, linearVelocity=new_vel)
            
            return True  # Effect successfully applied
        
        return SuperpowerEffect(

    def _create_quantum_entanglement(self) -> SuperpowerEffect:
        """

        correlated even when separated in the arena
        """
        def entanglement_effect(blade_id, blade_data, arena_data, time):
            # Find closest opponent to entangle with
            closest_opponent = None
            min_distance = float('inf')
            blade_pos, _ = p.getBasePositionAndOrientation(blade_id)
            
            for other_id in arena_data.get("active_blades", []):
                if other_id != blade_id:
                    other_pos, _ = p.getBasePositionAndOrientation(other_id)
                    distance = math.sqrt(sum((a - b) ** 2 for a, b in zip(blade_pos, other_pos)))
                    if distance < min_distance:
                        min_distance = distance
                        closest_opponent = other_id
            
            if closest_opponent is None or min_distance > 0.5:
                # No opponent close enough to entangle with
                return False
            
            # Create entanglement
            if blade_id not in self.blade_states:
                self.blade_states[blade_id] = {}
            
            # Store entanglement relationship
            self.blade_states[blade_id]["entangled_with"] = closest_opponent
            
            # Create quantum circuit for entangled states
            qc = QuantumCircuit(2, 2)
            qc.h(0)  # Put qubit 0 in superposition
            qc.cx(0, 1)  # CNOT gate to entangle qubits
            
            # Run circuit and store results
            result = execute(qc, self.simulator).result()
            state_vector = result.get_statevector()
            
            # Store state vector for later use in entanglement effects
            self.blade_states[blade_id]["entanglement_state"] = state_vector
            
            # Apply visual effect to show entanglement
            # Create a visible "tether" between the two RoboBladez
            blade_pos, _ = p.getBasePositionAndOrientation(blade_id)
            other_pos, _ = p.getBasePositionAndOrientation(closest_opponent)
            
            # Create line visual between entangled RoboBladez
            line_id = p.addUserDebugLine(

                other_pos,

                lineWidth=3.0,
                lifeTime=0.0  # Persistent until removed
            )
            
            self.blade_states[blade_id]["entanglement_line"] = line_id
            
            # Mirror the entanglement on the other blade
            if closest_opponent not in self.blade_states:
                self.blade_states[closest_opponent] = {}

    def _create_decoherence_blast(self) -> SuperpowerEffect:

            blast_radius = 0.8

            affected_blades = []
            for other_id in arena_data.get("active_blades", []):
                if other_id != blade_id:
                    other_pos, _ = p.getBasePositionAndOrientation(other_id)
                    distance = math.sqrt(sum((a - b) ** 2 for a, b in zip(blade_pos, other_pos)))
                    
                    if distance <= blast_radius:
                        affected_blades.append(other_id)
                        
                        # Cancel active superpowers on affected blade
                        if other_id in self.active_superpowers:
                            self.clear_superpower_effects(other_id)
                        
                        # Apply destabilizing effect
                        current_vel, current_ang_vel = p.getBaseVelocity(other_id)
                        
                        # Random disruption to velocity and angular velocity
                        disrupt_factor = 1.0 - (distance / blast_radius)  # Stronger effect closer to center
                        random_vel = [random.uniform(-3, 3) * disrupt_factor for _ in range(3)]
                        random_ang = [random.uniform(-5, 5) * disrupt_factor for _ in range(3)]
                        
                        # Apply disrupted velocity
                        new_vel = [current_vel[i] * 0.5 + random_vel[i] for i in range(3)]
                        new_ang_vel = [current_ang_vel[i] * 0.5 + random_ang[i] for i in range(3)]
                        
                        p.resetBaseVelocity(
                            other_id,
                            linearVelocity=new_vel,
                            angularVelocity=new_ang_vel
                        )
            
            # Create visual blast effect
            p.addUserDebugLine(
                [blade_pos[0] - blast_radius, blade_pos[1], blade_pos[2]],

        return SuperpowerEffect(
            name="Probability Amplification",
            duration=5.0,
            cooldown=15.0,
            energy_cost=60.0,
            integrity_cost=5.0,
            special_effect=probability_effect,
            visual_effect_id="probability_glow",
            sound_effect_id="reality_bend"
        )
    
    def can_activate_superpower(self, blade_id: int, superpower_name: str, blade_data: Dict) -> bool:
        """Check if a superpower can be activated"""
        # Check if superpower exists
        if superpower_name not in self.superpower_registry:
            return False
        
        # Check if superpower is on cooldown
        current_time = blade_data.get("simulation_time", 0)
        if blade_id in self.cooldowns and superpower_name in self.cooldowns[blade_id]:
            if current_time < self.cooldowns[blade_id][superpower_name]:
                return False
        
        # Check if blade has enough energy
        blade_energy = blade_data.get("energy", 0)
        energy_cost = self.superpower_registry[superpower_name].energy_cost
        if blade_energy < energy_cost:
            return False
        
        # Check if blade has enough integrity
        blade_integrity = blade_data.get("integrity", 100)
        integrity_cost = self.superpower_registry[superpower_name].integrity_cost
        if blade_integrity < integrity_cost:
            return False
        
        # All checks passed
        return True
    
    def activate_superpower(self, blade_id: int, superpower_name: str, blade_data: Dict, arena_data: Dict) -> bool:
        """Activate a superpower for a RoboBlade"""
        # Check if superpower can be activated
        if not self.can_activate_superpower(blade_id, superpower_name, blade_data):
            return False
        
        # Get superpower effect
        effect = self.superpower_registry[superpower_name]
        
        # Deduct costs
        blade_data["energy"] -= effect.energy_cost
        blade_data["integrity"] -= effect.integrity_cost
        
        # Store original physics parameters if not already stored
        if blade_id not in self.blade_states:
            self.blade_states[blade_id] = self._store_physics_state(blade_id)
        
        # Apply physics modifiers
        self._apply_physics_modifiers(blade_id, effect)
        
        # Apply special effect if defined
        current_time = blade_data.get("simulation_time", 0)
        if effect.special_effect:
            effect.special_effect(blade_id, blade_data, arena_data, current_time)
        
        # Set active superpower and cooldown
        if blade_id not in self.active_superpowers:
            self.active_superpowers[blade_id] = {}
        if blade_id not in self.cooldowns:
            self.cooldowns[blade_id] = {}
        
        self.active_superpowers[blade_id][superpower_name] = current_time + effect.duration
        self.cooldowns[blade_id][superpower_name] = current_time + effect.duration + effect.cooldown
        
        return True
    
    def _store_physics_state(self, blade_id: int) -> Dict:
        """Store original physics state of a RoboBlade"""
        dynamics_info = p.getDynamicsInfo(blade_id, -1)
        return {
            "original_mass": dynamics_info[0],
            "original_lateral_friction": dynamics_info[1],
            "original_restitution": dynamics_info[5],
            "original_gravity": p.getGravity()
        }
    
    def _apply_physics_modifiers(self, blade_id: int, effect: SuperpowerEffect) -> None:
        """Apply physics modifiers from superpower effect"""
        # Get original values
        original_state = self.blade_states[blade_id]
        
        # Apply modifiers
        p.changeDynamics(
            blade_id, -1,
            mass=original_state["original_mass"] * effect.mass_modifier,
            lateralFriction=original_state["original_lateral_friction"] * effect.friction_modifier,
            restitution=original_state["original_restitution"] * effect.restitution_modifier
        )
        
        # Apply gravity modifier if not 1.0
        if effect.gravity_modifier != 1.0:
            original_gravity = original_state["original_gravity"]
            modified_gravity = [g * effect.gravity_modifier for g in original_gravity]
            p.setGravity(*modified_gravity)
    
    def update(self, blade_data_dict: Dict[int, Dict], arena_data: Dict) -> None:
        """Update all active superpowers"""
        current_time = arena_data.get("simulation_time", 0)
        expired_powers = []
        
        # Check for expired superpowers
        for blade_id, active_powers in self.active_superpowers.items():
            for power_name, end_time in active_powers.items():
                if current_time >= end_time:
                    expired_powers.append((blade_id, power_name))
                else:
                    # Handle special ongoing effects that need updates
                    effect = self.superpower_registry[power_name]
                    if power_name == "quantum_entanglement" and effect.special_effect:
                        # Update entanglement effects (e.g., update tether line position)
                        self._update_entanglement(blade_id, blade_data_dict.get(blade_id, {}), arena_data)
        
        # Remove expired superpowers
        for blade_id, power_name in expired_powers:
            self.clear_superpower_effect(blade_id, power_name)
    
    def _update_entanglement(self, blade_id: int, blade_data: Dict, arena_data: Dict) -> None:
        """Update entanglement visual and effects"""
        if blade_id not in self.blade_states or "entangled_with" not in self.blade_states[blade_id]:
            return
        
        entangled_id = self.blade_states[blade_id]["entangled_with"]
        if entangled_id not in self.blade_states:
            return
        
        # Update visual tether
        if "entanglement_line" in self.blade_states[blade_id]:
            line_id = self.blade_states[blade_id]["entanglement_line"]
            p.removeUserDebugItem(line_id)
            
            # Create new line with updated positions
            blade_pos, _ = p.getBasePositionAndOrientation(blade_id)
            other_pos, _ = p.getBasePositionAndOrientation(entangled_id)
            
            new_line_id = p.addUserDebugLine(
                blade_pos,
                other_pos,
                lineColorRGB=[0.8, 0.2, 0.8],
                lineWidth=3.0,
                lifeTime=0.0
            )
            
            self.blade_states[blade_id]["entanglement_line"] = new_line_id
        
        # Implement entanglement effects
        # 1. Velocity correlation: if one blade speeds up, the other follows
        blade_vel, blade_ang_vel = p.getBaseVelocity(blade_id)
        other_vel, other_ang_vel = p.getBaseVelocity(entangled_id)
        
        # Calculate average linear and angular velocities
        avg_vel = [(a + b) * 0.5 for a, b in zip(blade_vel, other_vel)]
        avg_ang_vel = [(a + b) * 0.5 for a, b in zip(blade_ang_vel, other_ang_vel)]
        
        # Apply correlated velocities (with some random variation to make it interesting)
        blade_random = [random.uniform(-0.1, 0.1) for _ in range(3)]
        other_random = [random.uniform(-0.1, 0.1) for _ in range(3)]
        
        p.resetBaseVelocity(
            blade_id,
            linearVelocity=[avg_vel[i] + blade_random[i] for i in range(3)],
            angularVelocity=avg_ang_vel
        )
        
        p.resetBaseVelocity(
            entangled_id,
            linearVelocity=[avg_vel[i] + other_random[i] for i in range(3)],
            angularVelocity=avg_ang_vel
        )
    
    def clear_superpower_effect(self, blade_id: int, power_name: str) -> None:
        """Clear a specific superpower effect"""
        if blade_id not in self.active_superpowers or power_name not in self.active_superpowers[blade_id]:
            return
        
        # Remove from active superpowers
        del self.active_superpowers[blade_id][power_name]
        
        # Get original state
        original_state = self.blade_states.get(blade_id, {})
        if not original_state:
            return
        
        # Restore original physics parameters
        if "original_mass" in original_state:
            p.changeDynamics(
                blade_id, -1,
                mass=original_state["original_mass"],
                lateralFriction=original_state["original_lateral_friction"],
                restitution=original_state["original_restitution"]
            )
        
        if "original_gravity" in original_state:
            p.setGravity(*original_state["original_gravity"])
        
        # Cleanup special effect artifacts
        self._cleanup_power_artifacts(blade_id, power_name)
    
    def clear_superpower_effects(self, blade_id: int) -> None:
        """Clear all superpower effects for a blade"""
        if blade_id not in self.active_superpowers:
            return
        
        active_powers = list(self.active_superpowers[blade_id].keys())
        for power_name in active_powers:
            self.clear_superpower_effect(blade_id, power_name)
    
    def _cleanup_power_artifacts(self, blade_id: int, power_name: str) -> None:
        """Clean up artifacts created by superpowers"""
        if blade_id not in self.blade_states:
            return
        
        # Superposition cleanup
        if power_name == "quantum_superposition" and "phantom_ids" in self.blade_states[blade_id]:
            for phantom_id in self.blade_states[blade_id]["phantom_ids"]:
                p.removeBody(phantom_id)
            del self.blade_states[blade_id]["phantom_ids"]
        
        # Tunneling cleanup
        if power_name == "quantum_tunneling":
            # Restore collision properties
            p.setCollisionFilterGroupMask(blade_id, -1, 1, 0)  # Reset to default group
            
            # Restore original appearance
            p.changeVisualShape(blade_id, -1, rgbaColor=[1.0, 1.0, 1.0, 1.0])
        
        # Entanglement cleanup
        if power_name == "quantum_entanglement" and "entanglement_line" in self.blade_states[blade_id]:
            p.removeUserDebugItem(self.blade_states[blade_id]["entanglement_line"])
            
            if "entangled_with" in self.blade_states[blade_id]:
                entangled_id = self.blade_states[blade_id]["entangled_with"]
                if entangled_id in self.blade_states and "entangled_with" in self.blade_states[entangled_id]:
                    del self.blade_states[entangled_id]["entangled_with"]
            
            del self.blade_states[blade_id]["entanglement_line"]
            del self.blade_states[blade_id]["entangled_with"]
            
        # Probability amplification cleanup
        if power_name == "probability_amplification" and "particle_ids" in self.blade_states[blade_id]:
            for particle_id in self.blade_states[blade_id]["particle_ids"]:
                p.removeUserDebugItem(particle_id)
            del self.blade_states[blade_id]["particle_ids"]

class SuperpowerTrigger:
    """Defines triggers for automatic superpower activation based on battle conditions"""
    
    def __init__(self):
        self.triggers = {}  # blade_id -> {trigger_name: trigger_condition}
    
    def add_trigger(self, blade_id: int, superpower_name: str, condition_function: Callable) -> None:
        """Add a trigger for a superpower"""
        if blade_id not in self.triggers:
            self.triggers[blade_id] = {}
        
        self.triggers[blade_id][superpower_name] = condition_function
    
    def check_triggers(self, blade_id: int, blade_data: Dict, arena_data: Dict, quantum_powers: QuantumSuperpower) -> None:
        """Check all triggers for a blade and activate superpowers if conditions are met"""
        if blade_id not in self.triggers:
            return
        
        for power_name, condition_func in self.triggers[blade_id].items():
            # Check if trigger condition is met
            if condition_func(blade_data, arena_data):
                # Activate superpower if possible
                quantum_powers.activate_superpower(blade_id, power_name, blade_data, arena_data)

# Example trigger conditions
def low_health_trigger(blade_data: Dict, arena_data: Dict) -> bool:
    """Trigger when blade integrity is below 30%"""
    return blade_data.get("integrity", 100) < 30

def center_control_trigger(blade_data: Dict, arena_data: Dict) -> bool:
    """Trigger when blade is near the center of the arena"""
    blade_pos = blade_data.get("position", [0, 0, 0])
    distance_from_center = math.sqrt(blade_pos[0]**2 + blade_pos[1]**2)
    return distance_from_center < 0.3

def opponent_nearby_trigger(blade_data: Dict, arena_data: Dict) -> bool:
    """Trigger when an opponent is nearby"""
    blade_id = blade_data.get("id", -1)
    blade_pos = blade_data.get("position", [0, 0, 0])
    
    for other_id in arena_data.get("active_blades", []):
        if other_id != blade_id:
            other_data = arena_data.get("blade_data", {}).get(other_id, {})
            other_pos = other_data.get("position", [0, 0, 0])
            distance = math.sqrt(sum((a - b)**2 for a, b in zip(blade_pos, other_pos)))
            if distance < 0.4:
                return True
    
    return False

# Example usage
def setup_quantum_superpowers(blade_id, ai_strategy):
    """Set up quantum superpowers for a RoboBlade based on AI strategy"""
    quantum_powers = QuantumSuperpower()
    triggers = SuperpowerTrigger()
    
    # Set up triggers based on AI strategy
    if ai_strategy == "aggressive":
        triggers.add_trigger(blade_id, "quantum_superposition", opponent_nearby_trigger)
        triggers.add_trigger(blade_id, "decoherence_blast", opponent_nearby_trigger)
    elif ai_strategy == "defensive":
        triggers.add_trigger(blade_id, "quantum_tunneling", low_health_trigger)
        triggers.add_trigger(blade_id, "probability_amplification", low_health_trigger)
    elif ai_strategy == "strategic":
        triggers.add_trigger(blade_id, "quantum_entanglement", center_control_trigger)
        triggers.add_trigger(blade_id, "probability_amplification", center_control_trigger)
    else:  # balanced
        triggers.add_trigger(blade_id, "quantum_superposition", opponent_nearby_trigger)
        triggers.add_trigger(blade_id, "quantum_tunneling", low_health_trigger)
    
    return quantum_powers, triggers

@dataclass
class ArenaProperties:
    """Physical properties and mechanics of a battle arena"""
    name: str
    size: Tuple[float, float, float]  # width, length, height
    base_friction: float
    restitution: float
    gravity: Tuple[float, float, float]  # x, y, z components

class BeybladeArenaPhysics:
    """Manages the physics and dynamics of Beyblade battle arenas"""
    
    def __init__(self, physics_client_id=0):
        """Initialize the arena physics system"""
        self.physics_client_id = physics_client_id
        self.arena_id = None
        self.properties = None
        self.simulation_time = 0.0
        self.time_step = 1/240.0  # Physics simulation time step
        self.chance_vector = [1.0, 0.0, 0.0]  # [luck, friction_var, quirk]
        self.active_blades = []
        self.blade_data = {}
        self.special_objects = {}
        self.visual_effects = {}
        
        # Register built-in arena types
        self.arena_registry = {
            "quantum_field": self._create_quantum_field_arena,
            "circuit": self._create_circuit_arena,
            "vortex_labyrinth": self._create_vortex_labyrinth_arena,
            "neural_nexus": self._create_neural_nexus_arena,
            "eldritch_gateway": self._create_eldritch_gateway_arena
        }
    
    def create_arena(self, arena_type: str) -> int:
        """Create a battle arena of the specified type"""
        if arena_type not in self.arena_registry:
            raise ValueError(f"Unknown arena type: {arena_type}")
        
        # Remove any existing arena
        if self.arena_id is not None:
            p.removeBody(self.arena_id)
            self.arena_id = None
            self.properties = None
            self.special_objects = {}
            self.visual_effects = {}
        
        # Create new arena
        arena_creator = self.arena_registry[arena_type]
        self.arena_id, self.properties = arena_creator()
        
        # Set gravity based on arena properties
        p.setGravity(*self.properties.gravity, physicsClientId=self.physics_client_id)
        
        return self.arena_id
    
    def _create_quantum_field_arena(self) -> Tuple[int, ArenaProperties]:
        """Create a Quantum Field arena with a dynamic, wave-like surface"""
        # Define arena properties
        properties = ArenaProperties(
            name="Quantum Field",
            size=(10.0, 10.0, 2.0),
            base_friction=0.03,
            restitution=0.9,
            gravity=(0.0, 0.0, -9.8),
            surface_type="dynamic",
            special_mechanics=["quantum_wave", "probability_zones"],
            visual_effect_ids=["quantum_glow", "wave_ripple"]
        )
        
        # Create base arena shape
        arena_radius = 5.0
        height_segments = 32
        radius_segments = 32
        
        # Create a heightfield shape for the dynamic surface
        height_data = np.zeros((height_segments, radius_segments))
        for i in range(height_segments):
            for j in range(radius_segments):
                # Initial flat surface with small height variation
                x = (i / (height_segments - 1) - 0.5) * 2.0 * arena_radius
                y = (j / (radius_segments - 1) - 0.5) * 2.0 * arena_radius
                
                # Initial small height variations
                height_data[i][j] = 0.05 * math.sin(x * 0.5) * math.cos(y * 0.5)
        
        terrain_shape = p.createCollisionShape(
            shapeType=p.GEOM_HEIGHTFIELD,
            meshScale=[arena_radius / height_segments, arena_radius / radius_segments, 1.0],
            heightfieldData=height_data.flatten(),
            numHeightfieldRows=height_segments,
            numHeightfieldColumns=radius_segments,
            physicsClientId=self.physics_client_id
        )
        
        # Create the visual shape
        visual_shape = p.createVisualShape(
            shapeType=p.GEOM_HEIGHTFIELD,
            meshScale=[arena_radius / height_segments, arena_radius / radius_segments, 1.0],
            heightfieldData=height_data.flatten(),
            numHeightfieldRows=height_segments,
            numHeightfieldColumns=radius_segments,
            rgbaColor=[0.4, 0.7, 0.9, 1.0],  # Bluish color
            physicsClientId=self.physics_client_id
        )
        
        # Create the arena body
        arena_id = p.createMultiBody(
            baseMass=0,  # Static body
            baseCollisionShapeIndex=terrain_shape,
            baseVisualShapeIndex=visual_shape,
            basePosition=[0, 0, 0],
            physicsClientId=self.physics_client_id
        )
        
        # Set arena material properties
        p.changeDynamics(
            arena_id, -1,
            lateralFriction=properties.base_friction,
            restitution=properties.restitution,
            physicsClientId=self.physics_client_id
        )
        
        # Create probability zones
        zone_positions = [
            [1.5, 1.5, 0.1],
            [-1.5, -1.5, 0.1],
            [1.5, -1.5, 0.1],
            [-1.5, 1.5, 0.1]
        ]
        
        for pos in zone_positions:
            zone_visual = p.createVisualShape(
                shapeType=p.GEOM_CYLINDER,
                radius=0.5,
                length=0.05,
                rgbaColor=[0.7, 0.3, 0.9, 0.5],  # Purple, semi-transparent
                physicsClientId=self.physics_client_id
            )
            
            zone_id = p.createMultiBody(
                baseMass=0,  # Static body
                baseVisualShapeIndex=zone_visual,
                basePosition=pos,
                physicsClientId=self.physics_client_id
            )
            
            self.special_objects[f"prob_zone_{len(self.special_objects)}"] = {
                "id": zone_id,
                "type": "probability_zone",
                "position": pos,
                "radius": 0.5,
                "effect": "luck_boost",
                "magnitude": 1.5  # 50% increase in luck
            }
        
        return arena_id, properties
    
    def _create_circuit_arena(self) -> Tuple[int, ArenaProperties]:
        """Create a Circuit arena with electric pulses and hexagonal sections"""
        # Define arena properties
        properties = ArenaProperties(
            name="Circuit",
            size=(10.0, 10.0, 1.0),
            base_friction=0.05,
            restitution=0.7,
            gravity=(0.0, 0.0, -9.8),
            surface_type="sectioned",
            special_mechanics=["electric_pulse", "conductive_zones"],
            visual_effect_ids=["electric_arc", "power_glow"]
        )
        
        # Create hexagonal prism for the arena
        hexagon_radius = 4.0  # Distance from center to vertex
        num_sides = 6
        
        # Create hexagonal shape points
        vertices = []
        for i in range(num_sides):
            angle = 2 * math.pi * i / num_sides
            x = hexagon_radius * math.cos(angle)
            y = hexagon_radius * math.sin(angle)
            vertices.append([x, y, 0])  # Bottom face
            vertices.append([x, y, 0.2])  # Top face
        
        # Create indices for hexagonal prism faces
        indices = []
        
        # Bottom face (hexagon)
        bottom_indices = []
        for i in range(num_sides):
            bottom_indices.append(i * 2)  # Every even index is a bottom vertex
        indices.extend(self._triangulate_polygon(bottom_indices))
        
        # Top face (hexagon)
        top_indices = []
        for i in range(num_sides):
            top_indices.append(i * 2 + 1)  # Every odd index is a top vertex
        indices.extend(self._triangulate_polygon(top_indices))
        
        # Side faces (rectangles)
        for i in range(num_sides):
            i1 = i * 2
            i2 = i * 2 + 1
            i3 = ((i + 1) % num_sides) * 2 + 1
            i4 = ((i + 1) % num_sides) * 2
            indices.extend([i1, i2, i3, i1, i3, i4])  # Two triangles per rectangular face
        
        # Create mesh collision shape
        mesh_scale = [1.0, 1.0, 1.0]
        collision_shape = p.createCollisionShape(
            shapeType=p.GEOM_MESH,
            vertices=vertices,
            indices=indices,
            meshScale=mesh_scale,
            physicsClientId=self.physics_client_id
        )
        
        # Create visual shape with different sections
        visual_shapes = []
        for i in range(num_sides):
            # Section vertices (a triangular segment of the hexagon)
            section_vertices = [
                [0, 0, 0],  # Center bottom
                [0, 0, 0.2],  # Center top
                vertices[i * 2],  # Bottom corner 1
                vertices[i * 2 + 1],  # Top corner 1
                vertices[((i + 1) % num_sides) * 2],  # Bottom corner 2
                vertices[((i + 1) % num_sides) * 2 + 1]  # Top corner 2
            ]
            
            # Section indices (two triangular prisms)
            section_indices = [
                0, 2, 4,  # Bottom triangle
                1, 3, 5,  # Top triangle
                2, 3, 5, 2, 5, 4,  # Side face 1
                0, 1, 3, 0, 3, 2,  # Side face 2
                0, 4, 5, 0, 5, 1   # Side face 3
            ]
            
            # Alternating colors for sections
            if i % 2 == 0:
                color = [0.3, 0.5, 0.8, 1.0]  # Blue
            else:
                color = [0.2, 0.7, 0.4, 1.0]  # Green
            
            visual_shape = p.createVisualShape(
                shapeType=p.GEOM_MESH,
                vertices=section_vertices,
                indices=section_indices,
                meshScale=mesh_scale,
                rgbaColor=color,
                physicsClientId=self.physics_client_id
            )
            
            visual_shapes.append(visual_shape)
        
        # Create the base arena body
        arena_id = p.createMultiBody(
            baseMass=0,  # Static body
            baseCollisionShapeIndex=collision_shape,
            baseVisualShapeIndex=-1,  # No base visual, we'll use link visuals
            basePosition=[0, 0, 0],
            linkMasses=[0] * num_sides,
            linkCollisionShapeIndices=[-1] * num_sides,  # No collision for links (using base collision)
            linkVisualShapeIndices=visual_shapes,
            linkPositions=[[0, 0, 0]] * num_sides,
            linkOrientations=[[0, 0, 0, 1]] * num_sides,
            linkInertialFramePositions=[[0, 0, 0]] * num_sides,
            linkInertialFrameOrientations=[[0, 0, 0, 1]] * num_sides,
            linkParentIndices=[0] * num_sides,
            linkJointTypes=[p.JOINT_FIXED] * num_sides,
            linkJointAxis=[[0, 0, 0]] * num_sides,
            physicsClientId=self.physics_client_id
        )
        
        # Set arena material properties
        p.changeDynamics(
            arena_id, -1,
            lateralFriction=properties.base_friction,
            restitution=properties.restitution,
            physicsClientId=self.physics_client_id
        )
        
        # Create conductive paths between sections
        path_width = 0.1
        path_length = hexagon_radius * 0.6
        
        for i in range(num_sides):
            angle = 2 * math.pi * i / num_sides + math.pi / num_sides

    def _update_hyperdimensional_shift(self, blade_id: int, current_time: float) -> None:
        """Update the Hyperdimensional Shift effect"""
        if blade_id not in self.blade_states:
            return
            
        if "teleport_progress" not in self.blade_states[blade_id]:
            return
            
        # Get teleport parameters
        start_pos = self.blade_states[blade_id]["teleport_start"]
        end_pos = self.blade_states[blade_id]["teleport_end"]
        
        # Update progress based on 4D path (simulated as interpolation in 3D)
        effect_start_time = self.active_effects[blade_id]["hyperdimensional_shift"] - self.superpower_registry["hyperdimensional_shift"].duration
        progress = min(1.0, (current_time - effect_start_time) / 0.5)  # Complete in 0.5 seconds
        
        self.blade_states[blade_id]["teleport_progress"] = progress
        
        if progress < 1.0:
            # During teleport, move along 4D path
            # For visualization, we make the RoboBlade partially transparent
            alpha = 0.3 + 0.7 * (1.0 - math.sin(progress * math.pi))  # Fade out then in
            
            # Change visual appearance during teleport
            p.changeVisualShape(
                blade_id, -1,
                rgbaColor=[0.5, 0.7, 1.0, alpha],
                physicsClientId=self.physics_client_id
            )
            
            # Interpolate position (with nonlinear curve for 4D effect)
            if progress < 0.5:
                # First half: fade out from start
                t = progress * 2
                interp_factor = math.sin(t * math.pi / 2)  # Ease out
                
                # Stay near start but begin to fade
                interp_pos = [
                    start_pos[0] * (1.0 - interp_factor * 0.2),
                    start_pos[1] * (1.0 - interp_factor * 0.2), 
                    start_pos[2] + interp_factor * 0.1  # Slight rise
                ]
            else:
                # Second half: fade in at destination
                t = (progress - 0.5) * 2
                interp_factor = math.sin(t * math.pi / 2)  # Ease in
                
                # Move toward destination
                interp_pos = [
                    end_pos[0] * interp_factor + start_pos[0] * 0.2 * (1.0 - interp_factor),
                    end_pos[1] * interp_factor + start_pos[1] * 0.2 * (1.0 - interp_factor),
                    end_pos[2] + 0.1 * (1.0 - interp_factor)  # Descend to final height
                ]
            
            # Set position
            p.resetBasePositionAndOrientation(
                blade_id,
                interp_pos,
                p.getBaseOrientation(blade_id, physicsClientId=self.physics_client_id),
                physicsClientId=self.physics_client_id
            )
            
            # During teleport, disable collisions
            p.setCollisionFilterGroupMask(
                blade_id, -1, 0, 0,
                physicsClientId=self.physics_client_id
            )
        else:
            # Teleport complete
            # Reset appearance
            p.changeVisualShape(
                blade_id, -1,
                rgbaColor=[1.0, 1.0, 1.0, 1.0],
                physicsClientId=self.physics_client_id
            )
            
            # Re-enable collisions
            p.setCollisionFilterGroupMask(
                blade_id, -1, 1, 0,
                physicsClientId=self.physics_client_id
            )
    
    def _update_non_euclidean_geometry(self, blade_id: int, current_time: float) -> None:
        """Update the Non-Euclidean Geometry effect"""
        if blade_id not in self.blade_states:
            return
            
        # Get current position
        blade_pos, blade_orn = p.getBasePositionAndOrientation(blade_id, physicsClientId=self.physics_client_id)
        blade_vel, blade_ang_vel = p.getBaseVelocity(blade_id, physicsClientId=self.physics_client_id)
        
        # Update distortion field visualizations
        if "distortion_indicators" in self.blade_states[blade_id]:
            # Remove old indicators
            for indicator_id in self.blade_states[blade_id]["distortion_indicators"]:
                p.removeUserDebugItem(indicator_id, physicsClientId=self.physics_client_id)
                
            # Create new indicators with non-Euclidean distortion
            field_radius = 1.0
            num_indicators = 8
            indicator_ids = []
            
            for i in range(num_indicators):
                angle_base = 2 * math.pi * i / num_indicators
                
                # Non-Euclidean distortion: angles don't add up to 360 degrees
                angle_factor = self.blade_states[blade_id].get("angle_factor", 1.5)
                angle = angle_base * angle_factor
                
                # Create distorted distance
                distance_factor = self.blade_states[blade_id].get("distance_factor", 0.7)
                # Add time-varying distortion
                distortion = 0.3 * math.sin(3 * angle + current_time * 2)
                r = field_radius * (1.0 + distortion) * distance_factor
                
                x = blade_pos[0] + r * math.cos(angle)
                y = blade_pos[1] + r * math.sin(angle)
                
                # Create visual indicator
                point_id = p.addUserDebugLine(
                    [x, y, blade_pos[2]],
                    [x, y, blade_pos[2] + 0.1],
                    lineColorRGB=[0.5, 0.0, 0.8],
                    lineWidth=2.0,
                    lifeTime=0.1,
                    physicsClientId=self.physics_client_id
                )
                
                indicator_ids.append(point_id)
            
            self.blade_states[blade_id]["distortion_indicators"] = indicator_ids
        # Apply non-Euclidean physics effect: velocity direction doesn't match movement
        if blade_vel[0]**2 + blade_vel[1]**2 > 0.01:  # If moving
            # Calculate normal expected direction
            speed = math.sqrt(blade_vel[0]**2 + blade_vel[1]**2)
            norm_dir = [blade_vel[0]/speed, blade_vel[1]/speed]
            
            # Apply non-Euclidean transformation to direction
            # In non-Euclidean space, parallel lines can converge or diverge
            # Simulate this by adding a curved component to movement
            effect_strength = 0.3  # Strength of non-Euclidean effect
            curvature = math.sin(current_time * 3) * effect_strength
            
            # Rotate direction vector by curvature amount
            cos_curve = math.cos(curvature)
            sin_curve = math.sin(curvature)
            curved_dir = [
                norm_dir[0] * cos_curve - norm_dir[1] * sin_curve,
                norm_dir[0] * sin_curve + norm_dir[1] * cos_curve
            ]
            
            # Apply a small force in the curved direction
            force_magnitude = 5.0 * self.blade_states[blade_id].get("original_mass", 1.0)
            force = [d * force_magnitude for d in curved_dir]
            
            # Apply the force
            p.applyExternalForce(
                blade_id, -1,
                force + [0],  # Add Z component
                blade_pos,
                p.WORLD_FRAME,
                physicsClientId=self.physics_client_id
            )

    def _update_phase_space_manipulation(self, blade_id: int, current_time: float) -> None:
        """Update the Phase Space Manipulation effect"""
        if blade_id not in self.blade_states or "phase_phantoms" not in self.blade_states[blade_id]:
            return
            
        # Update phantom instances
        phantom_ids = self.blade_states[blade_id]["phase_phantoms"]
        
        # Get current state
        blade_pos, blade_orn = p.getBasePositionAndOrientation(blade_id, physicsClientId=self.physics_client_id)
        blade_vel, blade_ang_vel = p.getBaseVelocity(blade_id, physicsClientId=self.physics_client_id)
        
        # Animate phantoms
        effect_start_time = self.active_effects[blade_id]["phase_space_manipulation"] - self.superpower_registry["phase_space_manipulation"].duration
        effect_progress = min(1.0, (current_time - effect_start_time) / self.superpower_registry["phase_space_manipulation"].duration)
        
        # Phase space manipulation allows violating normal physics relationships
        # between position and momentum
        
        # Position phantoms (first half of phantoms)
        position_phantoms = phantom_ids[:len(phantom_ids)//2]
        for i, phantom_id in enumerate(position_phantoms):
            # Get current phantom state
            phantom_pos, phantom_orn = p.getBasePositionAndOrientation(phantom_id, physicsClientId=self.physics_client_id)
            
            # Calculate new position (oscillate around the main blade)
            phase_offset = 2 * math.pi * i / len(position_phantoms)
            orbit_radius = 0.3
            
            # Orbital motion around the blade
            orbit_x = blade_pos[0] + orbit_radius * math.cos(current_time * 3 + phase_offset)
            orbit_y = blade_pos[1] + orbit_radius * math.sin(current_time * 3 + phase_offset)
            
            # Set new position but maintain phantom's velocity
            p.resetBasePositionAndOrientation(
                phantom_id,
                [orbit_x, orbit_y, blade_pos[2]],
                blade_orn,
                physicsClientId=self.physics_client_id
            )
            
            # Update appearance based on effect progress
            alpha = 0.5 * (1.0 - effect_progress)  # Fade out as effect ends
            p.changeVisualShape(
                phantom_id, -1,
                rgbaColor=[0.8, 0.2, 0.8, alpha],
                physicsClientId=self.physics_client_id
            )
        
        # Velocity phantoms (second half of phantoms)
        velocity_phantoms = phantom_ids[len(phantom_ids)//2:]
        for i, phantom_id in enumerate(velocity_phantoms):
            # Calculate new velocity (rotate around the main blade's velocity)
            phase_offset = 2 * math.pi * i / len(velocity_phantoms)
            
            # Base velocity magnitude
            speed = math.sqrt(sum(v*v for v in blade_vel))
            if speed < 0.1:
                # If almost stationary, use a default velocity
                base_vel = [1.0, 0.0, 0.0]  # Arbitrary direction
                speed = 1.0
            else:
                base_vel = [v/speed for v in blade_vel]  # Normalized direction
            
            # Rotate velocity vector
            rot_angle = current_time * 2 + phase_offset
            cos_rot = math.cos(rot_angle)
            sin_rot = math.sin(rot_angle)
            
            rotated_vel = [
                base_vel[0] * cos_rot - base_vel[1] * sin_rot,
                base_vel[0] * sin_rot + base_vel[1] * cos_rot,
                0  # Keep Z component zero
            ]
            
            # Apply velocity with original magnitude
            phantom_vel = [v * speed for v in rotated_vel]
            
            # Reset velocity
            p.resetBaseVelocity(
                phantom_id,
                linearVelocity=phantom_vel,
                angularVelocity=blade_ang_vel,
                physicsClientId=self.physics_client_id
            )
            
            # Update appearance based on effect progress
            alpha = 0.5 * (1.0 - effect_progress)  # Fade out as effect ends
            p.changeVisualShape(
                phantom_id, -1,
                rgbaColor=[0.2, 0.6, 0.8, alpha],
                physicsClientId=self.physics_client_id
            )
        
        # Apply phase space effect to the main blade
        # This allows temporary violation of physics constraints
        
        # Example effect: allow blade to gain momentum without corresponding position change
        # (like storing energy for a burst)
        if effect_progress > 0.8:
            # Near the end of the effect, prepare for a burst
            # Store energy as a multiplier for the final burst
            self.blade_states[blade_id]["phase_burst_energy"] = 2.0 + effect_progress * 3  # 2.0-5.0x
        
        if effect_progress > 0.95:
            # At the very end, release the energy burst
            if "phase_burst_energy" in self.blade_states[blade_id]:
                energy_multiplier = self.blade_states[blade_id]["phase_burst_energy"]
                
                # Apply a strong impulse in the forward direction
                forward_vec = self._get_forward_vector(blade_orn)
                impulse_force = [f * energy_multiplier * 50.0 for f in forward_vec]
                
                p.applyExternalForce(
                    blade_id, -1,
                    impulse_force + [0],  # Add Z component
                    blade_pos,
                    p.WORLD_FRAME,
                    physicsClientId=self.physics_client_id
                )
                
                # Remove stored energy
                del self.blade_states[blade_id]["phase_burst_energy"]

            # For visualization, we make the RoboBlade partially transparent
            alpha = 0.3 + 0.7 * (1.0 - math.sin(progress * math.pi))  # Fade out then in
            
            # Change visual appearance during teleport
            p.changeVisualShape(

            )
            
            # Interpolate position (with nonlinear curve for 4D effect)
            if progress < 0.5:
                # First half: fade out from start
                t = progress * 2
                interp_factor = math.sin(t * math.pi / 2)  # Ease out
                
                # Stay near start but begin to fade
                interp_pos = [
                    start_pos[0] * (1.0 - interp_factor * 0.2),
                    start_pos[1] * (1.0 - interp_factor * 0.2), 
                    start_pos[2] + interp_factor * 0.1  # Slight rise
                ]
            else:
                # Second half: fade in at destination
                t = (progress - 0.5) * 2
                interp_factor = math.sin(t * math.pi / 2)  # Ease in
                
                # Move toward destination
                interp_pos = [
                    end_pos[0] * interp_factor + start_pos[0] * 0.2 * (1.0 - interp_factor),
                    end_pos[1] * interp_factor + start_pos[1] * 0.2 * (1.0 - interp_factor),
                    end_pos[2] + 0.1 * (1.0 - interp_factor)  # Descend to final height
                ]
            
            # Set position
            p.resetBasePositionAndOrientation(
                blade_id,
                interp_pos,
                p.getBaseOrientation(blade_id, physicsClientId=self.physics_client_id),
                physicsClientId=self.physics_client_id
            )
            
            # During teleport, disable collisions
            p.setCollisionFilterGroupMask(
                blade_id, -1, 0, 0,
                physicsClientId=self.physics_client_id
            )
        else:
            # Teleport complete
            # Reset appearance
            p.changeVisualShape(
                blade_id, -1,
                rgbaColor=[1.0, 1.0, 1.0, 1.0],
                physicsClientId=self.physics_client_id
            )
            
            # Re-enable collisions
            p.setCollisionFilterGroupMask(
                blade_id, -1, 1, 0,
                physicsClientId=self.physics_client_id
            )

    def _create_quantum_model(self):
        """Create a QNode with PyTorch integration for quantum calculations"""
        @qml.qnode(self.device, interface="torch")
        def quantum_circuit(inputs, weights):
            """
            Parameterized quantum circuit for superpower calculations
            
            Args:
                inputs: Input parameters (typically from blade state)
                weights: Trainable weights for the quantum circuit
            
            Returns:
                Measurement results from the quantum circuit
            """
            # Encode inputs into quantum state
            for i in range(self.num_qubits):
                qml.RY(inputs[i], wires=i)
            
            # Apply parameterized quantum layers
            for layer in range(2):  # Two layers of parameterized gates
                for i in range(self.num_qubits):
                    qml.RX(weights[layer][i][0], wires=i)
                    qml.RY(weights[layer][i][1], wires=i)
                    qml.RZ(weights[layer][i][2], wires=i)
                
                # Entanglement layer
                for i in range(self.num_qubits - 1):
                    qml.CNOT(wires=[i, i + 1])
                qml.CNOT(wires=[self.num_qubits - 1, 0])  # Connect last qubit back to first
            
            # Return measurements of all qubits
            return [qml.expval(qml.PauliZ(i)) for i in range(self.num_qubits)]
        
        # Create randomly initialized weights for the quantum circuit
        weight_shapes = {"weights": (2, self.num_qubits, 3)}  # 2 layers, num_qubits, 3 rotation gates
        
        # Return the quantum circuit as a PyTorch model
        model = qml.qnn.TorchLayer(quantum_circuit, weight_shapes)
        
        return model

    def _create_hadamard_superposition_circuit(self, inputs):
        """
        Create a Hadamard superposition circuit that puts
        all qubits in superposition for spatial projection
        """
        @qml.qnode(self.device, interface="torch")
        def circuit(inputs):
            # Initialize qubits based on inputs
            for i in range(self.num_qubits):
                qml.RY(inputs[i], wires=i)
            
            # Apply Hadamard gates to create superposition
            for i in range(self.num_qubits):
                qml.Hadamard(wires=i)
            
            # Apply controlled phase gates to create entanglement
            for i in range(self.num_qubits - 1):
                qml.CZ(wires=[i, i + 1])
            
            # Measure in superposition basis
            return [qml.expval(qml.PauliX(i)) for i in range(self.num_qubits)]
        
        return circuit(inputs)
    
    def _create_wavefunction_collapse_circuit(self, inputs):
        """
        Create a quantum fluid dynamics circuit that
        simulates fluid-like movement patterns
        """
        @qml.qnode(self.device, interface="torch")
        def circuit(inputs):
            # Initialize qubits based on inputs
            for i in range(self.num_qubits):
                qml.RY(inputs[i], wires=i)
            
            # Create superposition
            for i in range(self.num_qubits):
                qml.Hadamard(wires=i)
            
            # Apply rotation gates parameterized by inputs
            for i in range(self.num_qubits):
                qml.RX(inputs[i % len(inputs)], wires=i)
                qml.RY(inputs[(i + 1) % len(inputs)], wires=i)
            
            # Create entanglement pattern for fluid dynamics
            for i in range(self.num_qubits):
                qml.CRZ(inputs[i % len(inputs)], wires=[i, (i + 1) % self.num_qubits])
            
            # Add another layer of rotations
            for i in range(self.num_qubits):
                qml.RZ(inputs[i % len(inputs)], wires=i)
            
            # Measure in computational basis
            return [qml.expval(qml.PauliZ(i)) for i in range(self.num_qubits)]
        
        return circuit(inputs)
    
    def _create_braid_operator_circuit(self, inputs):
        """
        Create a topological vortex circuit that
        implements braid operators for vortex effects
        """
        @qml.qnode(self.device, interface="torch")
        def circuit(inputs):
            # Initialize qubits based on inputs
            for i in range(self.num_qubits):
                qml.RY(inputs[i], wires=i)
            
            # Create a braid pattern by applying SWAP gates
            # This mimics the topological braiding in anyonic systems
            for layer in range(2):
                # Even-indexed braids
                for i in range(0, self.num_qubits - 1, 2):
                    qml.SWAP(wires=[i, i + 1])
                
                # Apply phase based on inputs
                for i in range(self.num_qubits):
                    qml.PhaseShift(inputs[i % len(inputs)], wires=i)
                
                # Odd-indexed braids
                for i in range(1, self.num_qubits - 1, 2):
                    qml.SWAP(wires=[i, i + 1])
                
                # Apply phase based on inputs
                for i in range(self.num_qubits):
                    qml.PhaseShift(inputs[(i + 1) % len(inputs)], wires=i)
            
            # Measure in computational basis
            return [qml.expval(qml.PauliZ(i)) for i in range(self.num_qubits)]
        
        return circuit(inputs)
    
    def _create_hilbert_space_transformation_circuit(self, inputs):
        """
        Create a dimensional shapeshifting circuit that
        transforms the state through Hilbert space
        """
        @qml.qnode(self.device, interface="torch")
        def circuit(inputs):
            # Initialize qubits based on inputs
            for i in range(self.num_qubits):
                qml.RY(inputs[i], wires=i)
            
            # Apply transformation layers
            for layer in range(3):
                # Rotation layer
                for i in range(self.num_qubits):
                    qml.Rot(
                        inputs[i % len(inputs)],
                        inputs[(i + 1) % len(inputs)],
                        inputs[(i + 2) % len(inputs)],
                        wires=i
                    )
                
                # Entanglement layer - creates a fully connected graph
                for i in range(self.num_qubits):
                    for j in range(i + 1, self.num_qubits):
                        qml.CRZ(inputs[i % len(inputs)], wires=[i, j])
                
                # Basis transformation
                if layer < 2:  # Skip on last layer
                    for i in range(self.num_qubits):
                        qml.Hadamard(wires=i)
            
            # Measure in computational basis
            return [qml.expval(qml.PauliZ(i)) for i in range(self.num_qubits)]
        
        return circuit(inputs)
    
    def activate_hyperspatial_projection(self, blade_id: int, blade_data: Dict, arena_data: Dict, p) -> bool:
        """
        Activate the Hyperspatial Projection superpower
        
        Args:
            blade_id: ID of the RoboBlade to apply the effect to
            blade_data: Current state data for the blade
            arena_data: Current state data for the arena
            p: PyBullet physics client
        
        Returns:
            Success status of the superpower activation
        """
        superpower = self.superpower_registry["hyperspatial_projection"]
        
        # Check if blade has enough energy and integrity
        if blade_data.get("energy", 0) < superpower.energy_cost:
            return False
        
        if blade_data.get("integrity", 100) < superpower.integrity_cost:
            return False
        
        # Get current blade state
        position, orientation = p.getBasePositionAndOrientation(blade_id)
        linear_vel, angular_vel = p.getBaseVelocity(blade_id)
        
        # Store original state if not already stored
        if blade_id not in self.blade_states:
            self.blade_states[blade_id] = {
                "original_position": position,
                "original_orientation": orientation,
                "original_linear_vel": linear_vel,
                "original_angular_vel": angular_vel,
                "original_visual": p.getVisualShapeData(blade_id)[0][7]  # RGBA color
            }
        
        # Create input tensor from blade data
        inputs = torch.tensor([
            position[0], position[1], position[2],
            linear_vel[0], linear_vel[1],
        ], dtype=torch.float32)
        
        # Normalize inputs to range suitable for quantum circuit
        inputs = torch.tanh(inputs) * math.pi
        
        # Run quantum circuit to determine projection parameters
        results = self._create_hadamard_superposition_circuit(inputs)
        
        # Convert quantum results to spatial projection parameters
        projection_distance = 3.0 * (results[0] + 1) / 2  # Scale to 0-3 range
        projection_angle = math.pi * results[1]
        projection_height = 0.5 * results[2]
        
        # Calculate target position for hyperspatial projection
        forward_vec = self._get_forward_vector(orientation)
        
        # Combine forward direction with quantum projection angle
        direction_x = forward_vec[0] * math.cos(projection_angle) - forward_vec[1] * math.sin(projection_angle)
        direction_y = forward_vec[0] * math.sin(projection_angle) + forward_vec[1] * math.cos(projection_angle)
        
        target_position = [
            position[0] + direction_x * projection_distance,
            position[1] + direction_y * projection_distance,
            position[2] + projection_height
        ]
        
        # Store projection parameters
        self.blade_states[blade_id].update({
            "projection_start": position,
            "projection_target": target_position,
            "projection_start_time": arena_data.get("simulation_time", 0),
            "projection_results": results
        })
        
        # Apply visual effect - shift to bluish glow
        p.changeVisualShape(
            blade_id, -1,
            rgbaColor=[0.3, 0.5, 1.0, 0.7]  # Blue, semi-transparent
        )
        
        # Consume resources
        blade_data["energy"] -= superpower.energy_cost
        blade_data["integrity"] -= superpower.integrity_cost
        
        # Set active effect and cooldown
        current_time = arena_data.get("simulation_time", 0)
        
        if blade_id not in self.active_effects:
            self.active_effects[blade_id] = {}
        if blade_id not in self.cooldowns:
            self.cooldowns[blade_id] = {}
        
        self.active_effects[blade_id]["hyperspatial_projection"] = current_time + superpower.duration
        self.cooldowns[blade_id]["hyperspatial_projection"] = current_time + superpower.duration + superpower.cooldown
        
        return True

    def _create_vortex_visuals(self, blade_id: int, position: List[float], radius: float, height: float, rotation: float, p) -> None:

        visual_ids = []
        
        # Create spiral lines for the vortex
        num_spirals = 8
        points_per_spiral = 32
        
        for spiral_idx in range(num_spirals):
            spiral_phase = 2 * math.pi * spiral_idx / num_spirals
            
            # Create points for this spiral
            spiral_points = []
            for point_idx in range(points_per_spiral):
                t = point_idx / (points_per_spiral - 1)  # Parameter from 0 to 1
                
                # Spiral equation with rotation
                angle = rotation * t + spiral_phase
                r = radius * t
                
                x = position[0] + r * math.cos(angle)
                y = position[1] + r * math.sin(angle)
                z = position[2] + height * t
                
                spiral_points.append([x, y, z])
            
            # Create lines connecting the points
            for i in range(len(spiral_points) - 1):
                line_id = p.addUserDebugLine(
                    spiral_points[i],
                    spiral_points[i + 1],
                    lineColorRGB=[0.8, 0.2, 0.8],  # Purple
                    lineWidth=2,
                    lifeTime=0.1
                )
                visual_ids.append(line_id)
        
        # Store visual IDs
        self.blade_states[blade_id]["vortex_visual_ids"] = visual_ids

class DimensionalShapeshifter:

    def __init__(self, physics_client_id=0):
        """Initialize the dimensional shapeshifter system"""
        self.physics_client_id = physics_client_id
        self.active_transformations = {}  # blade_id -> transformation_data
        self.blade_states = {}  # blade_id -> original_state
        
        # Set up quantum device using PennyLane
        self.num_qubits = 6
        self.device = qml.device("default.qubit", wires=self.num_qubits)
        self.quantum_model = self._create_quantum_model()
        
        # Prepare shape morphing parameters
        self.shape_library = self._initialize_shape_library()
    
    def _create_quantum_model(self):
        """Create the quantum model for computing dimensional transformations"""
        @qml.qnode(self.device, interface="torch")
        def quantum_circuit(inputs, weights):
            # Encode input parameters
            for i in range(self.num_qubits):
                qml.RY(inputs[i % len(inputs)], wires=i)
            
            # Apply parameterized quantum layers
            for layer in range(3):
                # Rotation layer
                for i in range(self.num_qubits):
                    qml.RX(weights[layer][i][0], wires=i)
                    qml.RY(weights[layer][i][1], wires=i)
                    qml.RZ(weights[layer][i][2], wires=i)
                
                # Entanglement layer
                for i in range(self.num_qubits - 1):
                    qml.CNOT(wires=[i, i + 1])
                qml.CNOT(wires=[self.num_qubits - 1, 0])
            
            # Return measurements
            return [qml.expval(qml.PauliZ(i)) for i in range(self.num_qubits)]
        
        # Define weight shapes for the model
        weight_shapes = {"weights": (3, self.num_qubits, 3)}  # 3 layers, num_qubits, 3 rotation gates
        
        # Create PyTorch layer
        return qml.qnn.TorchLayer(quantum_circuit, weight_shapes)

    def activate_4d_shapeshifting(self, blade_id: int, blade_data: Dict, arena_data: Dict) -> bool:
        """
        Activate 4D dimensional shapeshifting for a RoboBlade
        
        Args:
            blade_id: ID of the RoboBlade to apply the effect to
            blade_data: Current state data for the blade
            arena_data: Current state data for the arena
        
        Returns:
            Success status of the shapeshifting activation
        """
        # Check if already shapeshifting
        if blade_id in self.active_transformations:
            return False
        
        # Check energy cost
        energy_cost = 80.0
        if blade_data.get("energy", 0) < energy_cost:
            return False
        
        # Get current blade state
        position, orientation = p.getBasePositionAndOrientation(blade_id, physicsClientId=self.physics_client_id)
        linear_vel, angular_vel = p.getBaseVelocity(blade_id, physicsClientId=self.physics_client_id)
        
        # Store original state if not already stored
        if blade_id not in self.blade_states:
            # Get current collision shape
            collision_shapes = p.getCollisionShapeData(blade_id, -1, physicsClientId=self.physics_client_id)
            
            # Get dynamics information
            dynamics_info = p.getDynamicsInfo(blade_id, -1, physicsClientId=self.physics_client_id)
            
            # Store original state
            self.blade_states[blade_id] = {
                "original_shape": collision_shapes,
                "original_mass": dynamics_info[0],
                "original_lateral_friction": dynamics_info[1],
                "original_restitution": dynamics_info[5],
                "original_linear_damping": dynamics_info[6],
                "original_angular_damping": dynamics_info[7],
                "original_position": position,
                "original_orientation": orientation,
                "original_visual": p.getVisualShapeData(blade_id, physicsClientId=self.physics_client_id)[0]
            }
        
        # Create input tensor from blade and arena data
        inputs = torch.tensor([
            position[0], position[1], position[2],  # Position
            linear_vel[0], linear_vel[1], linear_vel[2],  # Linear velocity
            angular_vel[0], angular_vel[1], angular_vel[2],  # Angular velocity
            blade_data.get("integrity", 100) / 100,  # Normalized integrity
            arena_data.get("chance_vector", [1.0, 0.0, 0.0])[0]  # Luck factor
        ], dtype=torch.float32)
        
        # Normalize inputs to suitable range for quantum circuit
        inputs = torch.clamp(inputs / 5.0, -1.0, 1.0) * math.pi
        
        # Process through quantum model to get transformation parameters
        with torch.no_grad():
            results = self.quantum_model(inputs)
        
        # Convert quantum results to shapeshifting parameters
        shapeshifting_params = self._quantum_results_to_params(results)
        
        # Calculate morph sequence based on quantum results
        morph_sequence = self._calculate_morph_sequence(shapeshifting_params)
        
        # Store active transformation data
        self.active_transformations[blade_id] = {
            "start_time": arena_data.get("simulation_time", 0),
            "duration": 8.0,  # Total duration of shapeshifting effect
            "params": shapeshifting_params,
            "morph_sequence": morph_sequence,
            "current_morph_index": 0,
            "current_morph_progress": 0.0,
            "visual_elements": [],
            "phase_particles": [],
            "hyperspace_trail": []
        }
        
        # Start the first shape transformation
        self._apply_shape_transformation(
            blade_id, 
            morph_sequence[0]["shape"], 
            morph_sequence[0]["scale"],
            morph_sequence[0]["dynamics"]
        )
        
        # Apply initial visual effects
        self._create_hyperspace_visuals(blade_id, position)
        
        # Deduct energy
        blade_data["energy"] -= energy_cost
        
        return True
    
    def _quantum_results_to_params(self, results: torch.Tensor) -> Dict:
        """Convert quantum circuit results to shapeshifting parameters"""
        # Normalize results to 0-1 range
        normalized = (results + 1) / 2
        
        # Extract key parameters
        return {
            "morph_rate": 0.5 + normalized[0] * 2.0,  # 0.5-2.5 shapes per second
            "dimension_factor": normalized[1] * 0.8 + 0.2,  # 0.2-1.0 dimensionality factor
            "non_linearity": normalized[2] * 0.9 + 0.1,  # 0.1-1.0 non-linearity factor
            "shape_preferences": normalized[3:].tolist(),  # Preferences for shape selection
            "spatial_curvature": normalized[0] * normalized[1] * 2.0  # 0.0-2.0 spatial curvature
        }
    
    def _calculate_morph_sequence(self, params: Dict) -> List[Dict]:
        """Calculate a sequence of shapes to morph through based on parameters"""
        # Number of shapes in the sequence
        num_morphs = max(3, min(7, round(params["morph_rate"] * 3)))
        
        # Select shapes based on preferences
        available_shapes = list(self.shape_library.keys())
        shape_scores = {
            shape: params["shape_preferences"][i % len(params["shape_preferences"])] 
            for i, shape in enumerate(available_shapes)
        }
        
        # Sort shapes by preference score
        sorted_shapes = sorted(shape_scores.items(), key=lambda x: x[1], reverse=True)
        
        # Create morph sequence
        morph_sequence = []
        
        for i in range(num_morphs):
            # Select shape with some randomness influenced by preferences
            if random.random() < 0.7:  # 70% chance to follow preferences
                shape_name = sorted_shapes[i % len(sorted_shapes)][0]
            else:
                shape_name = random.choice(available_shapes)
            
            # Get base shape data
            shape_data = self.shape_library[shape_name]
            
            # Calculate scale factor with some dimension-specific modulation
            scale_factor = 1.0 + (params["dimension_factor"] - 0.5) * 0.6  # 0.7-1.3 scale
            scale_factor *= 1.0 + math.sin(i * 0.8) * 0.2  # Add oscillation
            
            # Calculate dynamics adaptations based on non-linearity
            dynamics_mods = {
                "friction": max(0.01, shape_data["dynamic_properties"]["friction"] * (1.0 - params["non_linearity"] * 0.8)),
                "restitution": min(0.99, shape_data["dynamic_properties"]["restitution"] * (1.0 + params["non_linearity"] * 0.3)),
                "angular_damping": shape_data["dynamic_properties"]["angular_damping"] * (1.0 - params["non_linearity"] * 0.5)
            }
            
            # Add to sequence
            morph_sequence.append({
                "shape": shape_name,
                "scale": scale_factor,
                "dynamics": dynamics_mods,
                "duration": 1.0 / params["morph_rate"],  # Time to spend in this shape
                "phase_shift": (i / num_morphs) * 2 * math.pi  # Phase for visual effects
            })
        
        return morph_sequence
    
    def _apply_shape_transformation(self, blade_id: int, shape_name: str, scale_factor: float, dynamics_mods: Dict) -> None:

    def _create_hyperspace_visuals(self, blade_id: int, position: List[float]) -> None:
        """Create visual effects for 4D hyperspace interaction"""
        if blade_id not in self.active_transformations:
            return
        
        # Create phase particles that show higher dimensional interactions
        num_particles = 30
        params = self.active_transformations[blade_id]["params"]
        
        for i in range(num_particles):
            # Calculate particle position in a hypersphere projection
            # Use 4D coordinates (w, x, y, z) and project to 3D
            w = random.uniform(-1, 1)
            theta = random.uniform(0, 2 * math.pi)
            phi = random.uniform(0, math.pi)
            
            # 4D to 3D projection with w component affecting scale
            r = params["dimension_factor"] * (1.0 + 0.5 * w)
            x = position[0] + r * math.sin(phi) * math.cos(theta)
            y = position[1] + r * math.sin(phi) * math.sin(theta)
            z = position[2] + r * math.cos(phi)
            
            # Store particle with 4D properties
            self.active_transformations[blade_id]["phase_particles"].append({
                "position": [x, y, z],
                "w_coordinate": w,
                "velocity": [
                    random.uniform(-0.5, 0.5) * params["non_linearity"],
                    random.uniform(-0.5, 0.5) * params["non_linearity"],
                    random.uniform(-0.2, 0.2) * params["non_linearity"],
                    random.uniform(-0.3, 0.3)  # w-dimension velocity
                ],
                "size": 0.02 + random.uniform(0, 0.04) * (1.0 + w),
                "color": [0.3, 0.7, 0.9, 0.7],  # Bluish, semi-transparent
                "lifetime": random.uniform(0.5, 1.5),
                "visual_id": None
            })
    
    def update(self, blade_data_dict: Dict[int, Dict], arena_data: Dict, delta_time: float) -> None:
        """Update all active shapeshifting transformations"""
        current_time = arena_data.get("simulation_time", 0)
        
        # Process each blade with active transformation
        for blade_id in list(self.active_transformations.keys()):
            if blade_id not in self.active_transformations:
                continue  # May have been removed during processing
                
            transform_data = self.active_transformations[blade_id]
            
            # Check if effect has expired
            effect_duration = transform_data["duration"]
            if current_time - transform_data["start_time"] >= effect_duration:
                self._revert_to_original_shape(blade_id)
                continue
            
            # Get current position for visual effects
            position, _ = p.getBasePositionAndOrientation(blade_id, physicsClientId=self.physics_client_id)
            
            # Update current morph progress
            current_morph = transform_data["morph_sequence"][transform_data["current_morph_index"]]
            transform_data["current_morph_progress"] += delta_time
            
            # Check if it's time to morph to the next shape
            if transform_data["current_morph_progress"] >= current_morph["duration"]:
                # Reset progress and move to next shape
                transform_data["current_morph_progress"] = 0
                transform_data["current_morph_index"] = (transform_data["current_morph_index"] + 1) % len(transform_data["morph_sequence"])
                
                # Apply the new shape transformation
                next_morph = transform_data["morph_sequence"][transform_data["current_morph_index"]]
                self._apply_shape_transformation(
                    blade_id,
                    next_morph["shape"],
                    next_morph["scale"],
                    next_morph["dynamics"]
                )
                
                # Create dimensional shift visual effect
                self._create_dimensional_shift_effect(blade_id, position)
            
            # Update hyperspace trail
            self._update_hyperspace_trail(blade_id, position)
            
            # Update phase particles
            self._update_phase_particles(blade_id, delta_time)
            
            # Apply 4D movement mechanics
            self._apply_4d_movement(blade_id, delta_time, transform_data["params"])
    
    def _update_hyperspace_trail(self, blade_id: int, position: List[float]) -> None:
        """Update the hyperspace trail following the RoboBlade"""
        if blade_id not in self.active_transformations:
            return
            
        transform_data = self.active_transformations[blade_id]
        
        # Add current position to trail with limited length
        transform_data["hyperspace_trail"].append(position.copy())
        if len(transform_data["hyperspace_trail"]) > 20:
            transform_data["hyperspace_trail"].pop(0)
        
        # Clear old trail visuals
        for visual_id in transform_data.get("trail_visuals", []):
            p.removeUserDebugItem(visual_id, physicsClientId=self.physics_client_id)
        
        # Create new trail visuals
        trail_visuals = []
        if len(transform_data["hyperspace_trail"]) >= 2:
            for i in range(len(transform_data["hyperspace_trail"]) - 1):
                start = transform_data["hyperspace_trail"][i]
                end = transform_data["hyperspace_trail"][i + 1]
                
                # Calculate color with fade effect
                alpha = 0.7 * (i / (len(transform_data["hyperspace_trail"]) - 1))
                
                # Create trail segment
                visual_id = p.addUserDebugLine(
                    start, 
                    end,
                    lineColorRGB=[0.3, 0.6, 0.9 * alpha],
                    lineWidth=3 * alpha,
                    lifeTime=0.1,
                    physicsClientId=self.physics_client_id
                )
                trail_visuals.append(visual_id)
        
        transform_data["trail_visuals"] = trail_visuals
    
    def _update_phase_particles(self, blade_id: int, delta_time: float) -> None:
        """Update 4D phase particles"""
        if blade_id not in self.active_transformations:
            return
            
        transform_data = self.active_transformations[blade_id]
        
        # Remove old visual particles
        for particle in transform_data["phase_particles"]:
            if particle["visual_id"] is not None:
                p.removeUserDebugItem(particle["visual_id"], physicsClientId=self.physics_client_id)
                particle["visual_id"] = None
        
        # Update particles
        remaining_particles = []
        for particle in transform_data["phase_particles"]:
            # Update particle lifetime
            particle["lifetime"] -= delta_time
            if particle["lifetime"] <= 0:
                continue
                
            # Update position including 4D component
            particle["position"][0] += particle["velocity"][0] * delta_time
            particle["position"][1] += particle["velocity"][1] * delta_time
            particle["position"][2] += particle["velocity"][2] * delta_time
            particle["w_coordinate"] += particle["velocity"][3] * delta_time
            
            # Apply 4D to 3D projection effects
            w_factor = 1.0 + 0.3 * math.sin(particle["w_coordinate"] * 2)
            size = particle["size"] * w_factor
            
            # Create visual representation (point or small line)
            end_pos = [
                particle["position"][0],
                particle["position"][1],
                particle["position"][2] + size
            ]
            
            # Adjust alpha based on w-coordinate (particles fade in and out as they move through 4D)
            alpha = 0.7 * (0.5 + 0.5 * math.cos(particle["w_coordinate"] * 3))
            color = [
                particle["color"][0],
                particle["color"][1],
                particle["color"][2],
                alpha
            ]
            
            # Create visual
            particle["visual_id"] = p.addUserDebugLine(
                particle["position"],
                end_pos,
                lineColorRGB=color[:3],  # RGB component
                lineWidth=2 * alpha,
                lifeTime=0.1,
                physicsClientId=self.physics_client_id
            )
            
            remaining_particles.append(particle)
        
        transform_data["phase_particles"] = remaining_particles
        
        # Create new particles if needed
        if len(remaining_particles) < 15:

    def _calculate_manifold_params(self, inputs: List[float]) -> Dict:
        """Calculate manifold parameters using quantum circuit"""
        # Define quantum circuit
        @qml.qnode(self.device)
        def circuit(inputs):
            # Encode inputs
            for i in range(self.num_qubits):
                qml.RY(inputs[i % len(inputs)], wires=i)
            
            # Apply parameterized gates
            for i in range(self.num_qubits):
                qml.RX(inputs[i % len(inputs)], wires=i)
            
            # Create entanglement
            for i in range(self.num_qubits - 1):
                qml.CNOT(wires=[i, i + 1])
            
            # Apply final rotation
            for i in range(self.num_qubits):
                qml.RY(inputs[(i + 1) % len(inputs)], wires=i)
            
            # Measure all qubits
            return [qml.expval(qml.PauliZ(i)) for i in range(self.num_qubits)]
        
        # Run circuit
        results = circuit(inputs)
        
        # Convert quantum results to manifold parameters
        # Normalize to 0-1 range
        normalized = [(r + 1) / 2 for r in results]
        
        return {
            "manifold_preference": normalized[0],  # Preference for manifold type
            "curvature_factor": normalized[1] * 2.0 - 1.0,  # -1.0 to 1.0
            "non_linearity": normalized[2] * 0.9 + 0.1,  # 0.1 to 1.0
            "dimension_weight": normalized[3] * 0.8 + 0.2,  # 0.2 to 1.0
            "speed_factor": normalized[4] * 2.0 + 0.5,  # 0.5 to 2.5
            "oscillation": normalized[5] * 0.7  # 0.0 to 0.7

    def update(self, blade_data_dict: Dict[int, Dict], arena_data: Dict, delta_time: float) -> None:
        """Update non-linear movement for all active blades"""
        current_time = arena_data.get("simulation_time", 0)
        
        # Process each active blade
        for blade_id in list(self.active_blades.keys()):
            if blade_id not in self.active_blades:
                continue  # May have been removed during processing
                
            blade_data = self.active_blades[blade_id]
            
            # Check if effect has expired
            if current_time - blade_data["start_time"] >= blade_data["duration"]:
                self._deactivate_non_linear_movement(blade_id)
                continue
            
            # Calculate progress along path
            elapsed = current_time - blade_data["start_time"]
            total_progress = elapsed / blade_data["duration"]
            
            # Calculate target segment
            path = blade_data["movement_path"]
            target_segment = min(len(path) - 1, int(total_progress * (len(path) - 1)))
            
            # If we've moved to a new segment
            if target_segment > blade_data["current_segment"]:
                # Update current segment
                blade_data["current_segment"] = target_segment
            
            # Calculate position within current segment
            segment_progress = (total_progress * (len(path) - 1)) - target_segment
            
            # Get segment endpoints
            if target_segment < len(path) - 1:
                start_pos = path[target_segment][:3]  # 3D coordinates
                end_pos = path[target_segment + 1][:3]
                
                # Interpolate position
                target_pos = [
                    start_pos[0] * (1 - segment_progress) + end_pos[0] * segment_progress,
                    start_pos[1] * (1 - segment_progress) + end_pos[1] * segment_progress,
                    start_pos[2] * (1 - segment_progress) + end_pos[2] * segment_progress
                ]
                
                # Get current position
                current_pos, current_orn = p.getBasePositionAndOrientation(blade_id, physicsClientId=self.physics_client_id)
                
                # Calculate velocity needed to reach target position
                vel_scale = 10.0  # Adjust this to control how quickly the blade follows the path
                target_vel = [
                    (target_pos[0] - current_pos[0]) * vel_scale,
                    (target_pos[1] - current_pos[1]) * vel_scale,
                    (target_pos[2] - current_pos[2]) * vel_scale
                ]
                
                # Apply force to guide blade along path
                mass = blade_data["original_mass"]
                force = [
                    target_vel[0] * mass,
                    target_vel[1] * mass,
                    target_vel[2] * mass
                ]
                
                p.applyExternalForce(
                    blade_id, -1,
                    force,
                    current_pos,
                    p.WORLD_FRAME,
                    physicsClientId=self.physics_client_id
                )
                
                # Store last position for trail effect
                blade_data["last_position"] = current_pos
                
                # Add movement particles

    def spawn_roboblade(self, position: List[float], design_params: Dict) -> int:

        Args:
            position: Initial position [x, y, z]
            design_params: RoboBlade design parameters
            
        Returns:
            blade_id: ID of the created RoboBlade
        """
        # Create RoboBlade collision and visual shapes
        radius = design_params.get("radius", 0.1)

        collision_id = p.createCollisionShape(

        blade_id = p.createMultiBody(

    def update(self, delta_time: float) -> None:
        """Update the entire Beyblade Battle Arena system for one time step"""
        start_time = time.time()
        
        # Update arena physics state
        physics_start = time.time()
        self.arena_physics.update_arena_dynamics(delta_time)
        physics_time = time.time() - physics_start
        
        # Update RoboBlade states
        for blade_id in list(self.active_blades.keys()):
            # Skip if blade has been removed
            if blade_id not in self.active_blades:
                continue
                
            # Get current physics state
            position, orientation = p.getBasePositionAndOrientation(
                blade_id, physicsClientId=self.physics_client_id
            )
            linear_vel, angular_vel = p.getBaseVelocity(
                blade_id, physicsClientId=self.physics_client_id
            )
            
            # Update blade data
            self.active_blades[blade_id].update({
                "position": position,
                "orientation": orientation,
                "linear_velocity": linear_vel,
                "angular_velocity": angular_vel
            })
            
            # Check if blade is out of bounds
            if abs(position[0]) > 10 or abs(position[1]) > 10 or position[2] < -2:
                self._handle_blade_out_of_bounds(blade_id)
                continue
            
            # Natural energy regeneration
            if self.active_blades[blade_id]["energy"] < 100:
                regen_rate = 1.0  # Base regen rate (energy per second)
                energy_rate = self.active_blades[blade_id].get("energy_rate", 1.0)
                self.active_blades[blade_id]["energy"] += regen_rate * energy_rate * delta_time
                self.active_blades[blade_id]["energy"] = min(100, self.active_blades[blade_id]["energy"])
            
            # Update center control time
            distance_from_center = math.sqrt(position[0]**2 + position[1]**2)
            if distance_from_center < 0.5:
                self.active_blades[blade_id]["center_control_time"] += delta_time
                
                # Bonus energy for controlling center
                self.active_blades[blade_id]["energy"] += 1.0 * delta_time
                self.active_blades[blade_id]["energy"] = min(100, self.active_blades[blade_id]["energy"])
        
        # Update arena data
        self.arena_data["simulation_time"] += delta_time
        
        # Update chance vector every 100ms
        if int(self.arena_data["simulation_time"] * 10) % 1 == 0:
            self.arena_data["chance_vector"] = [
                random.uniform(0.9, 1.1),  # luck
                random.uniform(-0.01, 0.01),  # friction_var
                random.uniform(-0.05, 0.05)  # quirk
            ]
        
        # Update quantum superpowers
        quantum_start = time.time()
        self.quantum_superpowers.update_effects(self.active_blades, self.arena_data, delta_time)
        self.dimensional_shapeshift.update(self.active_blades, self.arena_data, delta_time)
        self.non_linear_movement.update(self.active_blades, self.arena_data, delta_time)
        quantum_time = time.time() - quantum_start
        
        # Update AI controllers
        ai_start = time.time()
        self.ai_evolution.update(self.active_blades, self.arena_data, delta_time)
        ai_time = time.time() - ai_start
        
        # Step the physics simulation
        p.stepSimulation(physicsClientId=self.physics_client_id)
        
        # Process collisions
        self._process_collisions()
        
        # Update performance metrics
        total_time = time.time() - start_time
        rendering_time = total_time - physics_time - quantum_time - ai_time
        
        self.performance_metrics.update({
            "physics_time": physics_time,
            "quantum_time": quantum_time,
            "ai_time": ai_time,
            "rendering_time": rendering_time
        })
        
        # Update FPS counter
        self.fps_counter += 1
        current_time = time.time()
        if current_time - self.last_fps_time >= 1.0:
            self.performance_metrics["avg_fps"] = self.fps_counter
            self.fps_counter = 0
            self.last_fps_time = current_time
    
    def _process_collisions(self) -> None:
        """Process collisions between RoboBladez"""
        # Get all contact points
        contact_points = p.getContactPoints(physicsClientId=self.physics_client_id)
        
        # Process each contact point
        for contact in contact_points:
            body1 = contact[1]
            body2 = contact[2]
            
            # Check if both bodies are RoboBladez
            if body1 in self.active_blades and body2 in self.active_blades:
                # Get contact info
                force = contact[9]  # Normal force
                position = contact[5]  # Contact position
                
                # Increment collision counter for both blades
                self.active_blades[body1]["collision_count"] += 1
                self.active_blades[body2]["collision_count"] += 1
                
                # Add energy based on collision
                energy_gain = min(10, force * 0.1)
                self.active_blades[body1]["energy"] += energy_gain
                self.active_blades[body2]["energy"] += energy_gain
                
                self.active_blades[body1]["energy"] = min(100, self.active_blades[body1]["energy"])
                self.active_blades[body2]["energy"] = min(100, self.active_blades[body2]["energy"])
                
                # Calculate damage based on collision intensity
                damage = min(5, force * 0.05)
                
                # Apply damage to integrity
                self.active_blades[body1]["integrity"] -= damage
                self.active_blades[body2]["integrity"] -= damage
                
                # Check for knockout
                if self.active_blades[body1]["integrity"] <= 0:
                    self._handle_blade_knockout(body1, body2)
                
                if self.active_blades[body2]["integrity"] <= 0:
                    self._handle_blade_knockout(body2, body1)
                
                # Create visual effect for collision
                self._create_collision_effect(position, force)
    
    def _handle_blade_knockout(self, defeated_id: int, victor_id: int) -> None:
        """Handle a blade being knocked out"""
        # Award victory points to the victor
        if victor_id in self.active_blades:
            self.active_blades[victor_id]["victory_points"] = self.active_blades[victor_id].get("victory_points", 0) + 1
        
        # Remove the defeated blade
        self._remove_blade(defeated_id)
        
        # Create knockout visual effect
        if defeated_id in self.active_blades:
            position = self.active_blades[defeated_id]["position"]
            self._create_knockout_effect(position)
    
    def _handle_blade_out_of_bounds(self, blade_id: int) -> None:
        """Handle a blade going out of bounds"""
        # Remove the blade
        self._remove_blade(blade_id)
    
    def _remove_blade(self, blade_id: int) -> None:
        """Remove a RoboBlade from the simulation"""
        # Remove from physics simulation
        p.removeBody(blade_id, physicsClientId=self.physics_client_id)
        
        # Remove from active blades
        if blade_id in self.active_blades:
            del self.active_blades[blade_id]
        
        # Remove from arena data
        if blade_id in self.arena_data["active_blades"]:
            self.arena_data["active_blades"].remove(blade_id)
    
    def _create_collision_effect(self, position: List[float], force: float) -> None:
        """Create visual effect for collision"""
        # Scale effect size with collision force
        scale = min(1.0, force * 0.01)
        
        # Create particle burst effect
        num_particles = int(10 * scale)
        for _ in range(num_particles):
            # Random direction
            angle = random.uniform(0, 2 * math.pi)
            velocity = random.uniform(0.5, 1.5) * scale
            
            # Calculate particle trajectory
            px = position[0] + math.cos(angle) * 0.1  # Start slightly offset from collision
            py = position[1] + math.sin(angle) * 0.1
            pz = position[2] + 0.05
            
            # End position
            ex = px + math.cos(angle) * velocity * 0.2
            ey = py + math.sin(angle) * velocity * 0.2
            ez = pz + random.uniform(0.05, 0.15) * scale
            
            # Add particle line
            p.addUserDebugLine(
                [px, py, pz],
                [ex, ey, ez],
                lineColorRGB=[1.0, 0.8, 0.2],  # Yellow-orange
                lineWidth=max(1, int(3 * scale)),
                lifeTime=0.5,
                physicsClientId=self.physics_client_id
            )
    
    def _create_knockout_effect(self, position: List[float]) -> None:
        """Create visual effect for a knockout"""
        # Create expanding ring
        num_segments = 16
        for r in range(5):  # 5 expanding rings
            radius = 0.2 * (r + 1)
            points = []
            
            # Create points for ring
            for i in range(num_segments + 1):
                angle = 2 * math.pi * i / num_segments
                px = position[0] + radius * math.cos(angle)
                py = position[1] + radius * math.sin(angle)
                pz = position[2] + 0.05
                points.append([px, py, pz])
            
            # Create ring segments
            for i in range(num_segments):
                p.addUserDebugLine(
                    points[i],
                    points[i+1],
                    lineColorRGB=[1.0, 0.3, 0.3],  # Red
                    lineWidth=3,
                    lifeTime=0.5 + r * 0.1,
                    physicsClientId=self.physics_client_id
                )
        
        # Add explosion particles
        num_particles = 30
        for _ in range(num_particles):
            # Random direction
            theta = random.uniform(0, math.pi)
            phi = random.uniform(0, 2 * math.pi)
            velocity = random.uniform(0.5, 2.0)
            
            # Calculate particle trajectory (spherical coordinates)
            px = position[0]
            py = position[1]
            pz = position[2]
            
            # End position
            ex = px + velocity * math.sin(theta) * math.cos(phi) * 0.5
            ey = py + velocity * math.sin(theta) * math.sin(phi) * 0.5
            ez = pz + velocity * math.cos(theta) * 0.5
            
            # Add particle line
            p.addUserDebugLine(
                [px, py, pz],
                [ex, ey, ez],
                lineColorRGB=[1.0, 0.3, 0.3],  # Red
                lineWidth=2,
                lifeTime=random.uniform(0.3, 0.8),
                physicsClientId=self.physics_client_id
            )
    
    def activate_superpower(self, blade_id: int, superpower_name: str) -> bool:
        """
        Activate a superpower for a RoboBlade
        
        Args:
            blade_id: ID of the RoboBlade
            superpower_name: Name of the superpower to activate
            
        Returns:
            Success status
        """
        # Check if blade exists
        if blade_id not in self.active_blades:
            return False
        
        # Determine which system handles this superpower
        if superpower_name in ["hyperspatial_projection", "quantum_fluid_dynamics", 
                              "topological_vortex", "dimensional_shapeshifting"]:
            # Quantum superpowers
            return self.quantum_superpowers.activate_superpower(
                blade_id, superpower_name, self.active_blades[blade_id], self.arena_data
            )
        
        elif superpower_name in ["hyperdimensional_shift", "non_euclidean_geometry",
                                "manifold_projection", "phase_space_manipulation"]:
            # Dimensional superpower
            method_name = f"activate_{superpower_name}"
            if hasattr(self.dimensional_shapeshift, method_name):
                method = getattr(self.dimensional_shapeshift, method_name)
                return method(blade_id, self.active_blades[blade_id], self.arena_data)
            return False
        
        elif superpower_name == "4d_shapeshifting":
            # Shapeshifting
            return self.dimensional_shapeshift.activate_4d_shapeshifting(
                blade_id, self.active_blades[blade_id], self.arena_data
            )
        
        elif superpower_name == "non_linear_movement":
            # Non-linear movement
            return self.non_linear_movement.activate_non_linear_movement(
                blade_id, self.active_blades[blade_id], self.arena_data
            )
        
        else:
            print(f"Unknown superpower: {superpower_name}")
            return False

    def initialize_ai_controller(self, blade_id: int, design_params: Dict) -> None:
        """Initialize an AI controller for a RoboBlade"""
        strategy = design_params.get("ai_strategy", "balanced")
        
        # Get strategy template
        template = self.strategy_templates.get(strategy, self.strategy_templates["balanced"])
        
        # Create neural network model
        model = self._create_neural_network()
        
        # Store controller data
        self.controllers[blade_id] = {
            "model": model,
            "strategy": strategy,
            "weights": template["weights"].copy(),
            "preferred_superpowers": template["preferred_superpowers"].copy(),
            "experience_buffer": [],
            "generations_evolved": 0,
            "battle_stats": {
                "victories": 0,

                "superpowers_used": 0
            }
        }
    
    def _create_neural_network(self) -> torch.nn.Module:
        """Create a neural network model for the AI controller"""
        model = torch.nn.Sequential(
            torch.nn.Linear(self.input_size, self.hidden_size),
            torch.nn.ReLU(),
            torch.nn.Linear(self.hidden_size, self.hidden_size),
            torch.nn.ReLU(),
            torch.nn.Linear(self.hidden_size, self.output_size),
            torch.nn.Tanh()
        )
        return model
    
    def update(self, blade_data_dict: Dict[int, Dict], arena_data: Dict, delta_time: float) -> None:
        """Update all AI controllers"""
        for blade_id, controller_data in list(self.controllers.items()):
            # Skip if blade no longer exists
            if blade_id not in blade_data_dict:
                if blade_id in self.controllers:
                    del self.controllers[blade_id]
                continue
            
            # Get current blade data
            blade_data = blade_data_dict[blade_id]
            
            # Get state observation
            state = self._get_state_observation(blade_id, blade_data, blade_data_dict, arena_data)
            
            # Get action from neural network
            action = self._get_action(blade_id, state)
            
            # Apply action
            self._apply_action(blade_id, action, blade_data, arena_data)
            
            # Store experience
            reward = self._calculate_reward(blade_id, blade_data, action)
            self.controllers[blade_id]["experience_buffer"].append({
                "state": state,
                "action": action,
                "reward": reward
            })
            
            # Limit buffer size
            if len(self.controllers[blade_id]["experience_buffer"]) > 1000:
                self.controllers[blade_id]["experience_buffer"].pop(0)
    
    def _get_state_observation(self, blade_id: int, blade_data: Dict, 
                              blade_data_dict: Dict[int, Dict], arena_data: Dict) -> torch.Tensor:
        """Get state observation for a RoboBlade"""
        # Extract position, orientation, velocities
        position = blade_data["position"]
        orientation = blade_data["orientation"]
        linear_vel = blade_data["linear_velocity"]
        angular_vel = blade_data["angular_velocity"]
        
        # Calculate distance and direction to center
        distance_to_center = math.sqrt(position[0]**2 + position[1]**2)
        direction_to_center = [-position[0]/max(0.1, distance_to_center), 
                              -position[1]/max(0.1, distance_to_center)]
        
        # Find closest opponent
        closest_opponent_distance = float('inf')
        closest_opponent_direction = [0, 0]
        closest_opponent_velocity = [0, 0, 0]
        
        for other_id, other_data in blade_data_dict.items():
            if other_id != blade_id:
                other_pos = other_data["position"]
                dist = math.sqrt((position[0] - other_pos[0])**2 + 
                                (position[1] - other_pos[1])**2)
                
                if dist < closest_opponent_distance:
                    closest_opponent_distance = dist
                    
                    # Calculate direction to opponent
                    dir_x = (other_pos[0] - position[0]) / max(0.1, dist)
                    dir_y = (other_pos[1] - position[1]) / max(0.1, dist)
                    closest_opponent_direction = [dir_x, dir_y]
                    
                    # Get opponent velocity
                    closest_opponent_velocity = other_data["linear_velocity"]
        
        # Get energy and integrity
        energy = blade_data["energy"] / 100.0  # Normalize to 0-1
        integrity = blade_data["integrity"] / 100.0  # Normalize to 0-1
        
        # Get arena data
        chance_vector = arena_data["chance_vector"]
        
        # Combine all features into state vector
        state = [
            # Position (normalized)
            position[0] / 10.0, position[1] / 10.0, position[2] / 2.0,
            
            # Velocity (normalized)
            linear_vel[0] / 5.0, linear_vel[1] / 5.0, linear_vel[2] / 5.0,
            
            # Angular velocity (normalized)
            angular_vel[0] / 10.0, angular_vel[1] / 10.0, angular_vel[2] / 10.0,
            
            # Distance and direction to center
            distance_to_center / 10.0, direction_to_center[0], direction_to_center[1],
            
            # Closest opponent info
            closest_opponent_distance / 10.0, 
            closest_opponent_direction[0], closest_opponent_direction[1],
            closest_opponent_velocity[0] / 5.0, closest_opponent_velocity[1] / 5.0,
            
            # Energy and integrity
            energy, integrity,
            
            # Arena chance vector
            chance_vector[0]
        ]
        
        return torch.tensor(state, dtype=torch.float32)
    
    def _get_action(self, blade_id: int, state: torch.Tensor) -> torch.Tensor:
        """Get action from neural network"""
        model = self.controllers[blade_id]["model"]
        
        with torch.no_grad():
            action = model(state)
        
        return action
    
    def _apply_action(self, blade_id: int, action: torch.Tensor, blade_data: Dict, arena_data: Dict) -> None:
        """Apply AI action to the RoboBlade"""
        # Convert action tensor to list
        action_list = action.numpy().tolist()
        
        # Extract movement and rotation components
        move_x = action_list[0]  # -1 to 1
        move_y = action_list[1]  # -1 to 1
        rotation = action_list[2]  # -1 to 1
        
        # Extract superpower activation signals (thresholded)
        superpower_signals = [a > 0.7 for a in action_list[3:]]
        
        # Calculate force to apply
        force_magnitude = 5.0 * blade_data.get("original_mass", 1.0)
        force_x = move_x * force_magnitude
        force_y = move_y * force_magnitude
        
        # Apply force
        p.applyExternalForce(
            blade_id, -1,
            [force_x, force_y, 0],
            blade_data["position"],
            p.WORLD_FRAME,
            physicsClientId=self.physics_client_id
        )
        
        # Apply torque for rotation
        torque_magnitude = 1.0 * blade_data.get("original_mass", 1.0)
        p.applyExternalTorque(
            blade_id, -1,
            [0, 0, rotation * torque_magnitude],
            p.WORLD_FRAME,
            physicsClientId=self.physics_client_id
        )
        
        # Check for superpower activation
        preferred_superpowers = self.controllers[blade_id]["preferred_superpowers"]
# Activate superpowers based on AI decision
        for i, should_activate in enumerate(superpower_signals):
            if should_activate and i < len(preferred_superpowers):
                superpower_name = preferred_superpowers[i]
                
                # Check if on cooldown
                cooldown_end = blade_data.get("superpower_cooldowns", {}).get(superpower_name, 0)
                if arena_data["simulation_time"] >= cooldown_end:
                    # Attempt to activate
                    # Note: This is a placeholder - the actual activation
                    # would need to be handled by the main system
                    print(f"AI trying to activate {superpower_name} for blade {blade_id}")
                    
                    # Track superpower usage
                    self.controllers[blade_id]["battle_stats"]["superpowers_used"] += 1
    
    def _calculate_reward(self, blade_id: int, blade_data: Dict, action: torch.Tensor) -> float:
        """Calculate reward for reinforcement learning"""
        # Base reward
        reward = 0.0
        
        # Strategy weights
        weights = self.controllers[blade_id]["weights"]
        
        # Reward for survival
        reward += 0.1  # Small reward per time step
        
        # Reward for center control
        distance_from_center = math.sqrt(blade_data["position"][0]**2 + blade_data["position"][1]**2)
        center_reward = max(0, (1.0 - distance_from_center / 5.0)) * weights["center_control"]
        reward += center_reward
        
        # Reward for collisions
        if blade_data.get("collision_count", 0) > self.controllers[blade_id].get("last_collision_count", 0):
            collision_reward = weights["collision_seeking"]
            reward += collision_reward
            self.controllers[blade_id]["last_collision_count"] = blade_data.get("collision_count", 0)
        
        # Penalty for low integrity
        integrity = blade_data["integrity"] / 100.0
        if integrity < 0.3:
            integrity_penalty = (0.3 - integrity) * 2.0
            reward -= integrity_penalty
        
        # Reward for energy management
        energy = blade_data["energy"] / 100.0
        energy_reward = energy * weights["energy_conservation"] * 0.1
        reward += energy_reward
        
        return reward
    
    def evolve_controller(self, blade_id: int, generations: int = 10) -> Dict:
        """
        Evolve the AI controller for a RoboBlade using genetic algorithm
        
        Args:
            blade_id: ID of the RoboBlade
            generations: Number of generations to evolve
            
        Returns:
            Evolution results
        """
        if blade_id not in self.controllers:
            return {"success": False, "message": "No controller found for this blade"}
        
        controller_data = self.controllers[blade_id]
        
        # Create initial population
        population = self._create_population(controller_data["model"])
        
        # Evolution metrics
        best_fitness = -float('inf')
        best_model = None
        fitness_history = []
        
        # Run evolution for specified generations
        for generation in range(generations):
            # Evaluate fitness of each model in population
            fitness_scores = []
            
            for model in population:
                # Calculate fitness based on experience buffer
                fitness = self._evaluate_model_fitness(model, controller_data["experience_buffer"])
                fitness_scores.append(fitness)
                
                # Track best model
                if fitness > best_fitness:
                    best_fitness = fitness
                    best_model = self._clone_model(model)
            
            # Record average fitness
            avg_fitness = sum(fitness_scores) / len(fitness_scores)
            fitness_history.append(avg_fitness)
            
            # Create next generation
            next_population = []
            
            # Elitism: keep best model
            best_idx = fitness_scores.index(max(fitness_scores))
            next_population.append(self._clone_model(population[best_idx]))
            
            # Create rest of population through selection, crossover, mutation
            while len(next_population) < self.population_size:
                # Selection
                parent1 = self._tournament_selection(population, fitness_scores)
                parent2 = self._tournament_selection(population, fitness_scores)
                
                # Crossover
                if random.random() < self.crossover_rate:
                    child = self._crossover(parent1, parent2)
                else:
                    child = self._clone_model(parent1)
                
                # Mutation
                child = self._mutate(child)
                
                # Add to next generation
                next_population.append(child)
            
            # Update population
            population = next_population
        
        # Update controller with best model
        if best_model is not None:
            controller_data["model"] = best_model
            controller_data["generations_evolved"] += generations
        
        # Return evolution results
        return {
            "success": True,
            "generations": generations,
            "final_fitness": best_fitness,
            "fitness_history": fitness_history,
            "total_generations_evolved": controller_data["generations_evolved"]
        }
    
    def _create_population(self, base_model: torch.nn.Module) -> List[torch.nn.Module]:
        """Create initial population based on base model"""
        population = [base_model]  # Include original model
        
        # Create variations
        for _ in range(self.population_size - 1):
            # Clone and mutate
            model = self._clone_model(base_model)
            model = self._mutate(model, mutation_strength=0.3)  # Stronger mutation for initial diversity
            population.append(model)
        
        return population
    
    def _clone_model(self, model: torch.nn.Module) -> torch.nn.Module:
        """Create a deep copy of a model"""
        clone = self._create_neural_network()  # Create new model with same architecture
        clone.load_state_dict(model.state_dict())  # Copy weights
        return clone
    
    def _evaluate_model_fitness(self, model: torch.nn.Module, experience_buffer: List[Dict]) -> float:
        """Evaluate fitness of a model based on experience"""
        if not experience_buffer:
            return 0.0
        
        total_reward = 0.0
        
        # Sample from experience buffer
        samples = random.sample(experience_buffer, min(100, len(experience_buffer)))
        
        for experience in samples:
            state = experience["state"]
            actual_reward = experience["reward"]
            
            # Generate action using this model
            with torch.no_grad():
                action = model(state)
            
            # Compare action to the one that was taken
            # Original action is not relevant here as we want to know
            # how well this model would perform on this state
            predicted_value = torch.mean(action).item()  # Simple approximation
            
            # Reward prediction error contributes to fitness
            prediction_error = abs(predicted_value - actual_reward)
            
            # For samples with positive reward, we want actions that would
            # lead to higher rewards
            if actual_reward > 0:
                total_reward += actual_reward * (1.0 - prediction_error * 0.5)
            else:
                # For negative rewards, we want to avoid those states
                total_reward += actual_reward * prediction_error * 0.5
        
        return total_reward / len(samples)
    
    def _tournament_selection(self, population: List[torch.nn.Module], 
                             fitness_scores: List[float]) -> torch.nn.Module:
        """Select a model using tournament selection"""
        # Select tournament candidates
        tournament_size = 3
        candidates = random.sample(range(len(population)), tournament_size)
        
        # Find best candidate
        best_idx = candidates[0]
        best_fitness = fitness_scores[best_idx]
        
        for idx in candidates[1:]:
            if fitness_scores[idx] > best_fitness:
                best_idx = idx
                best_fitness = fitness_scores[idx]
        
        return population[best_idx]
    
    def _crossover(self, parent1: torch.nn.Module, parent2: torch.nn.Module) -> torch.nn.Module:
        """Create a child model by crossing over two parent models"""
        child = self._create_neural_network()
        
        # Get state dictionaries
        parent1_dict = parent1.state_dict()
        parent2_dict = parent2.state_dict()
        child_dict = child.state_dict()
        
        # Crossover each parameter
        for name in child_dict:
            # Layer by layer crossover
            # For each layer, decide which parent to inherit from
            if random.random() < 0.5:
                child_dict[name] = parent1_dict[name].clone()
            else:
                child_dict[name] = parent2_dict[name].clone()
        
        # Load state dict into child
        child.load_state_dict(child_dict)
        
        return child
    
    def _mutate(self, model: torch.nn.Module, mutation_strength: float = None) -> torch.nn.Module:
        """Mutate a model by adding random noise to weights"""
        if mutation_strength is None:
            mutation_strength = self.mutation_rate
            
        state_dict = model.state_dict()
        
        for name, param in state_dict.items():
            # Add random noise to parameters
            if random.random() < mutation_strength:
                # Scale noise by parameter values
                noise = torch.randn_like(param) * param.std() * mutation_strength
                param.add_(noise)
        
        return model
