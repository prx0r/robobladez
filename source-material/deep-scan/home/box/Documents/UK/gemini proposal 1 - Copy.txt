# --- Abstract Base Classes (for modularity and extensibility) ---
class PCACRModule(nn.Module, ABC):
    def __init__(self, config: PCACRConfig):
        super().__init__()
        self.config = config
        self.module_name = self.__class__.__name__
        logger.info(f"Initializing {self.module_name}...")

    @abstractmethod
    def forward(self, *args, **kwargs) -> Any:
        pass

    def save_checkpoint(self, step: int):
        Path(self.config.output_dir / self.module_name).mkdir(parents=True, exist_ok=True)
        checkpoint_path = self.config.output_dir / self.module_name / f"checkpoint_step_{step}.pt"
        torch.save(self.state_dict(), checkpoint_path)
        logger.info(f"Saved {self.module_name} checkpoint to {checkpoint_path}")

    def load_checkpoint(self, checkpoint_path: Path):
        if checkpoint_path.exists():
            self.load_state_dict(torch.load(checkpoint_path, map_location=DEVICE))
            logger.info(f"Loaded {self.module_name} checkpoint from {checkpoint_path}")
        else:
            logger.warning(f"{self.module_name} checkpoint not found at {checkpoint_path}")

class SensoryProcessingUnit(PCACRModule):
    """
    Processes raw sensory input (audio, text) into features for downstream modules.
    Grounded in robust ASR/NLP models and standard feature extraction.
    """
    def __init__(self, config: PCACRConfig):
        super().__init__(config)
        # Audio Processing
        # TODO: Replace whisperx placeholder with actual whisperx loading and usage.
        # self.asr_model = whisperx.load_model("base.en", device=DEVICE, compute_type="float32")
        # For now, using torchaudio transforms as a stand-in for acoustic feature processing part
        self.mel_spectrogram = torchaudio.transforms.MelSpectrogram(
            sample_rate=config.audio_sample_rate,
            n_fft=config.audio_n_fft,
            hop_length=config.audio_hop_length,
            n_mels=config.audio_n_mels
        ).to(DEVICE)
        self.mfcc_transform = torchaudio.transforms.MFCC(
            sample_rate=config.audio_sample_rate,
            n_mfcc=13, # Standard number of MFCCs
            melkwargs={'n_fft': config.audio_n_fft, 'hop_length': config.audio_hop_length, 'n_mels': config.audio_n_mels}
        ).to(DEVICE)

        # Text Processing
        self.tokenizer = AutoTokenizer.from_pretrained(config.text_model_name)
        self.text_embedding_model = AutoModel.from_pretrained(config.text_model_name).to(DEVICE)
        self.text_feature_dim = self.text_embedding_model.config.hidden_size

        # Projection layer to unify feature dimensions if needed (e.g., to config.d_model)
        # This example assumes text_feature_dim might be different from d_model
        if self.text_feature_dim != self.config.d_model:
            self.text_projection = nn.Linear(self.text_feature_dim, config.d_model).to(DEVICE)
        else:
            self.text_projection = nn.Identity()

        # Example for audio features: assuming MFCCs (13 dim) + Mel Spectrogram (n_mels dim)
        # Adjust this based on actual features used and how they are combined
        audio_feature_dim = 13 + config.audio_n_mels
        if audio_feature_dim != self.config.d_model:
            self.audio_projection = nn.Linear(audio_feature_dim, config.d_model).to(DEVICE)
        else:
            self.audio_projection = nn.Identity()

        logger.info(f"{self.module_name} initialized with text model: {config.text_model_name}")

    def process_audio(self, audio_waveform: Optional[torch.Tensor]) -> Dict[str, Optional[torch.Tensor]]:
        """
        Processes an audio waveform to extract features and text.
        Args:
            audio_waveform (torch.Tensor): Raw audio waveform [channels, samples].
        Returns:
            A dictionary containing 'transcribed_text' (str) and 'acoustic_features' (Tensor).
        """
        if audio_waveform is None:
            return {"transcribed_text": None, "acoustic_features": None, "raw_acoustic_features": None}

        audio_waveform = audio_waveform.to(DEVICE)
        
        # TODO: Implement actual ASR using a library like Whisper
        # For now, this is a placeholder for transcription.
        # result = self.asr_model.transcribe(audio_waveform.cpu().numpy().squeeze()) # Assuming single channel
        # transcribed_text = " ".join([segment['text'] for segment in result['segments']])
        transcribed_text = "Placeholder: ASR not fully implemented in this skeleton."
        logger.debug(f"SPU: Transcribed text (placeholder): {transcribed_text}")

        # Acoustic features
        mel_spec = self.mel_spectrogram(audio_waveform).squeeze(0) # [n_mels, time_frames]
        mfccs = self.mfcc_transform(audio_waveform).squeeze(0)    # [n_mfcc, time_frames]

        # Pooling over time to get fixed-size vectors
        mel_spec_pooled = torch.mean(mel_spec, dim=1)
        mfccs_pooled = torch.mean(mfccs, dim=1)
        
        raw_acoustic_features = torch.cat([mfccs_pooled, mel_spec_pooled], dim=0) # [n_mfcc + n_mels]
        projected_acoustic_features = self.audio_projection(raw_acoustic_features) # [d_model]

        return {
            "transcribed_text": transcribed_text,
            "acoustic_features": projected_acoustic_features,
            "raw_acoustic_features": {"mfcc": mfccs, "mel_spec": mel_spec} # For more detailed analysis
        }

    def process_text(self, text_input: Optional[str]) -> Dict[str, Optional[torch.Tensor]]:
        """
        Processes text input to extract semantic embeddings.
        Args:
            text_input (str): Raw text string.
        Returns:
            A dictionary containing 'semantic_features' (Tensor).
        """
        if text_input is None:
            return {"semantic_features": None}

        inputs = self.tokenizer(text_input, return_tensors="pt", truncation=True, padding="max_length", max_length=self.config.max_seq_len).to(DEVICE)
        with torch.no_grad(): # Usually don't fine-tune the base embedder initially
            outputs = self.text_embedding_model(**inputs)
        
        # Using [CLS] token embedding or mean pooling of last hidden states
        # Here, using mean pooling for sentence-level representation
        semantic_features_raw = outputs.last_hidden_state.mean(dim=1).squeeze(0) # [text_feature_dim]
        semantic_features_projected = self.text_projection(semantic_features_raw) # [d_model]
        
        logger.debug(f"SPU: Processed text, feature shape: {semantic_features_projected.shape}")
        return {"semantic_features": semantic_features_projected}

    def forward(self, audio_input: Optional[torch.Tensor] = None, text_input: Optional[str] = None) -> Dict[str, Any]:
        """
        Main forward pass for SPU.
        Args:
            audio_input (torch.Tensor, optional): Audio waveform.
            text_input (str, optional): Text string.
        Returns:
            A dictionary containing processed 'transcribed_text', 'acoustic_features', 
            'semantic_features', and 'raw_inputs'.
        """
        audio_data = self.process_audio(audio_input)
        
        # Use transcribed text if original text_input is None
        effective_text_input = text_input if text_input is not None else audio_data["transcribed_text"]
        text_data = self.process_text(effective_text_input)

        # Combine outputs
        processed_sensory_data = {
            "timestamp": time.time(),
            "transcribed_text": audio_data["transcribed_text"], # Could be None if no audio
            "original_text": text_input, # Preserve original text if provided
            "acoustic_features": audio_data["acoustic_features"], # Projected, [d_model] or None
            "semantic_features": text_data["semantic_features"], # Projected, [d_model] or None
            "raw_acoustic_data": audio_data["raw_acoustic_features"] # Unprojected, detailed features
        }
        logger.info(f"{self.module_name} processed inputs.")
        return processed_sensory_data

class PredictiveProcessingUnit(PCACRModule):
    """
    Builds a generative model of sensory inputs, generates predictions, and calculates prediction errors.
    Inspired by Predictive Coding (Seth, Friston).
    """
    def __init__(self, config: PCACRConfig):
        super().__init__(config)
        # Generative model to predict next sensory state (features)
        # Input: current combined sensory features + previous action (optional)
        # Output: predicted next combined sensory features
        # Using an LSTM as a simple recurrent generative model. Could be a Transformer.
        # Input dimension: d_model (sensory) + d_model (action encoding, if used)
        # For simplicity, let's assume action is encoded to d_model. If no action, then d_model.
        self.generative_model_input_dim = config.d_model # If only predicting next sensory state
        # If predicting based on current sensory and previous action:
        # self.generative_model_input_dim = config.d_model + config.asgu_action_representation_dim 
        
        self.generative_model = nn.LSTM(
            input_size=self.generative_model_input_dim,
            hidden_size=config.ppu_recurrent_hidden_size,
            num_layers=config.ppu_recurrent_layers,
            batch_first=True # Expects input as (batch, seq, feature)
        ).to(DEVICE)
        
        # Output layer to map LSTM hidden state to predicted sensory feature dimension (d_model)
        self.prediction_head = nn.Linear(config.ppu_recurrent_hidden_size, config.d_model).to(DEVICE)

        self.hidden_state = None # To store LSTM hidden state across calls for stateless forward
        self.current_prediction_error = torch.zeros(config.d_model, device=DEVICE, dtype=DTYPE)
        
        # Hierarchy of prediction errors (conceptual for now)
        # TODO: Implement hierarchical predictive coding if desired.
        # self.error_hierarchy = [torch.zeros(config.d_model, device=DEVICE) for _ in range(3)] # e.g., 3 levels
        logger.info(f"{self.module_name} initialized with LSTM-based generative model.")

    def initialize_hidden_state(self, batch_size=1):
        # For LSTM: (num_layers, batch_size, hidden_size)
        h0 = torch.zeros(self.config.ppu_recurrent_layers, batch_size, self.config.ppu_recurrent_hidden_size).to(DEVICE)
        c0 = torch.zeros(self.config.ppu_recurrent_layers, batch_size, self.config.ppu_recurrent_hidden_size).to(DEVICE)
        self.hidden_state = (h0, c0)

    def generate_prediction(self, current_sensory_features: torch.Tensor, prev_action_representation: Optional[torch.Tensor] = None) -> torch.Tensor:
        """
        Generates a prediction of the next sensory state.
        Args:
            current_sensory_features (torch.Tensor): Current sensory features [d_model].
            prev_action_representation (torch.Tensor, optional): Representation of the last action taken.
        Returns:
            torch.Tensor: Predicted next sensory features [d_model].
        """
        if self.hidden_state is None:
            self.initialize_hidden_state(batch_size=1) # Assuming batch_size 1 for single instance processing

        # Prepare input for LSTM: (batch, seq_len, features) -> (1, 1, features)
        model_input = current_sensory_features.unsqueeze(0).unsqueeze(0)
        
        # TODO: Incorporate prev_action_representation if architecture supports it
        # if prev_action_representation is not None:
        #     model_input = torch.cat([model_input, prev_action_representation.unsqueeze(0).unsqueeze(0)], dim=2)

        lstm_output, self.hidden_state = self.generative_model(model_input, self.hidden_state)
        predicted_features = self.prediction_head(lstm_output.squeeze(0)) # Remove batch, seq_len=1 dim
        
        logger.debug(f"PPU: Generated prediction shape: {predicted_features.shape}")
        return predicted_features.squeeze(0) # Return [d_model]

    def calculate_prediction_error(self, actual_sensory_features: torch.Tensor, predicted_sensory_features: torch.Tensor) -> torch.Tensor:
        """
        Calculates the prediction error between actual and predicted sensory features.
        Args:
            actual_sensory_features (torch.Tensor): Actual incoming sensory features [d_model].
            predicted_sensory_features (torch.Tensor): Predicted sensory features [d_model].
        Returns:
            torch.Tensor: Prediction error [d_model].
        """
        # Simple difference, could be weighted by precision later
        prediction_error = actual_sensory_features - predicted_sensory_features
        self.current_prediction_error = prediction_error # Store for internal state / learning
        logger.debug(f"PPU: Calculated prediction error norm: {torch.norm(prediction_error).item()}")
        return prediction_error

    def update_generative_model(self, prediction_error: torch.Tensor, optimizer: optim.Optimizer):
        """
        Updates the generative model parameters to minimize prediction error.
        This method would be called by the LAU.
        Note: This implies the PPU's parameters are part of the overall model optimized by LAU.
              The PPU itself doesn't have its own optimizer here, it's trained externally.
        """
        # This is conceptual. The actual loss computation and optimizer.step()
        # would happen in the LAU or main training loop.
        # The PPU provides the error, the LAU uses it to update the PPU's parameters.
        # Example: loss = torch.mean(prediction_error**2)
        # loss.backward() -> handled by main training loop
        # optimizer.step() -> handled by main training loop
        pass # Actual update logic is external to PPU's forward pass

    def forward(self, current_sensory_combined: torch.Tensor, 
                actual_next_sensory_combined: Optional[torch.Tensor] = None) -> Dict[str, torch.Tensor]:
        """
        Main forward pass for PPU.
        Args:
            current_sensory_combined (torch.Tensor): Combined sensory features from SPU for current step [d_model].
                                                   This typically aggregates acoustic and semantic features.
            actual_next_sensory_combined (torch.Tensor, optional): Actual combined sensory features for the *next* step,
                                                                 used for calculating error during training/inference if available.
        Returns:
            A dictionary containing 'predicted_next_sensory_features' and 'prediction_error'.
        """
        # Detach hidden state to prevent gradients from flowing back indefinitely in long sequences
        # if self.hidden_state is not None:
        #     self.hidden_state = (self.hidden_state[0].detach(), self.hidden_state[1].detach())

        predicted_next_sensory = self.generate_prediction(current_sensory_combined)
        
        prediction_error = None
        if actual_next_sensory_combined is not None:
            prediction_error = self.calculate_prediction_error(actual_next_sensory_combined, predicted_next_sensory)
        else:
            # During pure inference without ground truth for next step, error is undefined or based on internal consistency
            self.current_prediction_error = torch.zeros_like(predicted_next_sensory) # Reset or estimate

        logger.info(f"{self.module_name} generated prediction.")
        return {
            "predicted_next_sensory_features": predicted_next_sensory, # Prediction for T+1 based on T
            "current_prediction_error": prediction_error if prediction_error is not None else self.current_prediction_error # Error of prediction for T based on T-1 prediction and T actual
        }

class InternalStateHomeostasisUnit(PCACRModule):
    """
    Manages internal state dynamics (e.g., using Hodgkin-Huxley like dynamics for excitability)
    and homeostatic regulation. Generates affective signals (valence, arousal).
    """
    def __init__(self, config: PCACRConfig):
        super().__init__(config)
        # --- Hodgkin-Huxley Neuron Group ---
        self.num_neurons = config.ishu_hh_neurons
        self.V = nn.Parameter(torch.full((self.num_neurons,), config.ishu_hh_EL, device=DEVICE, dtype=DTYPE)) # Initial voltage at resting potential
        self.m = nn.Parameter(torch.full((self.num_neurons,), self._alpha_m(self.V) / (self._alpha_m(self.V) + self._beta_m(self.V)), device=DEVICE, dtype=DTYPE))
        self.h = nn.Parameter(torch.full((self.num_neurons,), self._alpha_h(self.V) / (self._alpha_h(self.V) + self._beta_h(self.V)), device=DEVICE, dtype=DTYPE))
        self.n = nn.Parameter(torch.full((self.num_neurons,), self._alpha_n(self.V) / (self._alpha_n(self.V) + self._beta_n(self.V)), device=DEVICE, dtype=DTYPE))
        
        # Input mapping: project d_model features to an input current for HH neurons
        self.input_current_projection = nn.Linear(config.d_model, self.num_neurons).to(DEVICE)

        # --- Homeostatic Regulation ---
        self.homeostatic_setpoints = {
            var: torch.tensor(details["target"], device=DEVICE, dtype=DTYPE)
            for var, details in config.ishu_homeostatic_variables.items()
        }
        self.homeostatic_tolerances = {
            var: torch.tensor(details["tolerance"], device=DEVICE, dtype=DTYPE)
            for var, details in config.ishu_homeostatic_variables.items()
        }
        self.homeostatic_weights = { # Importance of regulating this variable
            var: torch.tensor(details["weight"], device=DEVICE, dtype=DTYPE)
            for var, details in config.ishu_homeostatic_variables.items()
        }
        
        # --- Affective State Generation ---
        # Input: Homeostatic errors, overall network activity (e.g. mean HH firing rate)
        # Output: Valence, Arousal
        # Using a simple LSTM to integrate homeostatic errors over time for affect
        self.affect_input_dim = len(config.ishu_homeostatic_variables) + 1 # errors + firing rate
        self.affect_lstm = nn.LSTM(
            input_size=self.affect_input_dim,
            hidden_size=config.ishu_affect_hidden_size,
            num_layers=config.ishu_affect_lstm_layers,
            batch_first=True
        ).to(DEVICE)
        self.affect_head = nn.Linear(config.ishu_affect_hidden_size, 2).to(DEVICE) # Output: [valence, arousal]
        self.affect_hidden_state = None

        self.current_affective_state = torch.zeros(2, device=DEVICE, dtype=DTYPE) # [valence, arousal]
        self.dt = 0.1 #ms - simulation time step for HH model (needs to be small)

        logger.info(f"{self.module_name} initialized with {self.num_neurons} HH neurons.")

    def _alpha_m(self, V): return 0.1 * (V + 40.0) / (1.0 - torch.exp(-(V + 40.0) / 10.0))
    def _beta_m(self, V):  return 4.0 * torch.exp(-(V + 65.0) / 18.0)
    def _alpha_h(self, V): return 0.07 * torch.exp(-(V + 65.0) / 20.0)
    def _beta_h(self, V):  return 1.0 / (1.0 + torch.exp(-(V + 35.0) / 10.0))
    def _alpha_n(self, V): return 0.01 * (V + 55.0) / (1.0 - torch.exp(-(V + 55.0) / 10.0))
    def _beta_n(self, V):  return 0.125 * torch.exp(-(V + 65.0) / 80.0)

    def _hodgkin_huxley_step(self, input_current: torch.Tensor):
        """Performs one step of Hodgkin-Huxley dynamics."""
        # Gating variable updates
        dm = (self._alpha_m(self.V) * (1.0 - self.m) - self._beta_m(self.V) * self.m) * self.dt
        dh = (self._alpha_h(self.V) * (1.0 - self.h) - self._beta_h(self.V) * self.h) * self.dt
        dn = (self._alpha_n(self.V) * (1.0 - self.n) - self._beta_n(self.V) * self.n) * self.dt
        self.m.data += dm
        self.h.data += dh
        self.n.data += dn

        # Ionic currents
        I_Na = self.config.ishu_hh_gNa * (self.m**3) * self.h * (self.V - self.config.ishu_hh_ENa)
        I_K = self.config.ishu_hh_gK * (self.n**4) * (self.V - self.config.ishu_hh_EK)
        I_L = self.config.ishu_hh_gL * (self.V - self.config.ishu_hh_EL)
        
        # Membrane potential update
        dV = (input_current - I_Na - I_K - I_L) / self.config.ishu_hh_Cm * self.dt
        self.V.data += dV
        
        # Spike detection (simple threshold)
        spikes = (self.V > -40.0).float() # Threshold for firing
        # Reset voltage for neurons that spiked (optional, simple reset)
        # self.V.data[spikes == 1] = self.config.ishu_hh_EL 
        return spikes, self.V.data # Return spikes and current voltages

    def _update_affective_state(self, homeostatic_errors: torch.Tensor, mean_firing_rate: torch.Tensor):
        if self.affect_hidden_state is None:
            h0 = torch.zeros(self.config.ishu_affect_lstm_layers, 1, self.config.ishu_affect_hidden_size).to(DEVICE)
            c0 = torch.zeros(self.config.ishu_affect_lstm_layers, 1, self.config.ishu_affect_hidden_size).to(DEVICE)
            self.affect_hidden_state = (h0,c0)

        affect_input = torch.cat([homeostatic_errors, mean_firing_rate.unsqueeze(0)], dim=0)
        affect_input = affect_input.unsqueeze(0).unsqueeze(0) # (batch, seq, feature)

        lstm_out, self.affect_hidden_state = self.affect_lstm(affect_input, self.affect_hidden_state)
        affect_output = self.affect_head(lstm_out.squeeze(0)) # [valence, arousal]
        self.current_affective_state = torch.tanh(affect_output.squeeze(0)) # constrain between -1 and 1
        logger.debug(f"ISHU: Updated affective state: Valence={self.current_affective_state[0].item():.2f}, Arousal={self.current_affective_state[1].item():.2f}")


    def forward(self, ppu_outputs: Dict[str, torch.Tensor], 
                iiwu_metrics: Optional[Dict[str, Any]] = None) -> Dict[str, Any]:
        """
        Main forward pass for ISHU.
        Args:
            ppu_outputs (Dict): Outputs from PPU, must include 'current_prediction_error'.
            iiwu_metrics (Dict, optional): Metrics from IIWU, e.g., 'phi_value', 'network_coherence'.
        Returns:
            A dictionary containing 'hh_spikes', 'hh_voltages', 'homeostatic_errors', 'affective_state'.
        """
        prediction_error_signal = ppu_outputs.get("current_prediction_error")
        if prediction_error_signal is None:
            # If no error signal (e.g., initial step), use a zero tensor or small random noise
            prediction_error_signal = torch.zeros(self.config.d_model, device=DEVICE, dtype=DTYPE)
            
        # 1. Modulate HH neurons based on prediction error
        # Project prediction error (d_model) to input currents for HH neurons
        input_current = self.input_current_projection(prediction_error_signal.detach()) # Detach to treat error as input, not to train projection via HH dynamics directly here
        
        # Simulate HH dynamics for one step (or a few internal steps)
        # TODO: Determine how many internal HH steps per single PCA-CR forward pass. For now, one.
        hh_spikes, hh_voltages = self._hodgkin_huxley_step(input_current)
        mean_firing_rate = torch.mean(hh_spikes)

        # 2. Calculate Homeostatic Errors
        current_homeostatic_values = {
            "prediction_error_level": torch.norm(prediction_error_signal) / self.config.d_model, # Normalized
            "phi_level": torch.tensor(iiwu_metrics.get("phi_value", self.config.ishu_homeostatic_variables["phi_level"]["target"]) if iiwu_metrics else self.config.ishu_homeostatic_variables["phi_level"]["target"], device=DEVICE, dtype=DTYPE),
            "internal_coherence": torch.tensor(iiwu_metrics.get("network_coherence", self.config.ishu_homeostatic_variables["internal_coherence"]["target"]) if iiwu_metrics else self.config.ishu_homeostatic_variables["internal_coherence"]["target"], device=DEVICE, dtype=DTYPE)
        }
        
        homeostatic_errors_dict = {}
        homeostatic_errors_tensor = []
        for var_name, target_val in self.homeostatic_setpoints.items():
            current_val = current_homeostatic_values[var_name]
            error = (current_val - target_val) * self.homeostatic_weights[var_name]
            homeostatic_errors_dict[var_name] = error.item()
            homeostatic_errors_tensor.append(error)
        
        homeostatic_errors_vec = torch.stack(homeostatic_errors_tensor) # Tensor of errors

        # 3. Update Affective State (Valence, Arousal)
        self._update_affective_state(homeostatic_errors_vec.detach(), mean_firing_rate.detach())
        
        logger.info(f"{self.module_name} updated internal and affective states.")
        return {
            "hh_spikes": hh_spikes, # [num_neurons]
            "hh_voltages": hh_voltages, # [num_neurons]
            "mean_firing_rate": mean_firing_rate.item(),
            "homeostatic_errors": homeostatic_errors_dict, # Dict of scalar errors
            "overall_homeostatic_error_magnitude": torch.norm(homeostatic_errors_vec).item(),
            "affective_state": self.current_affective_state # [valence, arousal]
        }

class InformationIntegrationWorkspaceUnit(PCACRModule):
    """
    Integrates information from various sources, calculates IIT's Phi (approximation),
    manages network connectivity, and implements GWT-like global broadcast.
    The underlying network could be the ISHU's HH neurons or a dedicated network.
    Here, we assume it operates on the state of ISHU's HH neurons.
    """
    def __init__(self, config: PCACRConfig):
        super().__init__(config)
        self.network_size = config.iiwu_network_size # Typically num_hh_neurons
        
        # Connectivity Matrix: W_ij is strength from j to i
        # Initialized as sparse random. Should be learnable/plastic.
        # Using a dense parameter for simplicity, but sparse is better for large N.
        self.connectivity_matrix = nn.Parameter(
            torch.randn(self.network_size, self.network_size, device=DEVICE, dtype=DTYPE) * 0.1
        )
        # Ensure no self-connections initially for some plasticity rules
        self.connectivity_matrix.data.fill_diagonal_(0)

        self.phi_calculator = PhiCalculator(config, self.network_size)
        self.current_phi = 0.0
        self.current_network_coherence = 0.0 # Example metric

        # Global Workspace (GWT-inspired)
        self.global_workspace = deque(maxlen=config.iiwu_global_workspace_capacity)
        self.workspace_item_salience = deque(maxlen=config.iiwu_global_workspace_capacity)

        logger.info(f"{self.module_name} initialized for a network of size {self.network_size}.")

    def _update_connectivity(self, neural_activity: torch.Tensor, prediction_error_magnitude: Optional[float] = None):
        """
        Updates the connectivity matrix based on neural activity and plasticity rules.
        Args:
            neural_activity (torch.Tensor): Current activity (e.g., spikes or voltages) of neurons [network_size].
            prediction_error_magnitude (float, optional): Overall system prediction error, can modulate plasticity.
        """
        # TODO: Implement a chosen plasticity rule rigorously.
        # Example: Simple Hebbian-like update (uncapped, needs normalization/decay)
        if self.config.iiwu_connectivity_update_rule == "hebbian":
            # Activity is [N], need outer product for [N, N]
            activity_i = neural_activity.unsqueeze(1) # [N, 1]
            activity_j = neural_activity.unsqueeze(0) # [1, N]
            delta_W = self.config.iiwu_plasticity_rate * (activity_i @ activity_j)
            # Add decay or normalization
            self.connectivity_matrix.data += delta_W
            self.connectivity_matrix.data -= self.config.iiwu_plasticity_rate * 0.01 * self.connectivity_matrix.data # Simple decay
            self.connectivity_matrix.data.fill_diagonal_(0) # No self-connections
            self.connectivity_matrix.data.clamp_(-1.0, 1.0) # Bound weights
            logger.debug("IIWU: Updated connectivity with Hebbian-like rule.")

        elif self.config.iiwu_connectivity_update_rule == "structural_fep":
        # TODO: More advanced rule, e.g., prune connections that increase free energy / prediction error,
        # or strengthen those that reduce it. This requires more complex computation.
            logger.warning("IIWU: 'structural_fep' plasticity not fully implemented.")
            pass
        else:
            logger.warning(f"IIWU: Unknown connectivity update rule: {self.config.iiwu_connectivity_update_rule}")
        
    def _calculate_network_coherence(self, neural_activity: torch.Tensor) -> float:
        """
        Calculates a measure of network coherence (e.g., mean phase coherence if using oscillators,
        or mean correlation of activity for simpler neurons).
        Args:
            neural_activity (torch.Tensor): Current activity (e.g., voltages) of neurons [network_size].
        Returns:
            float: Coherence value.
        """
        if self.network_size <= 1: return 0.0
        # Simple mean correlation for demonstration
        # For more complex activity (e.g. phases), use appropriate phase coherence measures.
        if neural_activity.ndim == 1:
            neural_activity_reshaped = neural_activity.unsqueeze(0) # Make it (1, N) for corrcoef
        else:
            neural_activity_reshaped = neural_activity

        if neural_activity_reshaped.shape[1] > 1: # Need at least 2 variables for correlation
            # Using .cpu().numpy() as torch.corrcoef can be memory intensive for large matrices or have issues on some devices
            # For very large networks, sampling pairs of neurons might be necessary.
            try:
                corr_matrix = torch.corrcoef(neural_activity_reshaped.cpu()) 
                # Mean of absolute off-diagonal elements
                mask = ~torch.eye(corr_matrix.shape[0], dtype=torch.bool)
                if torch.any(mask):
                    coherence = torch.mean(torch.abs(corr_matrix[mask])).item()
                else:
                    coherence = 0.0 # if only one neuron effectively
            except RuntimeError as e:
                logger.warning(f"IIWU: Could not compute correlation matrix for coherence: {e}")
                coherence = 0.0 # Default on error
        else:
            coherence = 0.0
            
        self.current_network_coherence = coherence
        return coherence

    def _update_global_workspace(self, information_sources: List[Tuple[torch.Tensor, float]]):

        # Add new salient items
        for info_tensor, salience in information_sources:
            if salience > self.config.iiwu_broadcast_threshold:
                # If workspace is full, remove least salient item if new item is more salient

                    min_salience_existing = min(self.workspace_item_salience) if self.workspace_item_salience else -1
                    if salience > min_salience_existing:
                        idx_to_remove = self.workspace_item_salience.index(min_salience_existing)
                        self.global_workspace.remove(self.global_workspace[idx_to_remove]) # This is inefficient for deque
                        self.workspace_item_salience.remove(min_salience_existing)
                        # Re-implement with proper deque removal or manage indices
                        # For simplicity of deque:
                        # self.global_workspace.popleft()
                        # self.workspace_item_salience.popleft()
                        # self.global_workspace.append(info_tensor.clone().detach())
                        # self.workspace_item_salience.append(salience)
                        # Correct way: Find and remove
                        # TODO: Efficiently manage replacement based on salience.
                        # For now, just adding if space or if more salient than oldest.
                        self.global_workspace.append(info_tensor.clone().detach())
                        self.workspace_item_salience.append(salience)

                elif len(self.global_workspace) < self.config.iiwu_global_workspace_capacity :
                    self.global_workspace.append(info_tensor.clone().detach())
                    self.workspace_item_salience.append(salience)
        
        # Sort by salience for easier access (optional, GWT implies availability not order)

        # self.global_workspace = deque([item[0] for item in sorted_workspace], maxlen=self.config.iiwu_global_workspace_capacity)

    def get_workspace_contents(self) -> List[torch.Tensor]:
        return list(self.global_workspace)

    def forward(self, ishu_outputs: Dict[str, Any], ppu_outputs: Dict[str, Any], spu_outputs: Dict[str, Any], current_time_step: int) -> Dict[str, Any]:
        """
        Main forward pass for IIWU.
        Args:
            ishu_outputs (Dict): Outputs from ISHU (e.g., 'hh_spikes', 'hh_voltages', 'affective_state').
            ppu_outputs (Dict): Outputs from PPU (e.g., 'current_prediction_error').
            spu_outputs (Dict): Outputs from SPU (e.g., 'semantic_features', 'acoustic_features').
            current_time_step (int): Current simulation time step.
        Returns:
            A dictionary containing 'phi_value', 'network_coherence', 'connectivity_matrix', 'workspace_contents'.
        """
        # Using HH spikes as the relevant neural activity for plasticity and Phi. Could be voltages.
        neural_activity = ishu_outputs.get("hh_spikes", torch.zeros(self.network_size, device=DEVICE, dtype=DTYPE))
        
        # 1. Update Network Connectivity (Plasticity)
        # TODO: Pass appropriate prediction error magnitude if rule requires it
        pred_error_mag = torch.norm(ppu_outputs.get("current_prediction_error", torch.tensor(0.0))).item()
        self._update_connectivity(neural_activity.detach(), pred_error_mag) # Detach activity to avoid direct backprop through this update

        # 2. Calculate Network Coherence
        # Using HH voltages for coherence as spikes are binary
        hh_voltages = ishu_outputs.get("hh_voltages", torch.zeros(self.network_size, device=DEVICE, dtype=DTYPE))
        self._calculate_network_coherence(hh_voltages)

        # 3. Calculate Phi (Integrated Information)
        # This is computationally expensive. Do it periodically.
        if current_time_step % self.config.iiwu_phi_calculation_interval == 0:
            # TODO: Ensure `neural_activity_history` is properly passed or managed by PhiCalculator
            # This requires a sequence of states. For now, using current and a dummy past.
            # Proper implementation needs `PhiCalculator` to maintain or receive history.
            # Here, `neural_activity` is current spikes. `past_neural_activity` would be spikes from t-1.
            # This needs to be passed from main loop or ISHU.
            past_neural_activity = getattr(self, "_past_neural_activity_for_phi", torch.zeros_like(neural_activity))
            self.current_phi = self.phi_calculator.calculate_phi_approx(
                current_state_discrete=neural_activity.bool(), # Assuming Phi on discrete states (spikes)
                past_state_discrete=past_neural_activity.bool(), 
                connectivity_matrix=self.connectivity_matrix.data.detach()
            )
            setattr(self, "_past_neural_activity_for_phi", neural_activity.clone().detach())
            logger.info(f"IIWU: Calculated Phi = {self.current_phi:.4f} at step {current_time_step}")


        # 4. Update Global Workspace (GWT-inspired)
        # Collect potential information sources and their salience
        # Salience can be defined in various ways (e.g., magnitude of error, intensity of affect)
        information_sources = []
        ppu_error = ppu_outputs.get("current_prediction_error")
        if ppu_error is not None:
            # Salience based on error magnitude
            salience_error = torch.norm(ppu_error).item() / (self.config.d_model**0.5) # Normalize roughly
            information_sources.append((ppu_error, salience_error))

        affective_state = ishu_outputs.get("affective_state") # [valence, arousal]
        if affective_state is not None:
            # Salience based on arousal or combined valence/arousal magnitude
            salience_affect = torch.norm(affective_state).item()
            information_sources.append((affective_state, salience_affect))
        
        # Add other potential sources, e.g., highly active neural patterns, strong sensory signals
        # For instance, if a particular sensory feature is very strong/novel
        if spu_outputs.get("semantic_features") is not None:
             # Salience could be novelty compared to PPU's prediction, or just norm
            salience_sem = torch.norm(spu_outputs["semantic_features"]).item() / (self.config.d_model**0.5)
            information_sources.append((spu_outputs["semantic_features"], salience_sem))

        self._update_global_workspace(information_sources)
        
        logger.info(f"{self.module_name} processed information. Coherence: {self.current_network_coherence:.3f}")
        return {
            "phi_value": self.current_phi,
            "network_coherence": self.current_network_coherence,
            "connectivity_matrix": self.connectivity_matrix.data, # For analysis/evolution
            "workspace_contents": self.get_workspace_contents() # List of tensors
        }

    def _shannon_entropy(self, probs: torch.Tensor) -> torch.Tensor:
        """ H(X) = - sum(p(x) log2 p(x)) """
        probs = probs[probs > 0] # Avoid log(0)
        return -torch.sum(probs * torch.log2(probs + 1e-9)) # Add epsilon for stability

    def _mutual_information(self, p_xy: torch.Tensor, p_x: torch.Tensor, p_y: torch.Tensor) -> torch.Tensor:
        """ I(X;Y) = H(X) + H(Y) - H(X,Y) """
        # Or, I(X;Y) = sum_{x,y} p(x,y) log2( p(x,y) / (p(x)p(y)) )
        # Assuming p_xy is joint, p_x and p_y are marginals.
        # This requires knowing the actual distributions.
        # TODO: Implement based on actual state probability distributions from TPM or data.
        # Placeholder for conceptual demonstration:
        if p_xy.ndim == 2 : # if p_xy is a joint probability matrix P(X,Y)
            # p_x = p_xy.sum(dim=1)
            # p_y = p_xy.sum(dim=0)
            mi = 0.0
            for i in range(p_xy.shape[0]):
                for j in range(p_xy.shape[1]):
                    if p_xy[i,j] > 1e-9 and p_x[i] > 1e-9 and p_y[j] > 1e-9:
                         mi += p_xy[i,j] * torch.log2(p_xy[i,j] / (p_x[i] * p_y[j] + 1e-9) + 1e-9)
            return mi
        return torch.tensor(0.0, device=DEVICE, dtype=DTYPE) # Highly simplified placeholder

            if len(past_subset_state) > 1 and len(current_subset_state) > 1:
                # This is still not Phi, just a statistical dependency measure.
                # Consider states as sequences of 0s and 1s.

class ActionSelectionGenerationUnit(PCACRModule):

        return action_index, action_probs

        # Project input features to match embedding requirements (num_qubits)
        quantum_input = self.input_projection_qpc(input_features)
        quantum_input = torch.tanh(quantum_input) * np.pi # Scale for angle embedding

        try:
            # Execute the quantum circuit
            quantum_output_raw = self.quantum_layer(quantum_input) # Shape [num_qubits]
            
            # Project quantum output back to d_model dimension
            classical_output = self.output_projection_qpc(quantum_output_raw)
            
            logger.info(f"{self.module_name} processed features.")
            return classical_output
        except Exception as e:
            logger.error(f"QPC: Error during quantum computation: {e}")
            # Fallback: return a zero tensor or a processed version of input_features classically
            return torch.zeros(self.config.d_model, device=DEVICE, dtype=DTYPE)


# ████████████████████████████████████████████████████████████████████████████████████████████████████
# ██ Main PCA-CR Model: Orchestrator                                                              ██
# ████████████████████████████████████████████████████████████████████████████████████████████████████

class PCA_CR_Model(nn.Module):
    """
    The main orchestrator for the Principled Cognitive Architecture.
    Connects all modules and manages the overall data flow and learning.
    """
    def __init__(self, config: PCACRConfig):
        super().__init__()
        self.config = config
        self.config.log_config() # Log the configuration used

        self.spu = SensoryProcessingUnit(config)
        self.ppu = PredictiveProcessingUnit(config)
        self.ishu = InternalStateHomeostasisUnit(config)
        self.iiwu = InformationIntegrationWorkspaceUnit(config)
        self.asgu = ActionSelectionGenerationUnit(config)
        
        # Collect all optimizable parameters for the LAU
        all_params = list(self.spu.parameters()) + \
                     list(self.ppu.parameters()) + \
                     list(self.ishu.parameters()) + \
                     list(self.iiwu.parameters()) + \
                     list(self.asgu.parameters())

        if config.qpc_enabled:
            self.qpc = QuantumProcessingCore(config)
            if self.qpc.quantum_layer is not None: # Check if QPC initialized successfully
                 all_params += list(self.qpc.parameters())
        else:
            self.qpc = None
            
        self.lau = LearningAdaptationUnit(config, all_params)

        self.global_step = 0
        self.current_sensory_features_t = None # Store S_t to be S_t-1 in the next step for PPU
        logger.info(f"PCA-CR Model initialized. Total optimizable parameters: {sum(p.numel() for p in all_params if p.requires_grad)}")

    def get_modules_dict(self) -> Dict[str, PCACRModule]:
        """ Helper to pass modules to LAU for complex operations like memory replay. """
        modules = {
            "SPU": self.spu, "PPU": self.ppu, "ISHU": self.ishu,
            "IIWU": self.iiwu, "ASGU": self.asgu
        }
        if self.qpc: modules["QPC"] = self.qpc
        return modules

    def forward(self, audio_input: Optional[torch.Tensor] = None, text_input: Optional[str] = None,
                target_next_sensory_features: Optional[torch.Tensor] = None, # For supervised PPU training
                is_training: bool = True) -> Dict[str, Any]:
        """
        A full step of perception, internal processing, action, and learning.
        Args:
            audio_input, text_input: Raw inputs for current time step T.
            target_next_sensory_features: Ground truth for sensory state at T+1 (if available, for training PPU).
            is_training: Flag to indicate if the model is in training mode (for LAU).
        Returns:
            Dictionary containing outputs from ASGU and key metrics.
        """
        logger.info(f"--- PCA-CR Step {self.global_step} ---")

        # 1. Sensory Processing (SPU) - Processes inputs for time T
        spu_data_t = self.spu(audio_input, text_input)
        
        # Combine SPU features (e.g., average or concatenate if both present, or use one if only one present)
        # This combined feature represents the "actual current sensory state S_t"
        s_t_actual_features_list = []
        if spu_data_t["semantic_features"] is not None: s_t_actual_features_list.append(spu_data_t["semantic_features"])
        if spu_data_t["acoustic_features"] is not None: s_t_actual_features_list.append(spu_data_t["acoustic_features"])
        
        if not s_t_actual_features_list: # No sensory input
            s_t_actual_combined = torch.zeros(self.config.d_model, device=DEVICE, dtype=DTYPE)
        elif len(s_t_actual_features_list) == 1:
            s_t_actual_combined = s_t_actual_features_list[0]
        else: # Both present, combine (e.g., average)
            s_t_actual_combined = torch.mean(torch.stack(s_t_actual_features_list), dim=0)
        
        # 2. Predictive Processing (PPU)
        # PPU needs S_t-1 (stored as self.current_sensory_features_t from *previous* step) to predict S_t.
        # Then, it compares this prediction of S_t with s_t_actual_combined to calculate error at T.
        # It also generates a prediction for S_t+1 using s_t_actual_combined.
        
        if self.current_sensory_features_t is None: # First step
            self.current_sensory_features_t = torch.zeros_like(s_t_actual_combined) # Initialize S_t-1
            self.ppu.initialize_hidden_state() # Initialize PPU hidden state

        # PPU predicts S_t based on S_t-1, then calculates error S_t_actual - S_t_predicted
        # And then PPU uses S_t_actual to predict S_t+1
        ppu_prediction_of_s_t = self.ppu.generate_prediction(self.current_sensory_features_t)
        s_t_prediction_error = self.ppu.calculate_prediction_error(s_t_actual_combined, ppu_prediction_of_s_t)
        
        # PPU then generates prediction for S_t+1 using actual S_t for next cycle's comparison
        ppu_prediction_of_s_t_plus_1 = self.ppu.generate_prediction(s_t_actual_combined.detach()) # Detach to ensure S_t is treated as fixed input for predicting S_t+1

        ppu_outputs = {
            "predicted_current_sensory_features": ppu_prediction_of_s_t, # Prediction of S_t
            "current_prediction_error": s_t_prediction_error,           # Error at S_t
            "predicted_next_sensory_features": ppu_prediction_of_s_t_plus_1 # Prediction of S_t+1
        }
        
        # Update for next cycle: current S_t becomes S_t-1
        self.current_sensory_features_t = s_t_actual_combined.detach()


        # 3. Internal State & Homeostasis (ISHU)
        # ISHU can be influenced by prediction error from PPU
        ishu_outputs = self.ishu(ppu_outputs)

        # 4. Information Integration & Workspace (IIWU)
        # IIWU integrates info from ISHU, PPU, SPU and calculates Phi, coherence, updates GWT
        iiwu_outputs = self.iiwu(ishu_outputs, ppu_outputs, spu_data_t, self.global_step)

        # (Optional) 5. Quantum Processing Core (QPC)
        qpc_output_features = None
        if self.qpc and self.config.qpc_enabled:
            # QPC might process some integrated information from IIWU or affect from ISHU
            qpc_input = iiwu_outputs.get("workspace_contents") # Example: use workspace content
            if qpc_input and isinstance(qpc_input[0], torch.Tensor): # Take first item for simplicity
                qpc_output_features = self.qpc(qpc_input[0]) 
            else: # Or some other salient feature
                qpc_output_features = self.qpc(s_t_actual_combined) # Fallback

        # 6. Action Selection & Generation (ASGU)
        # ASGU uses IIWU workspace, ISHU affect, and potentially QPC output to decide/generate action
        # The context for generation can be from SPU or IIWU
        # For now, ASGU primarily uses IIWU workspace and ISHU affect for action *selection*,
        # and SPU data (or IIWU context) for response *generation*.
        # If QPC provides features, they could modulate the state for ASGU.
        asgu_input_state_from_iiwu = iiwu_outputs["workspace_contents"][0] if iiwu_outputs["workspace_contents"] else torch.zeros(self.config.d_model, device=DEVICE)
        if qpc_output_features is not None: # Integrate QPC output if available
             asgu_input_state_from_iiwu = (asgu_input_state_from_iiwu + qpc_output_features) / 2.0 # Example combination

        asgu_outputs = self.asgu(iiwu_outputs, ishu_outputs, spu_data_t) # Pass spu_data_t for generation context

        # 7. Learning & Adaptation (LAU)
        lau_metrics = {}
        if is_training:
            # PPU's parameters will be updated based on its error in predicting S_t
            # We need the actual S_t (s_t_actual_combined) and PPU's prediction of S_t (ppu_prediction_of_s_t)
            # The PPU output already contains `current_prediction_error = s_t_actual - S_t_predicted`
            lau_metrics = self.lau(self, ppu_outputs, ishu_outputs, iiwu_outputs, spu_data_t, self.global_step)
            # Perform optimizer step after all gradients for the step are accumulated
            self.lau.optimizer.step()
            self.lau.optimizer.zero_grad()


        self.global_step += 1
        
        # Prepare final output for this step
        final_outputs = {
            "input_text": spu_data_t["original_text"] or spu_data_t["transcribed_text"],
            "response_text": asgu_outputs["generated_text"],
            "response_audio_path": asgu_outputs["generated_audio_path"],
            "metrics": {
                "phi_value": iiwu_outputs["phi_value"],
                "network_coherence": iiwu_outputs["network_coherence"],
                "affective_state": ishu_outputs["affective_state"].tolist(), # [valence, arousal]
                "mean_firing_rate": ishu_outputs["mean_firing_rate"],
                "prediction_error_norm": torch.norm(ppu_outputs["current_prediction_error"]).item() if ppu_outputs["current_prediction_error"] is not None else 0,
                "homeostatic_errors": ishu_outputs["homeostatic_errors"],
                "qpc_active": qpc_output_features is not None
            },
            "internal_states_for_analysis": { # For deeper analysis / debugging
                "spu_features": {k: v.shape if isinstance(v, torch.Tensor) else type(v) for k,v in spu_data_t.items()},
                "ppu_prediction_next": ppu_outputs["predicted_next_sensory_features"].shape,
                "ishu_voltages_sample": ishu_outputs["hh_voltages"][:5].tolist(), # Sample
                "iiwu_workspace_size": len(iiwu_outputs["workspace_contents"]),
            }
        }
        if "fep_loss" in lau_metrics: final_outputs["metrics"]["fep_loss"] = lau_metrics["fep_loss"]

        return final_outputs

    def save_full_model(self, epoch: int):
        path = self.config.output_dir / f"pca_cr_model_epoch_{epoch}.pt"
        torch.save(self.state_dict(), path)
        logger.info(f"Saved full PCA-CR model checkpoint to {path}")

    def load_full_model(self, path_str: str):
        path = Path(path_str)
        if path.exists():
            self.load_state_dict(torch.load(path, map_location=DEVICE))
            logger.info(f"Loaded full PCA-CR model from {path}")
        else:
            logger.error(f"Full PCA-CR model checkpoint not found at {path}")
