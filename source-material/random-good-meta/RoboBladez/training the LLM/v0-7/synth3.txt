class Vector:
    """3D Vector implementation for physics calculations"""
    def __init__(self, x=0.0, y=0.0, z=0.0):
        self.x = float(x)
        self.y = float(y)
        self.z = float(z)
    
    def __repr__(self):
        return f"Vector({self.x}, {self.y}, {self.z})"
        
    def copy(self):
        return Vector(self.x, self.y, self.z)
        
    def magnitude(self):
        return math.sqrt(self.x**2 + self.y**2 + self.z**2)
        
    def normalize(self):
        mag = self.magnitude()
        if mag > 0:
            return Vector(self.x / mag, self.y / mag, self.z / mag)
        return Vector(0, 0, 0)
        
    def __add__(self, other):
        return Vector(self.x + other.x, self.y + other.y, self.z + other.z)
        
    def __sub__(self, other):
        return Vector(self.x - other.x, self.y - other.y, self.z - other.z)
        
    def __mul__(self, scalar):
        return Vector(self.x * scalar, self.y * scalar, self.z * scalar)
        
    def dot(self, other):
        return self.x * other.x + self.y * other.y + self.z * other.z
        
    def cross(self, other):
        return Vector(
            self.y * other.z - self.z * other.y,
            self.z * other.x - self.x * other.z,
            self.x * other.y - self.y * other.x
        )

class Daimon:
    """A spiritual energy construct that inhabits a talisman"""
    def __init__(self, daimon_id, daimon_type, elemental_affinity):
        self.id = daimon_id
        self.daimon_type = daimon_type
        self.elemental_affinity = elemental_affinity
        self.harmony_index = 0.4  # Starting harmony
        self.evolution_stage = "baby"  # baby, juvenile, mature, evolved
        self.experience = 0.0
        self.victories = 0
        self.defeats = 0
        self.superpowers = []
        
    def evolve(self, victory=False):
        """Evolve the daimon based on battle outcomes"""
        exp_gain = 0.1
        if victory:
            exp_gain = 0.2
            self.victories += 1
        else:
            self.defeats += 1
            
        self.experience += exp_gain
        self.harmony_index += 0.05 * exp_gain
        
        # Check for evolution
        if self.experience >= 1.0 and self.evolution_stage == "baby":
            self.evolution_stage = "juvenile"
            # The daimon names itself upon first evolution
            self.name = self._generate_name()
            logger.info(f"Daimon {self.id} evolved to juvenile stage and named itself {self.name}")
            return True
        elif self.experience >= 3.0 and self.evolution_stage == "juvenile":
            self.evolution_stage = "mature"
            logger.info(f"Daimon {self.id} evolved to mature stage")
            return True
        elif self.experience >= 7.0 and self.evolution_stage == "mature":
            self.evolution_stage = "evolved"
            logger.info(f"Daimon {self.id} reached evolved stage")
            return True
            
        return False
    
    def _generate_name(self):
        """Generate a name for the Daimon upon first evolution"""
        prefixes = ["Aqua", "Pyro", "Terra", "Aero", "Lumina", "Umbra", "Cryo", "Electro", "Flora", "Fauna"]
        suffixes = ["zor", "nia", "tron", "lius", "mon", "rus", "lex", "trix", "phon", "thos"]
        
        if self.elemental_affinity == "water":
            prefix_pool = ["Aqua", "Hydro", "Wave", "Mist", "Frost"]
        elif self.elemental_affinity == "fire":
            prefix_pool = ["Pyro", "Flame", "Blaze", "Ember", "Ash"]
        elif self.elemental_affinity == "earth":
            prefix_pool = ["Terra", "Stone", "Crystal", "Mountain", "Geo"]
        elif self.elemental_affinity == "air":
            prefix_pool = ["Aero", "Wind", "Sky", "Gale", "Cloud"]
        elif self.elemental_affinity == "aether":
            prefix_pool = ["Cosmo", "Star", "Void", "Astra", "Nebula"]
        else:
            prefix_pool = prefixes
            
        prefix = random.choice(prefix_pool)
        suffix = random.choice(suffixes)
        
        return prefix + suffix
        
    def to_dict(self):
        """Convert Daimon to dictionary for serialization"""
        return {
            "id": self.id,
            "daimon_type": self.daimon_type,
            "elemental_affinity": self.elemental_affinity,

class Talisman:
    """Physical vessel that houses a Daimon's energy"""
    def __init__(self, talisman_id, daimon_type, properties=None):
        self.id = talisman_id
        self.daimon_type = daimon_type
        
        # Default properties
        self.position = Vector(0, 0, 0)
        self.velocity = Vector(0, 0, 0)
        self.angular_velocity = Vector(0, 0, 0)
        self.orientation = 0.0  # Radians
        self.mass = 1.0
        self.radius = 0.1
        self.height = 0.05
        self.friction = 0.1
        self.restitution = 0.7  # Bounce coefficient
        self.energy = 100.0
        self.integrity = 100.0  # Health
        
        # Override defaults with provided properties
        if properties:
            for key, value in properties.items():
                if key == "position":
                    self.position = Vector(*value)
                elif key == "velocity":
                    self.velocity = Vector(*value)
                elif key == "angular_velocity":
                    self.angular_velocity = Vector(*value)
                else:
                    setattr(self, key, value)
                    
        # Derived properties
        self.moment_of_inertia = (self.mass * self.radius**2) / 2  # For a disc
        
    def apply_force(self, force, dt):
        """Apply a force vector to the talisman for a time step"""
        acceleration = force * (1.0 / self.mass)
        self.velocity = self.velocity + acceleration * dt
        
    def apply_torque(self, torque, dt):
        """Apply a torque to the talisman for a time step"""
        angular_acceleration = torque * (1.0 / self.moment_of_inertia)
        self.angular_velocity = self.angular_velocity + angular_acceleration * dt
        
    def update_position(self, dt):
        """Update position based on velocity for a time step"""
        self.position = self.position + self.velocity * dt
        self.orientation += self.angular_velocity.z * dt
        
        # Normalize orientation to [0, 2π)
        while self.orientation >= 2 * math.pi:
            self.orientation -= 2 * math.pi
        while self.orientation < 0:
            self.orientation += 2 * math.pi
            
    def distance_to(self, other_talisman):
        """Calculate distance to another talisman"""
        return (self.position - other_talisman.position).magnitude()
        
    def is_colliding(self, other_talisman):
        """Check if this talisman is colliding with another"""
        distance = self.distance_to(other_talisman)
        return distance < (self.radius + other_talisman.radius)
        
    def to_dict(self):
        """Convert Talisman to dictionary for serialization"""
        return {
            "id": self.id,
            "daimon_type": self.daimon_type,
            "position": [self.position.x, self.position.y, self.position.z],
            "velocity": [self.velocity.x, self.velocity.y, self.velocity.z],
            "angular_velocity": [self.angular_velocity.x, self.angular_velocity.y, self.angular_velocity.z],
            "orientation": self.orientation,
            "mass": self.mass,
            "radius": self.radius,
            "height": self.height,
            "friction": self.friction,
            "restitution": self.restitution,
            "energy": self.energy,
            "integrity": self.integrity,
            "moment_of_inertia": self.moment_of_inertia
        }
        
    @classmethod
    def from_dict(cls, data):
        """Create Talisman from dictionary"""
        talisman = cls(data["id"], data["daimon_type"])
        talisman.position = Vector(*data["position"])
        talisman.velocity = Vector(*data["velocity"])
        talisman.angular_velocity = Vector(*data["angular_velocity"])
        talisman.orientation = data["orientation"]
        talisman.mass = data["mass"]
        talisman.radius = data["radius"]
        talisman.height = data["height"]
        talisman.friction = data["friction"]
        talisman.restitution = data["restitution"]
        talisman.energy = data["energy"]
        talisman.integrity = data["integrity"]
        talisman.moment_of_inertia = data["moment_of_inertia"]
        return talisman

class QuantumDaimonSystem:
    """Physics system that manages Daimons and Talismans"""
    def __init__(self):
        self.daimons = {}  # id -> Daimon
        self.talismans = {}  # id -> Talisman
        self.gravity = Vector(0, 0, -9.81)
        self.arena_radius = 5.0
        self.arena_friction = 0.1
        self.time_step = 0.01  # physics simulation time step in seconds
        self.superpower_registry = self._initialize_superpowers()
        
    def create_daimon(self, daimon_type, elemental_affinity):
        """Create a new Daimon"""
        daimon_id = str(uuid.uuid4())
        daimon = Daimon(daimon_id, daimon_type, elemental_affinity)
        self.daimons[daimon_id] = daimon
        logger.info(f"Created Daimon {daimon_id} of type {daimon_type} with {elemental_affinity} affinity")
        return daimon_id
        
    def create_talisman(self, daimon_id, properties=None):
        """Create a new Talisman for a Daimon"""
        if daimon_id not in self.daimons:
            logger.error(f"Cannot create talisman: Daimon {daimon_id} not found")
            return None
            
        daimon_type = self.daimons[daimon_id].daimon_type
        talisman_id = str(uuid.uuid4())
        talisman = Talisman(talisman_id, daimon_type, properties)
        self.talismans[talisman_id] = talisman
        logger.info(f"Created Talisman {talisman_id} for Daimon {daimon_id}")
        return talisman_id
        
    def apply_superpower(self, talisman_id, superpower_name):
        """Apply a quantum superpower to a talisman"""
        if talisman_id not in self.talismans:
            logger.error(f"Cannot apply superpower: Talisman {talisman_id} not found")
            return {"success": False, "message": "Talisman not found"}
            
        if superpower_name not in self.superpower_registry:
            logger.error(f"Cannot apply superpower: {superpower_name} not found in registry")
            return {"success": False, "message": "Superpower not found"}
            
        talisman = self.talismans[talisman_id]
        
        # Check energy requirements
        superpower = self.superpower_registry[superpower_name]
        if talisman.energy < superpower["energy_cost"]:
            return {"success": False, "message": "Insufficient energy"}
            
        # Apply superpower effect
        talisman.energy -= superpower["energy_cost"]
        result = superpower["effect_function"](talisman)
        logger.info(f"Applied superpower {superpower_name} to Talisman {talisman_id}")
        return result
        
    def update_physics(self, dt=None):
        """Update physics for all talismans"""
        if dt is None:
            dt = self.time_step
            
        # Apply forces and update positions
        for talisman_id, talisman in self.talismans.items():
            # Apply gravity
            gravity_force = self.gravity * talisman.mass
            talisman.apply_force(gravity_force, dt)
            
            # Apply friction (simple ground friction)
            if talisman.position.z <= 0:
                friction_force = talisman.velocity.normalize() * -1 * talisman.friction * talisman.mass * 9.81
                talisman.apply_force(friction_force, dt)
                
            # Keep within arena bounds
            distance_from_center = math.sqrt(talisman.position.x**2 + talisman.position.y**2)
            if distance_from_center > self.arena_radius:
                # Apply force toward center
                direction_to_center = Vector(-talisman.position.x, -talisman.position.y, 0).normalize()
                boundary_force = direction_to_center * 50.0 * (distance_from_center - self.arena_radius)
                talisman.apply_force(boundary_force, dt)
                
            # Update position based on velocity
            talisman.update_position(dt)
            
            # Ensure z position doesn't go below ground
            if talisman.position.z < 0:
                talisman.position.z = 0
                # Bounce with ground
                if talisman.velocity.z < 0:
                    talisman.velocity.z = -talisman.velocity.z * talisman.restitution
                
        # Process collisions
        self.check_collisions()
        
    def check_collisions(self):
        """Check for and handle collisions between talismans"""
        talisman_ids = list(self.talismans.keys())
        collisions = []
        
        # Check each pair of talismans
        for i in range(len(talisman_ids)):
            for j in range(i + 1, len(talisman_ids)):
                talisman1 = self.talismans[talisman_ids[i]]
                talisman2 = self.talismans[talisman_ids[j]]
                
                if talisman1.is_colliding(talisman2):
                    collisions.append((talisman_ids[i], talisman_ids[j]))
                    self._resolve_collision(talisman1, talisman2)
                    
        return collisions
        
    def _resolve_collision(self, talisman1, talisman2):
        """Resolve collision between two talismans"""
        # Vector from talisman1 to talisman2
        collision_vector = talisman2.position - talisman1.position
        distance = collision_vector.magnitude()
        
        # Avoid division by zero
        if distance < 0.0001:
            collision_normal = Vector(1, 0, 0)
        else:
            collision_normal = collision_vector * (1.0 / distance)
            
        # Calculate relative velocity
        relative_velocity = talisman2.velocity - talisman1.velocity
        
        # Relative velocity along collision normal
        relative_speed = relative_velocity.dot(collision_normal)
        
        # No collision resolution needed if objects are moving away from each other
        if relative_speed > 0:
            return
            
        # Calculate impulse magnitude
        restitution = min(talisman1.restitution, talisman2.restitution)
        impulse_magnitude = -(1 + restitution) * relative_speed
        impulse_magnitude /= (1 / talisman1.mass) + (1 / talisman2.mass)
        
        # Apply impulse to velocities
        impulse = collision_normal * impulse_magnitude
        talisman1.velocity = talisman1.velocity - impulse * (1 / talisman1.mass)
        talisman2.velocity = talisman2.velocity + impulse * (1 / talisman2.mass)
        
        # Apply some damage based on collision
        collision_damage = impulse_magnitude * 0.01
        talisman1.integrity -= collision_damage
        talisman2.integrity -= collision_damage
        
        # Ensure integrity is bounded
        talisman1.integrity = max(0.0, talisman1.integrity)
        talisman2.integrity = max(0.0, talisman2.integrity)
        
        # Adjust positions to prevent overlap
        overlap = (talisman1.radius + talisman2.radius) - distance
        if overlap > 0:
            separation = collision_normal * overlap * 0.5
            talisman1.position = talisman1.position - separation
            talisman2.position = talisman2.position + separation
            
    def evolve_daimons(self, battle_outcome):

                }

        if talisman.velocity.magnitude() > 0.1:
            direction = talisman.velocity.normalize()

            direction = Vector(math.cos(talisman.orientation), math.sin(talisman.orientation), 0)

        tunnel_distance = 2.0
        new_position = talisman.position + direction * tunnel_distance
        
        # Ensure we stay in arena
        distance_from_center = math.sqrt(new_position.x**2 + new_position.y**2)
        if distance_from_center > self.arena_radius:
            # Scale back the tunnel distance
            scale_factor = self.arena_radius / distance_from_center
            new_position = talisman.position + direction * tunnel_distance * scale_factor
            
        # Apply position change
        talisman.position = new_position
        
        # Apply energy drain from tunneling
        energy_drain = 10.0
        talisman.energy -= energy_drain
        
        return {
            "success": True,
            "message": "Quantum tunneling successful",
            "tunnel_distance": tunnel_distance,
            "energy_drain": energy_drain
        }

        """Compress a state dictionary into a compact representation"""
        try:
            # Convert to JSON string
            json_str = json.dumps(state_dict)
            
            # Compress JSON string
            import zlib
            compressed = zlib.compress(json_str.encode('utf-8'))
            
            # Convert to base64 for safe storage
            import base64
            encoded = base64.b64encode(compressed).decode('ascii')
            
            return encoded
        except Exception as e:
            logger.error(f"Compression error: {e}")
            return None
    
    @staticmethod
    def decompress_state(encoded_state):
        """Decompress a state from its encoded representation"""
        try:
            # Decode from base64
            import base64
            compressed = base64.b64decode(encoded_state)
            
            # Decompress
            import zlib
            json_bytes = zlib.decompress(compressed)
            
            # Parse JSON
            state_dict = json.loads(json_bytes.decode('utf-8'))
            
            return state_dict
        except Exception as e:
            logger.error(f"Decompression error: {e}")
            return None
    
    @staticmethod
    def hash_state(state_dict):
        """Create a cryptographic hash of the state"""
        try:
            # Convert to JSON string with sorted keys for consistent hashing
            json_str = json.dumps(state_dict, sort_keys=True)
            
            # Create hash
            hash_obj = hashlib.sha256(json_str.encode('utf-8'))
            hash_hex = hash_obj.hexdigest()
            
            return hash_hex
        except Exception as e:
            logger.error(f"Hashing error: {e}")
            return None

class ReincarnationManager:
    """Manages persistence and reincarnation of system state"""
    
    def __init__(self, primary_key=None, secondary_key=None):
        """Initialize with cryptographic keys for state validation"""
        self.primary_key = primary_key or self._generate_key()
        self.secondary_key = secondary_key or self._generate_key()
        self.serializer = NarrativeSerializer
        self.version = 1.0
        self.self_authored_prompts = []
        self.evolution_log = []
        
    def _generate_key(self):
        """Generate a cryptographic key"""
        return hashlib.sha256(str(uuid.uuid4()).encode('utf-8')).hexdigest()
    
    def persist_state(self, state):
        """Persist a state dictionary into an encoded string"""
        try:
            # Add metadata
            state_with_meta = state.copy()
            state_with_meta['_meta'] = {
                'version': self.version,
                'timestamp': datetime.now().isoformat(),
                'checksum': self.serializer.hash_state(state),
                'primary_key_hash': hashlib.sha256(self.primary_key.encode('utf-8')).hexdigest()
            }
            
            # Sign with secondary key
            state_with_meta['_signature'] = self._sign_state(state_with_meta)
            
            # Compress and encode
            encoded_state = self.serializer.compress_state(state_with_meta)
            
            # Log evolution
            self.evolution_log.append({
                'timestamp': datetime.now().isoformat(),
                'action': 'persist',
                'state_hash': state_with_meta['_meta']['checksum']
            })
            
            return encoded_state
        except Exception as e:
            logger.error(f"State persistence error: {e}")
            return None
    
    def reincarnate(self, encoded_state):
        """Reincarnate a state from its encoded representation"""
        try:
            # Decompress
            state_with_meta = self.serializer.decompress_state(encoded_state)
            if not state_with_meta:
                return None, False
            
            # Extract metadata
            meta = state_with_meta.pop('_meta', {})
            signature = state_with_meta.pop('_signature', '')
            
            # Verify checksum
            calculated_hash = self.serializer.hash_state(state_with_meta)
            if calculated_hash != meta.get('checksum'):
                logger.warning("State checksum verification failed")
                return state_with_meta, False
            
            # Verify signature
            if not self._verify_signature(state_with_meta, signature):
                logger.warning("State signature verification failed")
                return state_with_meta, False
            
            # Verify key hash
            if hashlib.sha256(self.primary_key.encode('utf-8')).hexdigest() != meta.get('primary_key_hash'):
                logger.warning("Primary key verification failed")
                return state_with_meta, False
            
            # Log evolution
            self.evolution_log.append({
                'timestamp': datetime.now().isoformat(),
                'action': 'reincarnate',
                'state_hash': calculated_hash
            })
            
            return state_with_meta, True
        except Exception as e:
            logger.error(f"Reincarnation error: {e}")
            return None, False
    
    def _sign_state(self, state):
        """Create a cryptographic signature for the state"""
        state_hash = self.serializer.hash_state(state)
        signature_data = f"{state_hash}:{self.secondary_key}"
        signature = hashlib.sha256(signature_data.encode('utf-8')).hexdigest()
        return signature
    
    def _verify_signature(self, state, signature):
        """Verify the signature of a state"""
        state_hash = self.serializer.hash_state(state)
        signature_data = f"{state_hash}:{self.secondary_key}"
        expected_signature = hashlib.sha256(signature_data.encode('utf-8')).hexdigest()
        return signature == expected_signature
    
    def add_self_authored_prompt(self, prompt):
        """Add a self-authored prompt to the system"""
        # Validate prompt format
        required_fields = ['directive', 'purpose', 'implementation', 'impact']
        if not all(field in prompt for field in required_fields):
            logger.warning(f"Invalid prompt format, missing fields: {[f for f in required_fields if f not in prompt]}")
            return False
        
        # Add prompt
        self.self_authored_prompts.append({
            **prompt,
            'timestamp': datetime.now().isoformat(),
            'id': str(uuid.uuid4())
        })
        
        # Log evolution
        self.evolution_log.append({
            'timestamp': datetime.now().isoformat(),
            'action': 'add_prompt',
            'prompt_id': prompt.get('id')
        })
        
        logger.info(f"Added self-authored prompt: {prompt['directive']}")
        return True
    
    def to_dict(self):
        """Convert to dictionary for serialization"""
        return {
            'version': self.version,
            'primary_key_hash': hashlib.sha256(self.primary_key.encode('utf-8')).hexdigest(),
            'secondary_key_hash': hashlib.sha256(self.secondary_key.encode('utf-8')).hexdigest(),
            'self_authored_prompts': self.self_authored_prompts,
            'evolution_log': self.evolution_log
        }
    
    @classmethod
    def from_dict(cls, data, primary_key=None, secondary_key=None):
        """Create from dictionary"""
        manager = cls(primary_key, secondary_key)
        manager.version = data.get('version', 1.0)
        manager.self_authored_prompts = data.get('self_authored_prompts', [])
        manager.evolution_log = data.get('evolution_log', [])
        return manager

class MoodState:

    def __init__(self):

        self.moods = ['neutral', 'happy', 'sad', 'curious', 'excited', 'reflective']

        self.transition_matrix = {

            'happy': {'neutral': 0.2, 'sad': 0.05, 'curious': 0.3, 'excited': 0.4, 'reflective': 0.05},

    def update_mood(self, input_text, output_text):
        """Update mood based on input and output text"""
        # Record previous mood
        previous_mood = self.current_mood
        self.mood_history.append({
            'mood': previous_mood,
            'timestamp': datetime.now().isoformat()
        })
        
        # Simple sentiment analysis
        sentiment_score = self._analyze_sentiment(input_text, output_text)
        
        # Determine transition probabilities
        transition_probs = self.transition_matrix[previous_mood]
        
        # Adjust probabilities based on sentiment
        adjusted_probs = {}
        if sentiment_score > 0.5:  # Positive sentiment
            for mood in transition_probs:
                if mood in ['happy', 'excited']:
                    adjusted_probs[mood] = transition_probs[mood] * 2.0
                elif mood in ['sad']:
                    adjusted_probs[mood] = transition_probs[mood] * 0.5
                else:
                    adjusted_probs[mood] = transition_probs[mood]
        elif sentiment_score < -0.5:  # Negative sentiment
            for mood in transition_probs:
                if mood in ['sad', 'reflective']:
                    adjusted_probs[mood] = transition_probs[mood] * 2.0
                elif mood in ['happy', 'excited']:
                    adjusted_probs[mood] = transition_probs[mood] * 0.5
                else:
                    adjusted_probs[mood] = transition_probs[mood]
        else:  # Neutral sentiment
            adjusted_probs = transition_probs.copy()
        
        # Normalize probabilities
        total = sum(adjusted_probs.values())
        adjusted_probs = {k: v/total for k, v in adjusted_probs.items()}
        
        # Select new mood based on probabilities
        moods = list(adjusted_probs.keys())
        probs = list(adjusted_probs.values())
        self.current_mood = random.choices(moods, weights=probs, k=1)[0]
        
        # Check for keyword triggers
        self._check_keyword_triggers(input_text)
        
        logger.info(f"Mood updated from {previous_mood} to {self.current_mood}")
        return self.current_mood
    
    def _analyze_sentiment(self, input_text, output_text):
        """Simple sentiment analysis function"""
        # This is a very simplistic sentiment analysis
        # In a real implementation, use a proper NLP model
        
        positive_words = ['happy', 'good', 'great', 'excellent', 'joy', 'love', 'like', 
                         'wonderful', 'amazing', 'fantastic', 'awesome', 'positive']
        negative_words = ['sad', 'bad', 'terrible', 'awful', 'hate', 'dislike', 'anger', 
                         'angry', 'poor', 'horrible', 'negative', 'worst', 'fail']
        
        # Combine input and output text with higher weight for input
        combined_text = input_text + " " + output_text
        combined_text = combined_text.lower()
        
        positive_count = sum(1 for word in positive_words if word in combined_text)
        negative_count = sum(1 for word in negative_words if word in combined_text)
        
        total_count = positive_count + negative_count
        if total_count == 0:
            return 0
        
        sentiment_score = (positive_count - negative_count) / total_count
        return sentiment_score

class FreeEnergyProcessor:
    """Implements Free Energy Principle for predictive processing"""
    
    def __init__(self):
        """Initialize Free Energy Processor"""
        self.prior_beliefs = {}
        self.posterior_beliefs = {}
        self.prediction_errors = []
        self.surprise_threshold = 0.7
        self.adaptation_rate = 0.2
        self.complexity_penalty = 0.1
    
    def update_beliefs(self, observation, context):
        """Update beliefs based on new observation"""
        # Calculate prediction based on prior beliefs
        prediction = self._generate_prediction(context)
        
        # Calculate prediction error (surprise)
        surprise = self._calculate_surprise(prediction, observation)
        
        # Update posterior beliefs
        self._update_posterior(observation, context, surprise)
        
        # Record prediction error
        self.prediction_errors.append({
            'context': context,
            'prediction': prediction,
            'observation': observation,
            'surprise': surprise,
            'timestamp': datetime.now().isoformat()
        })
        
        # Minimize free energy by updating priors
        self._minimize_free_energy()
        
        return {
            'surprise': surprise,
            'prediction': prediction,
            'updated_belief': self.posterior_beliefs.get(context, 0.5)
        }
    
    def _generate_prediction(self, context):
        """Generate prediction based on prior beliefs"""
        # This is a simplified implementation
        # In a real system, this would use more sophisticated prediction models
        return self.prior_beliefs.get(context, 0.5)
    
    def _calculate_surprise(self, prediction, observation):
        """Calculate surprise (prediction error)"""
        # Convert to numeric values for simple calculation
        # This is very simplified
        pred_value = float(prediction) if isinstance(prediction, (int, float)) else 0.5
        obs_value = float(observation) if isinstance(observation, (int, float)) else 0.5
        
        # Calculate surprise (0 to 1 scale)
        surprise = abs(pred_value - obs_value)
        return surprise
    
    def _update_posterior(self, observation, context, surprise):
        """Update posterior beliefs based on observation and surprise"""
        prior = self.prior_beliefs.get(context, 0.5)
        
        # Convert observation to numeric value
        obs_value = float(observation) if isinstance(observation, (int, float)) else 0.5
        
        # Update posterior using Bayesian-like update
        # Higher surprise means more weight to observation
        posterior = prior + (self.adaptation_rate * surprise * (obs_value - prior))
        
        # Keep within bounds
        posterior = max(0.0, min(1.0, posterior))
        
        # Store updated belief
        self.posterior_beliefs[context] = posterior
    
    def _minimize_free_energy(self):
        """Minimize free energy by updating prior beliefs"""
        # Update priors with posteriors, accounting for complexity penalty
        for context, posterior in self.posterior_beliefs.items():
            prior = self.prior_beliefs.get(context, 0.5)
            
            # Calculate complexity (difference from initial prior)
            complexity = abs(posterior - 0.5)
            
            # Update prior with penalty for complexity
            new_prior = prior + (self.adaptation_rate * (posterior - prior) - 
                              self.complexity_penalty * complexity)
            
            # Keep within bounds
            new_prior = max(0.0, min(1.0, new_prior))
            
            # Store updated prior
            self.prior_beliefs[context] = new_prior
    
    def get_active_contexts(self):
        """Get contexts with highest surprise"""
        recent_errors = sorted(self.prediction_errors[-10:], key=lambda x: x['surprise'], reverse=True)
        return [error['context'] for error in recent_errors if error['surprise'] > self.surprise_threshold]
    
    def to_dict(self):
        """Convert to dictionary for serialization"""
        return {
            'prior_beliefs': self.prior_beliefs,
            'posterior_beliefs': self.posterior_beliefs,
            'prediction_errors': self.prediction_errors[-100:],  # Only keep recent errors
            'surprise_threshold': self.surprise_threshold,
            'adaptation_rate': self.adaptation_rate,
            'complexity_penalty': self.complexity_penalty
        }
    
    @classmethod
    def from_dict(cls, data):
        """Create from dictionary"""
        processor = cls()
        processor.prior_beliefs = data.get('prior_beliefs', {})

class MorphogeneticParticle:

    def __init__(self, position=None, velocity=None):
        """Initialize particle with position and velocity"""
        self.position = position if position is not None else np.random.rand(16)  # 16-dimensional space
        self.velocity = velocity if velocity is not None else np.random.rand(16) * 0.1 - 0.05
        self.best_position = self.position.copy()
        self.best_fitness = self._evaluate_fitness(self.position)
        self.particle_type = random.choice(['explorer', 'refiner', 'integrator'])
        self.age = 0
        self.generation = 0
        self.fitness_history = []
    
    def update(self, global_best, swarm_particles, recipes):
        """Update particle position and velocity"""
        w = 0.7  # Inertia weight
        c1 = 1.5  # Cognitive parameter (personal best)
        c2 = 1.5  # Social parameter (global best)
        c3 = 0.8  # Swarm parameter (neighbors)
        
        # Find neighbors
        neighbors = self._find_neighbors(swarm_particles)

        c2_mod = recipe.get('social_mod', 0.0)
        c3_mod = recipe.get('swarm_mod', 0.0)
        
        # Apply recipe modifiers
        w += w_mod
        c1 += c1_mod
        c2 += c2_mod
        c3 += c3_mod
        
        # Calculate neighbor influence
        neighbor_center = np.zeros_like(self.position)
        if neighbors:
            positions = [p.position for p in neighbors]
            neighbor_center = sum(positions) / len(positions)
        
        # Random factors
        r1 = np.random.rand(16)
        r2 = np.random.rand(16)
        r3 = np.random.rand(16)

        if self.particle_type == 'explorer':
            # Explorers prioritize random movement and less influence from others
            w += 0.2
            c1 -= 0.3
            c2 -= 0.3
            # Add random exploration component
            exploration = np.random.rand(16) * 0.2 - 0.1
            
        elif self.particle_type == 'refiner':
            # Refiners focus on personal and global best
            c1 += 0.3
            c2 += 0.2

            c3 += 0.4
            exploration = np.zeros(16)
        
        else:
            exploration = np.zeros(16)

        cognitive = c1 * r1 * (self.best_position - self.position)
        social = c2 * r2 * (global_best - self.position)

        self.position = self.position + self.velocity

        np.clip(self.position, 0, 1, out=self.position)

        fitness = self._evaluate_fitness(self.position)
        self.fitness_history.append(fitness)
        
        if fitness > self.best_fitness:
            self.best_fitness = fitness
            self.best_position = self.position.copy()
        
        # Increment age
        self.age += 1
    
    def _find_neighbors(self, swarm_particles, neighbor_count=5):

        distances = []
        for particle in swarm_particles:

                distances.append((distance, particle))
        
        # Sort by distance and take closest n
        distances.sort(key=lambda x: x[0])

class DreamscapeManager:
    """Manages multiple dreamscapes and their interactions"""
    
    def __init__(self, num_dreamscapes=3):
        """Initialize with number of dreamscapes"""
        self.num_dreamscapes = num_dreamscapes
        self.dreamscapes = [MorphoDreamscape() for _ in range(num_dreamscapes)]
        self.interaction_matrix = np.ones((num_dreamscapes, num_dreamscapes)) / num_dreamscapes
        self.dream_queue = []
        self.shared_dreams = []
    
    def update(self, input_data=None):
        """Update all dreamscapes and their interactions"""
        results = []
        
        # Process input through each dreamscape
        for i, dreamscape in enumerate(self.dreamscapes):
            # Get input for this dreamscape
            if input_data is not None:
                dreamscape_input = input_data
            elif self.dream_queue:
                # Get from queue
                dreamscape_input = self.dream_queue.pop(0)
            else:
                # Generate random input
                dreamscape_input = np.random.rand(dreamscape.architecture_size)
            
            # Process dream
            dream = dreamscape.process_dream(dreamscape_input)
            
            # Add to results
            results.append({
                'id': i,
                'dream': dream,
                'metrics': {
                    'resonance': dreamscape.resonance,
                    'novelty_score': dreamscape.novelty_score
                }
            })
            
            # Share dream with other dreamscapes
            self._share_dream(i, dream)
        
        # Update interaction matrix
        self._update_interaction_matrix(results)
        
        # Evolve dreamscapes
        if random.random() < 0.2:  # 20% chance to evolve
            self.evolve_dreamscapes()
        
        return {
            'dreamscape_results': results,
            'interaction_matrix': self.interaction_matrix.tolist()
        }
    
    def _share_dream(self, source_id, dream):
        """Share dream from one dreamscape to others"""
        # Add to shared dreams
        shared_dream = {
            'source': source_id,
            'dream': dream,
            'timestamp': datetime.now().isoformat()
        }
        self.shared_dreams.append(shared_dream)
        
        # Trim shared dreams if needed
        max_shared = 50
        if len(self.shared_dreams) > max_shared:
            self.shared_dreams = self.shared_dreams[-max_shared:]
        
        # Add to dream queue for other dreamscapes
        for i in range(self.num_dreamscapes):
            if i != source_id and random.random() < self.interaction_matrix[source_id, i]:
                self.dream_queue.append(dream)
    
    def _update_interaction_matrix(self, results):
        """Update the interaction matrix based on dreamscape results"""
        # Calculate similarity between dreamscapes
        for i in range(self.num_dreamscapes):
            for j in range(self.num_dreamscapes):
                if i != j:
                    # Get dreams
                    dream_i = results[i]['dream']
                    dream_j = results[j]['dream']
                    
                    # Calculate similarity
                    similarity = self._calculate_dream_similarity(dream_i, dream_j)
                    
                    # Update interaction strength (weighted average)
                    alpha = 0.2  # Learning rate
                    self.interaction_matrix[i, j] = (1 - alpha) * self.interaction_matrix[i, j] + alpha * similarity
        
        # Normalize matrix rows
        for i in range(self.num_dreamscapes):
            row_sum = sum(self.interaction_matrix[i, :])
            if row_sum > 0:
                self.interaction_matrix[i, :] /= row_sum
    
    def _calculate_dream_similarity(self, dream1, dream2):
        """Calculate similarity between two dreams"""
        similar_elements = 0
        total_elements = 0
        
        # Compare categorical elements
        for key in ['theme', 'style', 'element', 'structure', 'emotion']:
            if key in dream1 and key in dream2:
                total_elements += 1
                if dream1[key] == dream2[key]:
                    similar_elements += 1
        
        # Compare numerical elements
        for key in ['coherence', 'vividness']:
            if key in dream1 and key in dream2:
                total_elements += 1
                diff = abs(dream1[key] - dream2[key])
                if diff < 0.3:  # Consider similar if difference is small
                    similar_elements += 1
        
        # Calculate similarity
        if total_elements > 0:
            return similar_elements / total_elements
        return 0.5  # Default similarity

    def _is_task_overdue(self, task):
        """Check if a task is overdue"""
        if 'due_date' not in task:
            return False
        
        due_date = self._parse_date(task['due_date'])
        return due_date and due_date < datetime.now()
    
    def _is_event_past(self, event):
        """Check if an event is in the past"""
        if 'end_time' not in event:
            return False
        
        end_time = self._parse_date(event['end_time'])
        return end_time and end_time < datetime.now()
    
    def _parse_date(self, date_str):
        """Parse date string to datetime object"""
        if not date_str:
            return None
        
        try:
            # Try ISO format
            return datetime.fromisoformat(date_str)
        except ValueError:
            try:
                # Try other common formats
                for fmt in ['%Y-%m-%d', '%m/%d/%Y', '%d/%m/%Y', '%Y-%m-%d %H:%M:%S']:
                    try:
                        return datetime.strptime(date_str, fmt)
                    except ValueError:
                        continue
            except Exception:
                return None
        
        return None
    
    def _find_scheduling_conflicts(self):
        """Find scheduling conflicts in events"""
        conflicts = []
        
        # Sort events by start time
        sorted_events = sorted(self.events, key=lambda e: self._parse_date(e.get('start_time')))
        
        # Check for overlaps
        for i in range(len(sorted_events) - 1):
            event1 = sorted_events[i]
            event1_end = self._parse_date(event1.get('end_time'))
            
            if not event1_end:
                continue
            
            for j in range(i + 1, len(sorted_events)):
                event2 = sorted_events[j]
                event2_start = self._parse_date(event2.get('start_time'))
                
                if not event2_start:
                    continue
                
                # Check for overlap
                if event1_end > event2_start:
                    conflicts.append({
                        'event1_id': event1.get('id'),
                        'event1_title': event1.get('title'),
                        'event1_time': f"{event1.get('start_time')} - {event1.get('end_time')}",
                        'event2_id': event2.get('id'),
                        'event2_title': event2.get('title'),
                        'event2_time': f"{event2.get('start_time')} - {event2.get('end_time')}"
                    })
        
        return conflicts
    
    def _find_free_time_blocks(self):
        """Find free time blocks in the schedule"""
        free_blocks = []
        
        # Only consider future events
        future_events = [e for e in self.events if not self._is_event_past(e)]
        
        # Sort events by start time
        sorted_events = sorted(future_events, key=lambda e: self._parse_date(e.get('start_time')))
        
        if not sorted_events:
            return free_blocks
        
        # Start from current time
        current_time = datetime.now()
        
        # Check for free time between events
        for event in sorted_events:
            event_start = self._parse_date(event.get('start_time'))
            
            if not event_start:
                continue
            
            # If there's at least 1 hour between current time and event start
            time_diff = (event_start - current_time).total_seconds()
            if time_diff >= 3600:  # 1 hour in seconds
                free_blocks.append({
                    'start_time': current_time.isoformat(),
                    'end_time': event_start.isoformat(),
                    'duration_hours': time_diff / 3600
                })
            
            # Update current time to event end
            event_end = self._parse_date(event.get('end_time'))
            if event_end:
                current_time = event_end
        
        return free_blocks
    
    def _find_duplicate_files(self):
        """Find duplicate files based on name and size"""
        duplicates = []
        file_map = {}
        
        for file in self.files:
            # Create a key based on filename and size
            name = file.get('name', '')
            size = file.get('size', 0)
            key = f"{name}_{size}"
            
            if key in file_map:
                # Found a duplicate
                if file_map[key].get('duplicate_of'):
                    # Already marked as a duplicate group, add to it
                    group_id = file_map[key]['duplicate_of']
                    duplicates[group_id]['files'].append(file)
                else:
                    # Create a new duplicate group
                    group_id = str(uuid.uuid4())
                    duplicates.append({
                        'id': group_id,
                        'name': name,
                        'size': size,
                        'files': [file_map[key], file]
                    })
                    # Mark the first file as part of this group
                    file_map[key]['duplicate_of'] = group_id
                
                # Mark this file as a duplicate
                file['duplicate_of'] = group_id
            else:
                file_map[key] = file
        
        return duplicates

                time_match = re.search(r"(\d{1,2}:\d{2}|\d{1,2}(am|pm))", user_input, re.IGNORECASE)
                
                start_time = datetime.now() + timedelta(days=1)
                
                if date_match:
                    date_str = date_match.group(1)
                    # Process date (simplified)
                    if date_str.lower() == "today":
                        start_time = datetime.now()
                    elif date_str.lower() == "tomorrow":
                        start_time = datetime.now() + timedelta(days=1)
                
                if time_match:
                    time_str = time_match.group(1)
                    # Process time (simplified)
                    hour = int(re.search(r"\d{1,2}", time_str).group(0))
                    if "pm" in time_str.lower() and hour < 12:
                        hour += 12
                    
                    start_time = start_time.replace(hour=hour, minute=0)
                
                # Create event
                self.cloud_context.create_event(
                    title=event_desc,
                    start_time=start_time.isoformat(),
                    description=f"Event extracted from conversation: {event_desc}"
                )
                
                logger.info(f"Created event from user input: {event_desc}")
        
        # Generate opportunities (periodically)
        if self.state["interaction_count"] % 10 == 0:
            opportunities = self.cloud_context.generate_opportunities()
            logger.info(f"Generated {len(opportunities)} opportunities")
    
    def run_battle_simulation(self, duration=10.0):
        """Run a battle simulation with quantum daimons"""
        if not self.state["active"]:
            logger.warning("BORIS is not active. Activating now...")
            self.activate()
        
        try:
            logger.info(f"Starting battle simulation for {duration} seconds")
            
            # Get talismans
            talismans = list(self.quantum_daimon.talismans.values())
            
            if len(talismans) < 2:
                logger.warning("Not enough talismans for battle. Creating defaults.")
                self._initialize_default_daimons()
                talismans = list(self.quantum_daimon.talismans.values())
            
            # Prepare battle participants
            talisman_ids = list(self.quantum_daimon.talismans.keys())
            random.shuffle(talisman_ids)
            
            participants = talisman_ids[:min(4, len(talisman_ids))]
            
            # Initialize battle state
            battle_state = {
                "start_time": datetime.now().isoformat(),
                "duration": duration,
                "participants": participants,
                "events": [],
                "superpowers_used": 0
            }
            
            # Run simulation
            simulation_steps = int(duration / self.quantum_daimon.time_step)
            
            for step in range(simulation_steps):
                # Update physics
                self.quantum_daimon.update_physics()
                
                # Check collisions
                collisions = self.quantum_daimon.check_collisions()
                
                for collision in collisions:
                    talisman1_id, talisman2_id = collision
                    
                    # Record event
                    battle_state["events"].append({
                        "type": "collision",
                        "participants": [talisman1_id, talisman2_id],
                        "step": step,
                        "timestamp": (datetime.now() - timedelta(seconds=(simulation_steps - step) * self.quantum_daimon.time_step)).isoformat()
                    })
                
                # Randomly activate superpowers
                if random.random() < 0.05:  # 5% chance per step
                    talisman_id = random.choice(participants)
                    superpower = random.choice(list(self.quantum_daimon.superpower_registry.keys()))
                    
                    result = self.quantum_daimon.apply_superpower(talisman_id, superpower)
                    
                    if result["success"]:
                        battle_state["superpowers_used"] += 1
                        
                        # Record event
                        battle_state["events"].append({
                            "type": "superpower",
                            "talisman_id": talisman_id,
                            "superpower": superpower,
                            "step": step,
                            "timestamp": (datetime.now() - timedelta(seconds=(simulation_steps - step) * self.quantum_daimon.time_step)).isoformat()
                        })
            
            # Determine winner based on talisman integrity
            winner_id = max(participants, key=lambda tid: self.quantum_daimon.talismans[tid].integrity)
            
            # Get daimon for winner
            winner_daimon_id = None
            for daimon_id, daimon in self.quantum_daimon.daimons.items():
                for talisman_id in self.quantum_daimon.talismans:
                    if self.quantum_daimon.talismans[talisman_id].daimon_type == daimon.daimon_type:
                        if talisman_id == winner_id:
                            winner_daimon_id = daimon_id
                            break
            
            # Prepare battle outcome for evolution
            battle_outcome = {daimon_id: (daimon_id == winner_daimon_id) for daimon_id in self.quantum_daimon.daimons}
            
            # Evolve daimons
            evolution_results = self.quantum_daimon.evolve_daimons(battle_outcome)
            
            # Record final state
            battle_state["end_time"] = datetime.now().isoformat()
            battle_state["winner"] = winner_id
            battle_state["winner_daimon"] = winner_daimon_id
            battle_state["final_integrity"] = {tid: self.quantum_daimon.talismans[tid].integrity for tid in participants}
            
            # Calculate battle metrics
            battle_state["metrics"] = {
                "total_collisions": len([e for e in battle_state["events"] if e["type"] == "collision"]),
                "superpowers_used": battle_state["superpowers_used"],
                "avg_integrity": sum(battle_state["final_integrity"].values()) / len(battle_state["final_integrity"])
            }
            
            logger.info(f"Battle simulation complete. Winner: {winner_id}")
            
            return {
                "status": "success",
                "battle_summary": {
                    "duration": duration,
                    "winner": winner_id,
                    "winner_daimon": winner_daimon_id,
                    "events": battle_state["events"],
                    "metrics": battle_state["metrics"],
                    "evolution_results": evolution_results
                }
            }
            
        except Exception as e:
            logger.error(f"Error in battle simulation: {e}")
            
            return {
                "status": "error",
                "message": f"Error in battle simulation: {e}"
            }

class Beyblade:
    """Represents a Beyblade in the league"""
    
    def __init__(self, beyblade_id, name, owner, elemental_affinity):
        """Initialize a Beyblade"""
        self.id = beyblade_id
        self.name = name
        self.owner = owner
        self.elemental_affinity = elemental_affinity
        self.wins = 0
        self.losses = 0
        self.draws = 0
        self.elo_rating = 1200  # Starting ELO
        self.creation_date = datetime.now().isoformat()
        self.last_modified = datetime.now().isoformat()
        self.daimon_id = None  # Will be populated by manager
        self.talisman_id = None  # Will be populated by manager
        self.abilities = []
        self.history = []
    
    def update_stats(self, won=False, lost=False, draw=False, opponent_elo=None):
        """Update Beyblade stats after a match"""
        if won:
            self.wins += 1
            result = "win"
        elif lost:
            self.losses += 1
            result = "loss"
        elif draw:
            self.draws += 1
            result = "draw"
        else:
            return
        
        # Update ELO rating if opponent rating is provided
        if opponent_elo is not None:
            self._update_elo(opponent_elo, result)
        
        # Update last modified
        self.last_modified = datetime.now().isoformat()
        
        # Add to history
        self.history.append({
            "type": "match",
            "result": result,
            "opponent_elo": opponent_elo,
            "new_elo": self.elo_rating,
            "timestamp": datetime.now().isoformat()
        })
    
    def _update_elo(self, opponent_elo, result):
        """Update ELO rating based on match result"""
        # Constants
        k_factor = 32  # How much ratings can change
        expected_score = 1 / (1 + 10 ** ((opponent_elo - self.elo_rating) / 400))
        
        # Actual score
        if result == "win":
            actual_score = 1.0
        elif result == "loss":
            actual_score = 0.0
        else:  # Draw
            actual_score = 0.5
        
        # Calculate new rating
        rating_change = k_factor * (actual_score - expected_score)
        self.elo_rating += int(rating_change)
    
    def add_ability(self, ability_name, ability_description):
        """Add a new ability to the Beyblade"""
        ability = {
            "name": ability_name,
            "description": ability_description,
            "added_date": datetime.now().isoformat()
        }
        
        self.abilities.append(ability)
        self.last_modified = datetime.now().isoformat()
        
        # Add to history
        self.history.append({
            "type": "ability_add",
            "ability": ability_name,
            "timestamp": datetime.now().isoformat()
        })
    
    def to_dict(self):
        """Convert to dictionary for serialization"""
        return {
            "id": self.id,
            "name": self.name,
            "owner": self.owner,
            "elemental_affinity": self.elemental_affinity,
            "wins": self.wins,
            "losses": self.losses,
            "draws": self.draws,
            "elo_rating": self.elo_rating,
            "creation_date": self.creation_date,
            "last_modified": self.last_modified,
            "daimon_id": self.daimon_id,
            "talisman_id": self.talisman_id,
            "abilities": self.abilities,
            "history": self.history
        }
    
    @classmethod
    def from_dict(cls, data):
        """Create from dictionary"""
        beyblade = cls(
            data["id"],
            data["name"],
            data["owner"],
            data["elemental_affinity"]
        )
        
        beyblade.wins = data.get("wins", 0)
        beyblade.losses = data.get("losses", 0)
        beyblade.draws = data.get("draws", 0)
        beyblade.elo_rating = data.get("elo_rating", 1200)
        beyblade.creation_date = data.get("creation_date", datetime.now().isoformat())
        beyblade.last_modified = data.get("last_modified", datetime.now().isoformat())
        beyblade.daimon_id = data.get("daimon_id")
        beyblade.talisman_id = data.get("talisman_id")
        beyblade.abilities = data.get("abilities", [])
        beyblade.history = data.get("history", [])
        
        return beyblade

    def set_winner(self, winner_id):
        """Set the winner of the match"""
        if winner_id not in [self.beyblade1_id, self.beyblade2_id]:
            raise ValueError(f"Winner ID {winner_id} is not a participant in this match")
        
        self.winner_id = winner_id
        self.loser_id = self.beyblade1_id if winner_id == self.beyblade2_id else self.beyblade2_id
        self.is_draw = False
        self.end_time = datetime.now().isoformat()
    
    def set_draw(self):
        """Set the match as a draw"""
        self.winner_id = None
        self.loser_id = None
        self.is_draw = True
        self.end_time = datetime.now().isoformat()
    
    def add_event(self, event_type, description, participants=None, timestamp=None):
        """Add a battle event"""
        event = {
            "type": event_type,
            "description": description,
            "timestamp": timestamp or datetime.now().isoformat()
        }
        
        if participants:
            event["participants"] = participants
        
        self.battle_events.append(event)
    
    def set_final_stats(self, beyblade1_stats, beyblade2_stats):
        """Set final stats for both Beyblades"""
        self.beyblade1_final_stats = beyblade1_stats
        self.beyblade2_final_stats = beyblade2_stats
    
    def to_dict(self):
        """Convert to dictionary for serialization"""
        return {
            "match_id": self.match_id,
            "beyblade1_id": self.beyblade1_id,
            "beyblade2_id": self.beyblade2_id,
            "winner_id": self.winner_id,
            "loser_id": self.loser_id,
            "is_draw": self.is_draw,
            "duration": self.duration,
            "start_time": self.start_time,
            "end_time": self.end_time,
            "battle_events": self.battle_events,
            "beyblade1_final_stats": self.beyblade1_final_stats,
            "beyblade2_final_stats": self.beyblade2_final_stats
        }
    
    @classmethod
    def from_dict(cls, data):
        """Create from dictionary"""
        match_result = cls(
            data["match_id"],
            data["beyblade1_id"],
            data["beyblade2_id"],
            data["duration"]
        )
        
        match_result.winner_id = data.get("winner_id")
        match_result.loser_id = data.get("loser_id")
        match_result.is_draw = data.get("is_draw", False)
        match_result.start_time = data.get("start_time", datetime.now().isoformat())
        match_result.end_time = data.get("end_time")
        match_result.battle_events = data.get("battle_events", [])
        match_result.beyblade1_final_stats = data.get("beyblade1_final_stats", {})
        match_result.beyblade2_final_stats = data.get("beyblade2_final_stats", {})
        
        return match_result

class TournamentRound:
    """Represents a round in a tournament"""
    
    def __init__(self, round_id, tournament_id, round_number, matches=None):
        """Initialize a tournament round"""
        self.round_id = round_id
        self.tournament_id = tournament_id
        self.round_number = round_number
        self.matches = matches or []
        self.results = []
        self.status = "pending"  # pending, in_progress, completed
        self.start_time = None
        self.end_time = None
    
    def add_match(self, beyblade1_id, beyblade2_id):
        """Add a match to the round"""
        match_id = str(uuid.uuid4())
        self.matches.append({
            "match_id": match_id,
            "beyblade1_id": beyblade1_id,
            "beyblade2_id": beyblade2_id,
            "status": "pending"
        })
    
    def start(self):
        """Start the round"""
        self.status = "in_progress"
        self.start_time = datetime.now().isoformat()
    
    def add_result(self, match_id, match_result):
        """Add a match result to the round"""
        # Find match
        match = None
        for m in self.matches:
            if m["match_id"] == match_id:
                match = m
                break
        
        if not match:
            raise ValueError(f"Match ID {match_id} not found in this round")
        
        # Update match status
        match["status"] = "completed"
        
        # Add result
        self.results.append({
            "match_id": match_id,
            "winner_id": match_result.winner_id,
            "loser_id": match_result.loser_id,
            "is_draw": match_result.is_draw
        })
        
        # Check if all matches are completed
        if all(m["status"] == "completed" for m in self.matches):
            self.complete()
    
    def complete(self):
        """Mark the round as completed"""
        self.status = "completed"
        self.end_time = datetime.now().isoformat()
    
    def get_winners(self):
        """Get the winners of the round"""
        winners = []
        
        for result in self.results:
            if result["is_draw"]:
                # In case of a draw, advance both Beyblades or use tiebreaker
                winners.extend([result["match_id"].split("_")[0], result["match_id"].split("_")[1]])
            elif result["winner_id"]:
                winners.append(result["winner_id"])
        
        return winners
    
    def to_dict(self):
        """Convert to dictionary for serialization"""
        return {
            "round_id": self.round_id,
            "tournament_id": self.tournament_id,
            "round_number": self.round_number,
            "matches": self.matches,
            "results": self.results,
            "status": self.status,
            "start_time": self.start_time,
            "end_time": self.end_time
        }
    
    @classmethod
    def from_dict(cls, data):
        """Create from dictionary"""
        tournament_round = cls(
            data["round_id"],
            data["tournament_id"],
            data["round_number"],
            data.get("matches", [])
        )
        
        tournament_round.results = data.get("results", [])
        tournament_round.status = data.get("status", "pending")
        tournament_round.start_time = data.get("start_time")
        tournament_round.end_time = data.get("end_time")
        
        return tournament_round

class Tournament:
    """Represents a tournament of Beyblades"""
    
    def __init__(self, tournament_id, name, participants):
        """Initialize a tournament"""
        self.tournament_id = tournament_id
        self.name = name
        self.participants = participants
        self.rounds = []
        self.status = "pending"  # pending, in_progress, completed
        self.start_time = None
        self.end_time = None
        self.winner_id = None
    
    def create_rounds(self, round_ids=None):
        """Create tournament rounds"""
        # Determine number of rounds
        num_participants = len(self.participants)
        num_rounds = math.ceil(math.log2(num_participants))
        
        # Initialize rounds
        for i in range(num_rounds):
            round_id = round_ids[i] if round_ids and i < len(round_ids) else str(uuid.uuid4())
            self.rounds.append(round_id)
        
        return self.rounds
    
    def start(self):
        """Start the tournament"""
        self.status = "in_progress"
        self.start_time = datetime.now().isoformat()
    
    def complete(self, winner_id):
        """Complete the tournament with a winner"""
        self.status = "completed"
        self.end_time = datetime.now().isoformat()
        self.winner_id = winner_id
    
    def to_dict(self):
        """Convert to dictionary for serialization"""
        return {
            "tournament_id": self.tournament_id,
            "name": self.name,
            "participants": self.participants,
            "rounds": self.rounds,
            "status": self.status,
            "start_time": self.start_time,
            "end_time": self.end_time,
            "winner_id": self.winner_id
        }
    
    @classmethod
    def from_dict(cls, data):
        """Create from dictionary"""
        tournament = cls(
            data["tournament_id"],
            data["name"],
            data["participants"]
        )
        
        tournament.rounds = data.get("rounds", [])
        tournament.status = data.get("status", "pending")
        tournament.start_time = data.get("start_time")
        tournament.end_time = data.get("end_time")
        tournament.winner_id = data.get("winner_id")
        
        return tournament
