FUNCTION SELECT_BEST_VARIATION(variations, optimization_targets):
    best_variation = NULL
    best_score = -INFINITY

        score = EVALUATE_VARIATION(variation, optimization_targets)

FUNCTION EVALUATE_VARIATION(variation, optimization_targets):
    score = 0.0

        score = score + (value * optimization_targets[target])
